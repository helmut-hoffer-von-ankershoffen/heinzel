1"""How Heinzelmännchen is installed, and the uv commands that change the installation. 

2 

3Heinzelmännchen is meant to be installed with `uv tool install`. uv records the tool's requirements — Heinzelmännchen 

4plus every 

5`--with <plugin>` — in `uv-receipt.toml`, and `uv tool upgrade` keeps them, whereas packages put into the tool 

6environment any other way are dropped on the next upgrade (ADR 0006). So adding a plugin means reinstalling the tool 

7with one more `--with`, and updating Heinzelmännchen means reinstalling it with the same `--with`s. Outside a uv tool 

8environment (a development checkout) packages are installed with `uv pip install` instead. 

9"""

454def pin_from_receipt(entry: dict[str, Any]) -> CorePin: 

455 """What a requirement from a uv tool receipt installs from and is pinned to. 

456 

457 uv records `git+URL@ref` as `?rev=ref` whether the ref is a branch, a tag or a commit, so a ref that looks like 

458 neither a release tag nor a commit is a `ref`: a branch, or a tag that is no release. 

459 """ 

460 name = str(entry["name"]) 

461 if extras := entry.get("extras"): 

462 name = f"{name}[{','.join(extras)}]" 

463 requirement = requirement_from_receipt(entry) 

464 if "git" in entry: 

465 parts = urlsplit(str(entry["git"])) 

466 query = parse_qs(parts.query) 

467 url = urlunsplit((parts.scheme, parts.netloc, parts.path, "", "")) 

468 subdirectory = query["subdirectory"][0] if "subdirectory" in query else None 

469 ref = next((query[key][0] for key in ("rev", "tag", "branch") if key in query), None) 

470 kind: PinKind 

471 if ref is None: 

472 kind = "default" 

473 elif "branch" not in query and _is_release_tag(ref): 

474 kind = "release" 

475 elif "branch" not in query and "tag" not in query and _COMMIT.fullmatch(ref): 

476 kind = "commit" 

477 else: 

478 kind = "ref" 

479 return CorePin(kind, requirement, ref, name, url, subdirectory) 

480 for key in ("directory", "editable", "path"): 

481 if key in entry: 

482 return CorePin("local", requirement, str(Path(entry[key]).resolve()), name) 

483 if "url" in entry: 

484 return CorePin("url", requirement, str(entry["url"]), name) 

485 return CorePin("index", requirement, entry.get("specifier") or None, name)
