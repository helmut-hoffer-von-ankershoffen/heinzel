1"""The MCP head: every operation becomes a tool for coding agents (Claude Code, Codex, Cursor, …). 

2 

3The tool's input schema comes from the operation's parameters (`Annotated[T, Field(description=...)]`), its 

4description from the docstring, its output schema from the return model, and `read_only` from the operation's flag. 

5""" 

6 

7import inspect 

8from collections.abc import Callable 

9from typing import Any, Literal 

10 

11from mcp.server import MCPServer 

12from mcp.server.mcpserver.exceptions import ToolError 

13from mcp.types import ToolAnnotations 

14 

15from heinzel import __version__ 

16from heinzel.heads._exposure import Exposed, exposed 

17from heinzel.kernel import HeinzelError, Kernel, boot 

18 

19HEAD = "mcp" 

20INSTRUCTIONS = ( 

21 "Heinzelmännchen helps product development teams with the busywork of running a team. Tools are named after " 

22 "their command path: `system_doctor` is `hz system doctor`. Start with `system_info` or `system_doctor`; tools " 

23 "marked as not read-only change state, so ask the user before calling them unless they asked for exactly that." 

24) 

25 

26Transport = Literal["stdio", "streamable-http"] 

27 

28 

29def build(kernel: Kernel) -> MCPServer: 

30 """An MCP server with one tool per exposed operation.""" 

31 server = MCPServer(name="heinzel", title="🧝 Heinzelmännchen", version=__version__, instructions=INSTRUCTIONS) 

32 for item in exposed(kernel, HEAD): 

33 server.add_tool( 

34 tool(kernel, item), 

35 name=item.name, 

36 description=item.operation.summary, 

37 annotations=ToolAnnotations( 

38 read_only_hint=item.operation.read_only, destructive_hint=not item.operation.read_only 

39 ), 

40 ) 

41 return server 

42 

43 

44def tool(kernel: Kernel, item: Exposed) -> Callable[..., Any]: 

45 """A function with the operation's public signature that runs it through the kernel.""" 

46 

47 def run(**kwargs: Any) -> Any: # noqa: ANN401 - whatever the operation returns 

48 try: 

49 return kernel.invoke(item.operation, **kwargs) 

50 except HeinzelError as exc: 

51 raise ToolError(str(exc)) from exc 

52 

53 run.__signature__ = inspect.Signature( # type: ignore[attr-defined] 

54 [p.replace(kind=inspect.Parameter.KEYWORD_ONLY) for p in item.parameters], return_annotation=item.returns 

55 ) 

56 run.__doc__ = inspect.getdoc(item.operation.func) 

57 run.__name__ = item.name 

58 return run 

59 

60 

61def serve(transport: Transport = "stdio", host: str = "127.0.0.1", port: int = 8765) -> None: 

62 """Boot the kernel and serve its operations over `transport` until interrupted.""" 

63 server = build(boot()) 

64 if transport == "stdio": 

65 server.run("stdio") 

66 else: 

67 server.run("streamable-http", host=host, port=port)
