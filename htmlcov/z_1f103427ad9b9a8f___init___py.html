1"""The MCP head: every operation becomes a tool for MCP clients such as coding agents (which can also use the CLI). 

2 

3The tool's input schema comes from the operation's parameters (`Annotated[T, Field(description=...)]`), its 

4description from the docstring, its output schema from the return model, and `read_only` from the operation's flag. 

5 

6With `chat=<session>` (`hz system serve mcp --chat <session>`, started over stdio by the agent of a chat, ADR 0045) 

7it serves the same tools to that one chat, and enforces the chat's rule in code: a read-only operation runs and 

8returns its result as JSON without email addresses; any other operation does not run. It is handed to the 

9`ActionGate` as a proposal, and the tool returns its action id: only the person's `hz chat confirm` (CLI or web head, 

10never MCP) runs it. Tools follow the MCP head's own exposure, so what a UI hides from beginners is still a tool here. 

11""" 

12

29HEAD = "mcp" 

30log = logging.getLogger(__name__) 

31INSTRUCTIONS = ( 

32 "Heinzelmännchen helps product development teams with the busywork of running a team. Tools are named after " 

33 "their command path: `system_doctor` is `hz system doctor`. Start with `system_info` or `system_doctor`; tools " 

34 "marked as not read-only change state, so ask the user before calling them unless they asked for exactly that." 

35) 

36CHAT_INSTRUCTIONS = ( 

37 "Heinzelmännchen's operations for one chat. Tools are named after their command path: `system_doctor` is " 

38 "`hz system doctor`. Read-only tools run and return JSON. Any other tool does not run: it is proposed to the " 

39 "person, who confirms or declines it in the chat, and the call returns its action id." 

40)

47def build(kernel: Kernel, transport: Transport = "stdio", *, chat: str | None = None) -> MCPServer: 

48 """An MCP server with one tool per exposed operation; for one chat session's agent when `chat` is given. 

49 

50 Over streamable HTTP, which has no authentication, the operations restricted to stdio (`mcp-stdio` in their 

51 `heads`, such as installing plugins) are left out. 

52 """ 

53 server = MCPServer( 

54 name="heinzel", 

55 title="🧝 Heinzelmännchen", 

56 version=__version__, 

57 instructions=INSTRUCTIONS if chat is None else CHAT_INSTRUCTIONS, 

58 ) 

59 items = exposed(kernel, HEAD if transport == "stdio" else MCP_HTTP) 

60 names = {item.name for item in items} 

61 for item in items: 

62 annotations = ToolAnnotations( 

63 read_only_hint=item.operation.read_only, destructive_hint=not item.operation.read_only 

64 ) 

65 server.add_tool( 

66 tool(kernel, item) if chat is None else chat_tool(kernel, item, chat), 

67 name=item.name, 

68 description=item.operation.summary, 

69 annotations=annotations, 

70 ) 

71 # Old tool names of renamed operations, for the one minor release both work (ADR 0051); a chat gets the 

72 # current names only, and a name a current tool has stays that tool's. 

73 for old in () if chat is not None else item.operation.aliases: 

74 if (alias := old.replace(" ", "_").replace("-", "_")) in names: 

75 continue 

76 names.add(alias) 

77 server.add_tool( 

78 _signed(tool(kernel, item), item, item.returns, name=alias), 

79 name=alias, 

80 description=f"Deprecated: use `{item.name}`, which this runs; goes in the next minor release. " 

81 f"{item.operation.summary}", 

82 annotations=annotations, 

83 ) 

84 return server

87def tool(kernel: Kernel, item: Exposed) -> Callable[..., Any]: 

88 """A function with the operation's public signature that runs it through the kernel.""" 

89 

90 def run(**kwargs: Any) -> Any: # noqa: ANN401 - whatever the operation returns 

91 try: 

92 return kernel.invoke(item.operation, **kwargs) 

93 except HeinzelError as exc: 

94 raise ToolError(str(exc)) from exc 

95 

96 return _signed(run, item, item.returns)

99def chat_tool(kernel: Kernel, item: Exposed, session: str) -> Callable[..., str]: 

100 """The tool of one chat session: runs a read-only operation, and only proposes any other (ADR 0045).""" 

101 

102 def run(**kwargs: Any) -> str: # noqa: ANN401 - the operation's own parameters 

103 try: 

104 if item.operation.read_only: 

105 result = kernel.invoke(item.operation, **kwargs) 

106 # Redacted before it is cut, so no address is cut into something the pattern misses. 

107 text = redact_emails(json.dumps(to_jsonable_python(result, fallback=str), ensure_ascii=False)) 

108 return text[:RESULT_LIMIT] + ("…(cut)" if len(text) > RESULT_LIMIT else "") 

109 arguments = to_jsonable_python(kwargs, fallback=str) 

110 call = ProposedCall( 

111 session=session, 

112 module=item.record.qualname, 

113 operation=item.operation.name, 

114 tool=item.name, 

115 arguments=arguments, 

116 command=command_line(item, arguments), 

117 ) 

118 action = kernel.resolve(ActionGate).propose(call) 

119 except HeinzelError as exc: 

120 raise ToolError(redact_emails(str(exc))) from exc 

121 except Exception as exc: # a bug's message may hold anything: the agent gets its type only 

122 log.exception("Chat tool %s failed", item.name) 

123 msg = f"{item.name} failed with a bug ({type(exc).__name__}); the person can see the log." 

124 raise ToolError(msg) from exc 

125 return redact_emails( 

126 f"Not run: `{call.command}` changes things, so it waits for the person's confirmation in the chat " 

127 f"(action {action}). Tell them briefly what it will do and why, then stop; don't call it again." 

128 ) 

129 

130 return _signed(run, item, str)

133def command_line(item: Exposed, arguments: dict[str, Any]) -> str: 

134 """The call as the person would type it: `hz system config set --key a.b --value 1`.""" 

135 words = ["hz", *item.path] 

136 for name, value in arguments.items(): 

137 if value is None: 137 ↛ 138line 137 didn't jump to line 138 because the condition on line 137 was never true

138 continue 

139 text = value if isinstance(value, str) else json.dumps(value, ensure_ascii=False) 

140 words += [f"--{name.replace('_', '-')}", shlex.quote(text)] 

141 return " ".join(words)

144def _signed[F: Callable[..., Any]](run: F, item: Exposed, returns: object, *, name: str | None = None) -> F: 

145 run.__signature__ = inspect.Signature( # type: ignore[attr-defined] 

146 [p.replace(kind=inspect.Parameter.KEYWORD_ONLY) for p in item.parameters], return_annotation=returns 

147 ) 

148 run.__doc__ = inspect.getdoc(item.operation.func) 

149 run.__name__ = name or item.name 

150 return run

153def serve( 

154 transport: Transport = "stdio", host: str = "127.0.0.1", port: int = 8765, *, chat: str | None = None 

155) -> None: 

156 """Boot the kernel and serve its operations over `transport` until interrupted; for one chat over stdio only. 

157 

158 Raises: 

159 HeinzelError: If `chat` is given with another transport than stdio. 

160 """ 

161 if chat is not None and transport != "stdio": 

162 msg = "A chat's tools are served over stdio only, never on a port." 

163 raise HeinzelError(msg) 

164 server = build(boot(), transport, chat=chat) 

165 if transport == "stdio": 165 ↛ 168line 165 didn't jump to line 168 because the condition on line 165 was always true

166 server.run("stdio") 

167 else: 

168 server.run("streamable-http", host=host, port=port)
