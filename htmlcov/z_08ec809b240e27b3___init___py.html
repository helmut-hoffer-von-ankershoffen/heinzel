1"""The web head: a React app (web/ in the repository) that the API head serves, started with `hz ui`. 

2 

3One origin serves both: the built app at `/`, every path the app routes falling back to its `index.html`, and the API 

4head's endpoints under `/api`, built for the `web` head. The API part keeps every check of the REST head (a loopback 

5Host on our port, no other Origin, no cross-site `Sec-Fetch-Site`) and adds a token: `hz ui` makes one per session and 

6opens the browser with it in the URL's fragment, which never reaches a server or a Referer; the app keeps it in 

7`sessionStorage` and sends it as `Authorization: Bearer <token>` with every call. Requests without it are refused, so 

8another page in the browser can't call operations even when it can reach the port. Every response carries a strict 

9Content Security Policy that allows this origin only (ADR 0039). 

10 

11`hz ui` listens on one fixed port (`[hz] ui_port`), so an installed app keeps its origin, and with its session token 

12the app pairs once for a device token it keeps, which every later `hz ui` accepts too (`_devices.py`, ADR 0046). The 

13app's manifest and service worker are top-level files of the build; the worker shows notifications and caches the 

14app's shell, never API data. 

15 

16Themes' font files are served outside `/api`, at `/theme-fonts/<theme>/<index>.woff2`, without the token: the browser 

17fetches a font for `FontFace` or `@font-face` itself and can't add a header. They pass the same Host, Origin and 

18`Sec-Fetch-Site` checks, hold nothing private (what an installed plugin ships), and come through `ThemeFonts`, which 

19checks each file again as it reads it (ADR 0044). Themes' logos are served the same way at 

20`/theme-assets/<theme>/<light|dark|mark>.svg`, as `image/svg+xml` under a CSP of their own that loads and runs nothing. 

21Soundscapes' audio files are served the same way at `/sound-assets/<name>/<layer-n|earcon-kind>.<ext>`, typed by their 

22checked signature (ADR 0058). The page itself (`index.html`) admits only fixed media frame origins from code: 

23a stream origin only while the soundscape in effect is a stream. The 

24configuration is read for each page, and any error admits no stream origin. 

25 

26`POST /api/restart` restarts the server in place, for changes a running process can't load, such as an added plugin 

27(`restart_required`, ADR 0047): the process executes `hz` again with the same arguments, handing over its listening 

28socket and its token, so the open tab keeps its URL and its token and only waits a moment. 

29 

30Ctrl-C stops `hz ui` at once: the notice stream and the terminals end before the server waits for them 

31(`_shutdown.py`). 

32"""

311def page_csp(kernel: Kernel) -> str: 

312 """The page CSP admits only fixed origins needed by currently installed capabilities. 

313 

314 A streaming frame is available only while a known stream is in effect. No 

315 origin comes from plugin, user, or configuration input. 

316 """ 

317 origins: list[str] = [] 

318 try: 

319 provider = kernel.resolve(SoundAssets).stream_provider() if kernel.provides(SoundAssets) else None 

320 except Exception: # noqa: BLE001 - fail closed: the page loads with the strict CSP, sound just can't stream 

321 provider = None 

322 stream = STREAM_PROVIDERS.get(provider) if provider else None 

323 if stream: 

324 origins.append(stream.frame_origin) 

325 origins.extend(stream.additional_frame_origins) 

326 return f"{CSP}; frame-src {' '.join(dict.fromkeys(origins))}" if origins else CSP

329def secure(response: Response, path: str) -> None: 

330 """Add the security headers to the answer for `path`, and what the answers of some paths need besides. 

331 

332 The page's own CSP (`page_csp`) is kept: only it may admit a frame, and only by extending `CSP`. 

333 """ 

334 own = response.headers.get(CSP_HEADER) 

335 response.headers.update(SECURITY_HEADERS) 

336 if ( 

337 own 

338 and own.startswith(CSP + "; frame-src ") 

339 and response.headers.get("content-type", "").startswith("text/html") 

340 ): 

341 response.headers[CSP_HEADER] = own 

342 if response.status_code != OK: 

343 return 

344 if path.startswith(SOUND_ROUTE + "/"): 

345 response.headers[CSP_HEADER] = SOUND_CSP # an audio file opened on its own loads and runs nothing 

346 if path.startswith(LOGO_ROUTE + "/"): 

347 response.headers["Content-Security-Policy"] = LOGO_CSP # an SVG opened on its own loads and runs nothing 

348 if path.startswith("/assets/"): 

349 response.headers["Cache-Control"] = IMMUTABLE # file names carry their hash

362 """Font `<index>.woff2` of the available theme `theme`, or 404; served without the token (ADR 0044).""" 

363 # Checked against fixed patterns, then looked up by theme name and index: a request never builds a path.

368 content = kernel.resolve(ThemeFonts).font(theme, int(index.group(1))) 

369 except HeinzelError: # system.theme can't be built (a module it needs is off): no theme ships fonts
