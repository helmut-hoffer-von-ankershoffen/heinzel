1"""A small client for Atlassian Cloud's REST APIs: Basic auth, JSON, and errors people can act on. 

2 

3Every failure becomes a `HeinzelError` that says what happened and what to do. Messages never contain the token or the 

4Basic credential built from it: headers are never part of a message, and Atlassian's own message text is redacted in 

5case it echoes either. Every string taken from an answer passes `clean` first. 

6"""

21_CONTROLS = re.compile(r"[\x00-\x08\x0b-\x1f\x7f-\x9f\u202a-\u202e\u2066-\u2069]") 

22MESSAGE_CHARS = 300 

23"""Characters of Atlassian's own error text shown, cut after redaction so no part of a secret survives the cut.""" 

24TOKEN_FIX = "hz system credential set atlassian-token" # noqa: S105 - a command, not a secret 

25SCOPED = ( 

26 "a token created with scopes only works through api.atlassian.com, which this adapter doesn't use yet: " 

27 "create one without scopes" 

28)

31def clean(text: str) -> str: 

32 """`text` without control characters (C0 but newline and tab, DEL, C1) and bidirectional overrides. 

33 

34 Text from Atlassian ends up in terminals and in agents' prompts; whoever can edit an issue or a page could 

35 otherwise hide or reorder what they show. 

36 """ 

37 return _CONTROLS.sub("", text)

47class AtlassianClient: 

48 """GETs JSON objects from one Atlassian Cloud site as one account.""" 

49 

50 def __init__(self, site_url: str, email: str, token: str, transport: Transport, timeout: float) -> None: 

51 """Talk to `site_url` (no trailing slash) as `email` with `token`, through `transport`.""" 

52 self._site_url = site_url 

53 pair = f"{email}:{token}" 

54 self._basic = base64.b64encode(pair.encode()).decode("ascii") 

55 self._secrets = (self._basic, pair, token) 

56 self._transport = transport 

57 self._timeout = timeout 

58 

59 def get(self, product: str, path: str, params: Mapping[str, str], subject: str) -> dict[str, Any]: 

60 """GET `path` with `params` from the site and return the JSON object it answers. 

61 

62 Args: 

63 product: `Jira` or `Confluence`, for messages. 

64 path: Absolute path on the site, e.g. `/rest/api/3/search/jql`. 

65 params: Query parameters. 

66 subject: What was asked for, for messages, e.g. `project WID` or `page 123`. 

67 

68 Raises: 

69 HeinzelError: If the site can't be reached, refuses, rate-limits or answers unexpectedly. 

70 """ 

71 url = f"{self._site_url}{path}?{urlencode(params)}" 

72 headers = { 

73 "Accept": "application/json", 

74 "Authorization": f"Basic {self._basic}", 

75 "User-Agent": _user_agent(), 

76 } 

77 try: 

78 response = self._transport.get(url, headers, self._timeout) 

79 except OSError as exc: 

80 reason = getattr(exc, "reason", None) or type(exc).__name__ 

81 msg = f"Could not reach {self._site_url}: {reason}. Check the network, or atlassian.site_url." 

82 raise HeinzelError(msg) from None 

83 if len(response.body) > MAX_BYTES: 

84 msg = f"{product}'s answer for {subject} is larger than {MAX_BYTES} bytes." 

85 raise HeinzelError(msg) 

86 if response.status != 200: # noqa: PLR2004 - both endpoints answer 200 OK 

87 raise HeinzelError(self._problem(product, subject, response)) 

88 data = _json(product, response.body) 

89 if not isinstance(data, dict): 

90 msg = f"{product} answered {subject} with something other than an object; is {self._site_url} an " 

91 msg += "Atlassian Cloud site?" 

92 raise HeinzelError(msg) 

93 return cast("dict[str, Any]", data) 

94 

95 def _problem(self, product: str, subject: str, response: Response) -> str: 

96 """What went wrong, for a person: never a secret, never a header value besides Retry-After.""" 

97 status, headers = response.status, response.headers 

98 said = self._redact(clean(_message(response.body)))[:MESSAGE_CHARS] # redacted whole, then cut 

99 if status == 429: # noqa: PLR2004 - Too Many Requests 

100 after = headers.get("retry-after", "") 

101 when = f"retry after {after} seconds" if after.isdigit() else "retry in a minute" 

102 return f"Atlassian's rate limit is reached for {subject}: {when}." 

103 if status == 401: # noqa: PLR2004 - Unauthorized 

104 return ( 

105 f"{product} rejected the email and token (401: {said}); check atlassian.email, replace the token with " 

106 f"`{TOKEN_FIX}`; {SCOPED}." 

107 ) 

108 if status == 403: # noqa: PLR2004 - Forbidden 

109 need = "Browse projects on it" if product == "Jira" else "View on its space" 

110 return f"{product} refused {subject} (403: {said}); the account needs the permission {need}." 

111 if status == 404: # noqa: PLR2004 - Not Found 

112 return ( 

113 f"{product} has no {subject}, or the account can't see it: check the spelling, or ask for access " 

114 f"for the account in atlassian.email." 

115 ) 

116 if status == 400: # noqa: PLR2004 - Bad Request: in Jira, an unknown or invisible project 

117 return f"{product} refused the request for {subject} (400: {said}); check that it exists and is visible." 

118 return f"{product} answered {subject} with {status}: {said}."

120 def _redact(self, text: str) -> str: 

121 for secret in self._secrets: 

122 text = text.replace(secret, "***") 

123 return text

126def _json(product: str, body: bytes) -> Any: # noqa: ANN401 - free-form JSON 

127 try: 

128 return json.loads(body) 

129 except ValueError: 

130 msg = f"{product}'s answer is not JSON; is atlassian.site_url an Atlassian Cloud site?" 

131 raise HeinzelError(msg) from None

134def _message(body: bytes) -> str: 

135 """Atlassian's own words from an error body, whole, or a placeholder; never the raw body, which could be anything. 

136 

137 Jira says `{"errorMessages": [...], "errors": {...}}`, Confluence v2 `{"errors": [{"title": ...}]}`, older 

138 Confluence `{"message": ...}`. 

139 """ 

140 try: 

141 data = json.loads(body) 

142 except ValueError: 

143 return "no details" 

144 if not isinstance(data, dict): 

145 return "no details" 

146 found: list[str] = [] 

147 payload = cast("dict[str, Any]", data) 

148 messages = payload.get("errorMessages") 

149 if isinstance(messages, list): 

150 found += [str(m) for m in cast("list[object]", messages)] 

151 errors = payload.get("errors") 

152 if isinstance(errors, dict): 

153 found += [str(m) for m in cast("dict[str, object]", errors).values()] 

154 elif isinstance(errors, list): 

155 for error in cast("list[object]", errors): 

156 if isinstance(error, dict): 156 ↛ 155line 156 didn't jump to line 155 because the condition on line 156 was always true

157 entry = cast("dict[str, Any]", error) 

158 found.append(str(entry.get("detail") or entry.get("title") or "")) 

159 if payload.get("message"): 

160 found.append(str(payload["message"])) 

161 return "; ".join(m for m in found if m) or "no details"

164def missing(settings: AtlassianSettings, credentials: Credentials) -> list[tuple[str, str]]: 

165 """What the adapters still need before they can send a request, each with the command that provides it.""" 

166 needs: list[tuple[str, str]] = [] 

167 if settings.site_url is None: 

168 needs.append(("atlassian.site_url", f"hz system config set atlassian.site_url {SITE_EXAMPLE}")) 

169 if settings.email is None: 

170 needs.append(("atlassian.email", "hz system config set atlassian.email <your Atlassian account email>")) 

171 if not credentials.has(TOKEN): 

172 needs.append(("the atlassian-token credential", TOKEN_FIX)) 

173 return needs

176def connect(settings: AtlassianSettings, credentials: Credentials, transport: Transport) -> AtlassianClient: 

177 """A client for the configured site and account. 

178 

179 Raises: 

180 HeinzelError: Before any request, if the site, the email or the token is missing; it names the commands. 

181 """ 

182 token = credentials.get(TOKEN) 

183 needs = missing(settings, credentials) 

184 if needs or settings.site_url is None or settings.email is None or not token: 

185 what = ", ".join(name for name, _ in needs) or "the atlassian-token credential" 

186 fixes = "; ".join(fix for _, fix in needs) or TOKEN_FIX 

187 msg = f"The Atlassian adapter needs {what}; there is no anonymous access. Run: {fixes}" 

188 raise HeinzelError(msg) 

189 return AtlassianClient(settings.site_url, settings.email, token, transport, settings.timeout)
