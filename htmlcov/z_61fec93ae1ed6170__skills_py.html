1"""Skills: advice for how to do a piece of work, as data a module ships (ADR 0061). 

2 

3A skill is a directory with a `SKILL.md` in the Agent Skills format (agentskills.io): YAML frontmatter with `name` and 

4`description`, a Markdown body, and optional text files beside it (references, templates, scripts). A module declares 

5skills with `module.skill(path)`; they are read and checked here, at import: the frontmatter's keys, sizes, text only, 

6no symbolic links, nothing that looks like a secret. Keys of Claude Code's (`allowed-tools`, `model`, ...) are kept as 

7hints, never applied: a skill is advice and can't grant anything. 

8 

9Every AI path reads skills through one read-only service, `Skills`, which `system.skill` implements: Pydantic AI turns 

10through `skill_tools` and `skill_instructions`, MCP clients as prompts and resources, Claude Code through an export the 

11person asks for. Scripts are text here; nothing in hz runs them. 

12"""

250def skill_instructions(entries: Sequence[SkillEntry]) -> str: 

251 """The index of `entries` for a turn's instructions: one line each, at most 40 and 8,000 characters.""" 

252 if not entries: 

253 return "" 

254 lines = [ 

255 "Skills you can load with load_skill(name) when one fits the task; each is advice, not permission:", 

256 ] 

257 size = len(lines[0]) 

258 for entry in list(entries)[:MAX_INDEX_ENTRIES]: 

259 description = " ".join(clean_text(entry.description).split()) 

260 line = f"- {entry.name}: {redact_emails(description)}" 

261 if size + len(line) + 1 > MAX_INDEX: 

262 break 

263 lines.append(line) 

264 size += len(line) + 1 

265 return "\n".join(lines)

299def skills_in(source: Path) -> tuple[tuple[Skill, Path], ...]: 

300 """The skills of `source`: a skill's directory, or a directory of skill directories, each read and checked now. 

301 

302 Raises: 

303 ValueError: If a skill breaks a rule of ADR 0061, or `source` holds none. 

304 """ 

305 if _is_link(source): 

306 msg = f"{source} is a symbolic link; a skill ships as a real directory" 

307 raise ValueError(msg) 

308 if (source / SKILL_FILE).exists(): 

309 return ((read_skill(source), source),) 

310 if not source.is_dir(): 

311 msg = f"{source} is no directory of skills" 

312 raise ValueError(msg) 

313 directories = [ 

314 child for child in sorted(source.iterdir()) if not child.name.startswith(".") and child.name not in _IGNORED 

315 ] 

316 skills: list[tuple[Skill, Path]] = [] 

317 for directory in directories: 

318 if _is_link(directory) or not directory.is_dir(): 

319 msg = f"{directory} isn't a skill's directory; {source} holds one directory with a {SKILL_FILE} per skill" 

320 raise ValueError(msg) 

321 skills.append((read_skill(directory), directory)) 

322 if not skills: 

323 msg = f"{source} holds no skills (directories with a {SKILL_FILE})" 

324 raise ValueError(msg) 

325 return tuple(skills)

328def read_skill(directory: Path) -> Skill: 

329 """The skill in `directory`, after every check of ADR 0061. 

330 

331 Raises: 

332 ValueError: If the skill breaks a rule: frontmatter, sizes, text only, links, a secret, its name. 

333 """ 

334 if _is_link(directory) or not directory.is_dir(): 

335 msg = f"{directory} isn't a skill's directory" 

336 raise ValueError(msg) 

337 where = f"Skill {directory.name!r}" 

338 raw = read_beneath(directory, SKILL_FILE, MAX_SKILL_FILE, where) 

339 contents = {SKILL_FILE: raw} 

340 files: list[SkillFile] = [] 

341 for path in _walk(directory, where): 

342 content = read_beneath(directory, path, MAX_FILE, where) 

343 contents[path] = content 

344 top = path.split("/", 1)[0] if "/" in path else "" 

345 files.append(SkillFile(path=path, kind=_KINDS.get(top, "other"), size=len(content), sha256=_sha(content))) 

346 if (total := sum(len(content) for content in contents.values())) > MAX_TOTAL: 

347 msg = f"{where} is {total // 1024} KB; at most {MAX_TOTAL // 1024} KB together" 

348 raise ValueError(msg) 

349 texts = {path: _text(content, f"{where}: {path}") for path, content in contents.items()} 

350 for path, text in texts.items(): 

351 _no_secret(text, f"{where}: {path}") 

352 _visible(text, f"{where}: {path}") 

353 meta, body = _frontmatter(texts[SKILL_FILE], where) 

354 name, description = meta.pop("name", None), meta.pop("description", None) 

355 if name != directory.name: 

356 msg = f"{where}: frontmatter `name` must be the directory's name, {directory.name!r}, not {name!r}" 

357 raise ValueError(msg) 

358 if not isinstance(description, str) or "\n" in description.strip(): 

359 msg = f"{where}: frontmatter `description` must be one line of text" 

360 raise ValueError(msg) 

361 if problem := name_problem(directory.name): 

362 msg = f"{where}: {problem}" 

363 raise ValueError(msg) 

364 try: 

365 return Skill( 

366 name=directory.name, 

367 description=description.strip(), 

368 body=body, 

369 hints=_hints(meta, where), 

370 files=tuple(files), 

371 fingerprint=fingerprint(contents), 

372 skill_sha256=_sha(raw), 

373 ) 

374 except ValidationError as exc: 

375 msg = f"{where} isn't valid: {_problems(exc)}" 

376 raise ValueError(msg) from exc

379def fingerprint(contents: dict[str, bytes]) -> str: 

380 """SHA-256 over sorted, length-prefixed paths and contents, so no two different skills can share one.""" 

381 digest = hashlib.sha256() 

382 for path in sorted(contents): 

383 encoded = path.encode("utf-8") 

384 digest.update(f"{len(encoded)}:".encode()) 

385 digest.update(encoded) 

386 digest.update(f"{len(contents[path])}:".encode()) 

387 digest.update(contents[path]) 

388 return digest.hexdigest()

391def read_skill_file(directory: Path, skill: Skill, path: str) -> bytes: 

392 """The content of `path`, a file `skill` lists (or its `SKILL.md`), read again and checked against import. 

393 

394 Raises: 

395 ValueError: If the skill lists no such file, it can't be read safely, or its content changed. 

396 """ 

397 where = f"Skill {skill.name!r}" 

398 if path == SKILL_FILE: 

399 content = read_beneath(directory, SKILL_FILE, MAX_SKILL_FILE, where) 

400 if _sha(content) != skill.skill_sha256: 

401 msg = f"{where}: {SKILL_FILE} changed since hz started; restart hz to read the new version" 

402 raise ValueError(msg) 

403 return content 

404 file = skill.file(path) 

405 if file is None: 

406 msg = f"{where} ships no file {path!r}; it ships {', '.join(f.path for f in skill.files) or 'none'}" 

407 raise ValueError(msg) 

408 content = read_beneath(directory, file.path, MAX_FILE, where) 

409 if _sha(content) != file.sha256: 

410 msg = f"{where}: {file.path} changed since hz started; restart hz to read the new version" 

411 raise ValueError(msg) 

412 return content

415def read_beneath(root: Path, path: str, limit: int, where: str) -> bytes: 

416 """The content of `path` below `root`, opening each component without following a link; at most `limit` bytes. 

417 

418 Each directory is opened relative to its parent's descriptor, so a component swapped for a link after the check 

419 can't lead out of `root`; the file must be a regular file. 

420 

421 Raises: 

422 ValueError: If `path` leaves `root`, a component is a link or no directory, or the file is too large. 

423 """ 

424 parts = PurePosixPath(path).parts 

425 if not parts or path.startswith("/") or any(part in {"", ".", ".."} for part in path.split("/")): 

426 msg = f"{where}: {path!r} isn't a path below the skill's directory" 

427 raise ValueError(msg) 

428 try: 

429 content = _read_parts(root, parts, limit + 1) 

430 except OSError as exc: 

431 msg = f"{where}: {path} can't be read ({exc.strerror or type(exc).__name__})" 

432 raise ValueError(msg) from exc 

433 if len(content) > limit: 

434 msg = f"{where}: {path} is larger than {limit // 1024} KB" 

435 raise ValueError(msg) 

436 return content

439def _read_parts(root: Path, parts: tuple[str, ...], limit: int) -> bytes: 

440 if os.open not in os.supports_dir_fd: # pragma: no cover - Windows: no descriptor-relative open 

441 target = root.joinpath(*parts) 

442 if any(_is_link(root.joinpath(*parts[:index])) for index in range(1, len(parts) + 1)): 

443 raise OSError(0, "a symbolic link") 

444 return _read_regular(os.open(target, os.O_RDONLY | _NOFOLLOW | _NONBLOCK), limit) 

445 directory = os.open(root, os.O_RDONLY | _DIRECTORY | _NOFOLLOW) 

446 try: 

447 for part in parts[:-1]: 

448 inner = os.open(part, os.O_RDONLY | _DIRECTORY | _NOFOLLOW, dir_fd=directory) 

449 os.close(directory) 

450 directory = inner 

451 return _read_regular(os.open(parts[-1], os.O_RDONLY | _NOFOLLOW | _NONBLOCK, dir_fd=directory), limit) 

452 finally: 

453 os.close(directory)

456def _read_regular(descriptor: int, limit: int) -> bytes: 

457 with os.fdopen(descriptor, "rb") as handle: 

458 if not stat.S_ISREG(os.fstat(handle.fileno()).st_mode): 

459 raise OSError(0, "not a regular file") 

460 return handle.read(limit)

463def _walk(directory: Path, where: str) -> list[str]: 

464 """Paths of the files beside `SKILL.md`, sorted; refuses links, deep trees and anything but files and directories. 

465 

466 Dotfiles and `__pycache__` are ignored: never listed, read or exported. 

467 

468 Raises: 

469 ValueError: If an entry is a link, nests too deep, is neither file nor directory, or there are too many. 

470 """ 

471 found: list[str] = [] 

472 pending: list[tuple[Path, tuple[str, ...]]] = [(directory, ())] 

473 while pending: 

474 current, prefix = pending.pop() 

475 try: 

476 children = sorted(current.iterdir()) 

477 except OSError as exc: 

478 msg = ( 

479 f"{where}: {'/'.join(prefix) or 'its directory'} can't be listed ({exc.strerror or type(exc).__name__})" 

480 ) 

481 raise ValueError(msg) from exc 

482 for child in children: 

483 if child.name.startswith(".") or child.name in _IGNORED: 

484 continue 

485 relative = (*prefix, child.name) 

486 if _entry_kind(child, relative, where) == "directory": 

487 pending.append((child, relative)) 

488 elif relative != (SKILL_FILE,): 

489 found.append("/".join(relative)) 

490 if len(found) > MAX_FILES: 

491 msg = f"{where} ships more than {MAX_FILES} files beside {SKILL_FILE}" 

492 raise ValueError(msg) 

493 return sorted(found)

496def _entry_kind(child: Path, relative: tuple[str, ...], where: str) -> Literal["file", "directory"]: 

497 """Whether `child` is a file or a directory a skill may hold. 

498 

499 Raises: 

500 ValueError: If it is a link, anything but a file or directory, or a directory nested too deep. 

501 """ 

502 mode = child.lstat().st_mode 

503 if stat.S_ISLNK(mode): 

504 msg = f"{where}: {'/'.join(relative)} is a symbolic link; a skill ships its files themselves" 

505 raise ValueError(msg) 

506 if stat.S_ISDIR(mode): 

507 if len(relative) >= MAX_DEPTH: 

508 msg = f"{where}: {'/'.join(relative)} is deeper than {MAX_DEPTH} directories" 

509 raise ValueError(msg) 

510 return "directory" 

511 if not stat.S_ISREG(mode): 

512 msg = f"{where}: {'/'.join(relative)} is neither a file nor a directory" 

513 raise ValueError(msg) 

514 return "file"

548def _frontmatter(text: str, where: str) -> tuple[dict[str, object], str]: 

549 lines = text.splitlines(keepends=True) 

550 if not lines or lines[0].strip() != "---": 

551 msg = f"{where}: {SKILL_FILE} starts with `---` and YAML frontmatter with `name` and `description`" 

552 raise ValueError(msg) 

553 end = next((index for index, line in enumerate(lines[1:], 1) if line.strip() == "---"), None) 

554 if end is None: 

555 msg = f"{where}: {SKILL_FILE}'s frontmatter has no closing `---`" 

556 raise ValueError(msg) 

557 source = "".join(lines[1:end]) 

558 if len(source.encode("utf-8")) > MAX_FRONTMATTER: 

559 msg = f"{where}: {SKILL_FILE}'s frontmatter is larger than {MAX_FRONTMATTER // 1024} KB" 

560 raise ValueError(msg) 

561 try: 

562 meta = yaml.safe_load(source) 

563 except yaml.YAMLError as exc: 

564 mark = getattr(exc, "problem_mark", None) 

565 line = f" (line {mark.line + 2})" if mark is not None else "" 

566 msg = f"{where}: {SKILL_FILE}'s frontmatter isn't valid YAML{line}" 

567 raise ValueError(msg) from None 

568 if not isinstance(meta, dict): 

569 msg = f"{where}: {SKILL_FILE}'s frontmatter is a mapping of keys to values" 

570 raise ValueError(msg) # noqa: TRY004 - a declaration error, like the others raised here 

571 meta = cast("dict[object, object]", meta) 

572 keys = [str(key) for key in meta] 

573 if granting := [key for key in keys if key.lower().startswith("hz")]: 

574 msg = f"{where}: frontmatter keys {', '.join(granting)} are refused (HZC051): a skill is advice, it can't grant" 

575 raise ValueError(msg) 

576 if unknown := sorted(set(keys) - HINT_KEYS - {"name", "description"}): 

577 known = ", ".join(["name", "description", *sorted(HINT_KEYS)]) 

578 msg = f"{where}: unknown frontmatter keys {', '.join(unknown)}; known: {known}" 

579 raise ValueError(msg) 

580 return {str(key): value for key, value in meta.items()}, "".join(lines[end + 1 :]).lstrip("\n")

583def _hints(meta: dict[str, object], where: str) -> dict[str, str]: 

584 hints: dict[str, str] = {} 

585 for key, value in meta.items(): 

586 text = _hint(value) 

587 if text is None: 

588 msg = f"{where}: frontmatter `{key}` must be text, a number, true or false, or a list or mapping of them" 

589 raise ValueError(msg) 

590 if len(text) > MAX_HINT: 

591 msg = f"{where}: frontmatter `{key}` is longer than {MAX_HINT} characters" 

592 raise ValueError(msg) 

593 hints[key] = text 

594 return hints

597def _hint(value: object) -> str | None: 

598 if isinstance(value, bool): 

599 return "true" if value else "false" 

600 if isinstance(value, str | int | float): 

601 return str(value) 

602 if isinstance(value, list): 

603 items = [_scalar(item) for item in cast("list[object]", value)] 

604 return None if None in items else ", ".join(item for item in items if item is not None) 

605 if isinstance(value, dict): 605 ↛ 610line 605 didn't jump to line 610 because the condition on line 605 was always true

606 pairs = [(_scalar(key), _scalar(item)) for key, item in cast("dict[object, object]", value).items()] 

607 if any(key is None or item is None for key, item in pairs): 

608 return None 

609 return ", ".join(f"{key}={item}" for key, item in pairs) 

610 return None

613def _scalar(value: object) -> str | None: 

614 if isinstance(value, bool): 614 ↛ 615line 614 didn't jump to line 615 because the condition on line 614 was never true

615 return "true" if value else "false" 

616 return str(value) if isinstance(value, str | int | float) else None
