1"""The daemon in the web app's status bar: running, paused, failing, stale or not installed (ADR 0056). 

2 

3A view over `hz system daemon status`. Installing and uninstalling change what runs at every login, so they open the 

4terminal with the command and the person presses Enter (the CLI is the only head that installs, ADR 0020); pausing and 

5resuming are operations the app runs only after the person confirms the exact call. 

6"""

36def status_item(status: DaemonStatus, now: datetime | None = None) -> StatusItem: 

37 """What the status bar says about the daemon, with the buttons that change it.""" 

38 now = now or datetime.now(UTC) 

39 if status.platform not in PLATFORMS: 

40 return StatusItem( 

41 text="Daemon unsupported here", icon="bot", detail=[DetailLine(text="Needs launchd or systemd")] 

42 ) 

43 if not status.installed: 

44 return StatusItem( 

45 text="Daemon not installed", 

46 icon="bot", 

47 detail=[DetailLine(text=f"Jobs that would run on their schedule: {status.jobs}")], 

48 actions=[INSTALL], 

49 ) 

50 last = status.last_tick.astimezone(UTC) if status.last_tick else None # a naive one is local time 

51 detail = [ 

52 DetailLine(text=f"Jobs: {status.jobs}"), 

53 DetailLine(text=f"Last tick {_ago(last, now)}" if last else "No tick yet"), 

54 *(DetailLine(text=f"Failing: {job}", tone="bad") for job in status.failing[:5]), 

55 ] 

56 detail.insert( 

57 1, 

58 DetailLine( 

59 text=f"Daemon version {status.version}" 

60 if status.version 

61 else "Daemon version unknown · restart it to record its version" 

62 ), 

63 ) 

64 if status.version_mismatch: 

65 detail.insert( 

66 2, 

67 DetailLine( 

68 text=f"Daemon runs hz {status.version}, not the installed {__version__}: restart it", tone="bad" 

69 ), 

70 ) 

71 restart = [RESTART] if status.version_mismatch or status.version is None else [] 

72 interval = timedelta(minutes=status.interval_minutes or 0) 

73 stale = bool(last and interval and now - last > STALE_INTERVALS * interval) 

74 if status.paused: 

75 return StatusItem( 

76 text="Daemon paused", tone="warn", icon="bot", detail=detail, actions=[*restart, RESUME, LOGS, UNINSTALL] 

77 ) 

78 if not status.loaded or status.failing or stale: 

79 text = ( 

80 "Daemon not loaded" 

81 if not status.loaded 

82 else f"Daemon failing: {len(status.failing)} of {status.jobs} jobs" 

83 if status.failing 

84 else "Daemon not ticking" 

85 ) 

86 return StatusItem(text=text, tone="bad", icon="bot", detail=detail, actions=[*restart, LOGS, PAUSE, UNINSTALL]) 

87 upcoming = last + interval if last and interval else None 

88 return StatusItem( 

89 text="Daemon needs a restart" 

90 if status.version_mismatch 

91 else "Daemon running · next tick" 

92 if upcoming 

93 else "Daemon running", 

94 tone="warn" if status.version_mismatch else "ok", 

95 icon="bot", 

96 live=LiveTime(at=upcoming) if upcoming else None, 

97 detail=detail, 

98 actions=[*restart, LOGS, PAUSE, UNINSTALL], 

99 )
