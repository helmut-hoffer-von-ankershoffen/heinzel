1"""The Claude Code agent runner: `claude -p` headless, isolated from the user's Claude setup (ADR 0029). 

2 

3Flags as in `claude --help` of Claude Code 2.1 (checked 2026-10-24): 

4 

5- `-p` reads the prompt from standard input; `--output-format stream-json --verbose` prints one JSON event per line 

6 (the tool calls, then a `result` event with the text, usage and cost). 

7- `--safe-mode` disables CLAUDE.md, skills, plugins, hooks, MCP servers, custom commands and agents; 

8 `--restricted` ignores user, project and local settings files (so their hooks, `env` and `apiKeyHelper`), removes 

9 the tools that run commands or code unless `--tools` names them, and confines file tools to the working directory; 

10 `--disable-slash-commands` turns off skills and commands; `--strict-mcp-config` without `--mcp-config` loads no 

11 MCP server. Managed (policy) settings still apply, as an administrator intends. Not `--bare`: it allows only API 

12 keys and would break OAuth logins. 

13- `--tools` limits the built-in tools available and `--allowed-tools` pre-approves them; `--permission-mode dontAsk` 

14 denies everything else instead of prompting. 

15- A run with MCP servers in its tool profile (a chat with Heinzelmännchen's operations as tools, ADR 0045) gets them 

16 in `--mcp-config`, a file in the private temporary directory, and `--allowed-tools mcp__<name>` for each. Claude 

17 Code 2.1.310 loads no `--mcp-config` server under `--safe-mode` (checked 2026-10-24: its init event lists none), so 

18 such a run drops `--safe-mode` and passes `--setting-sources ""` instead: no user, project or local settings, so no 

19 plugins, hooks or skills from them; `--restricted`, `--strict-mcp-config` and the rest stay. 

20- `--include-partial-messages` adds `stream_event` lines with the answer's text as it is written; they go to the 

21 injected `Progress`'s emitter as `text` events, and each tool call as a `tool` event, so a head can stream them. 

22- `--append-system-prompt-file` reads the contract from a file in a private temporary directory, so it never shows 

23 in the process list; `--max-budget-usd` stops a run at its budget. There is no turn limit flag, so a `max_turns` 

24 hint can't be enforced here. 

25"""

106 def command( 

107 self, *, contract_file: Path | None, tools: ToolProfile, budget: Budget, mcp_file: Path | None = None 

108 ) -> list[str]: 

109 """The command line of a run; the prompt goes to standard input and the contract to a file. 

110 

111 Raises: 

112 HeinzelError: If the CLI is missing, or the profile asks for a tool not configured as read-only. 

113 """ 

114 try: 

115 executable = resolve(self._settings.claude_executable, self._which, "Claude Code") 

116 except HeinzelError as exc: 

117 msg = f"{exc} Install it ({DOCS}), or bind the command runner: {BIND_COMMAND_RUNNER}" 

118 raise HeinzelError(msg) from None 

119 allowed = (tools.tools or list(self._settings.read_only_tools)) if tools.read_files else [] 

120 if unsafe := [tool for tool in allowed if tool not in self._settings.read_only_tools]: 

121 msg = ( 

122 f"The tool profile asks for {', '.join(unsafe)}, which [system.runner] read_only_tools doesn't list as " 

123 f"read-only; runs never get tools that can change things." 

124 ) 

125 raise HeinzelError(msg) 

126 names = ",".join(allowed) 

127 isolation = (*(NO_SETTINGS if tools.mcp_servers else SAFE_MODE), *ISOLATION) 

128 command = [executable, "-p", "--output-format", "stream-json", "--verbose", "--include-partial-messages"] 

129 command += [*isolation, "--tools", names, "--allowed-tools", ",".join([*allowed, *servers(tools)])] 

130 if mcp_file is not None: 

131 command += ["--mcp-config", str(mcp_file)] 

132 if contract_file is not None: 

133 command += ["--append-system-prompt-file", str(contract_file)] 

134 if self._settings.model: 

135 command += ["--model", self._settings.model] 

136 if budget.max_usd is not None: 

137 command += ["--max-budget-usd", f"{budget.max_usd:g}"] 

138 return command

140 def run(self, prompt: str, *, contract: str, tools: ToolProfile, timeout: timedelta, budget: Budget) -> AgentResult: 

141 """Run `prompt` under `contract` with read-only `tools`; stop at `timeout` or `budget`. 

142 

143 Raises: 

144 HeinzelError: If the CLI is missing, the profile isn't read-only, or the working directory is missing. 

145 """ 

146 cwd = working_dir(tools) 

147 with tempfile.TemporaryDirectory(prefix="hz-run-") as directory: # 0700: only this user reads the contract 

148 contract_file = None 

149 if contract: 

150 contract_file = Path(directory) / "contract.md" 

151 contract_file.write_text(contract, encoding="utf-8") 

152 mcp_file = None 

153 if tools.mcp_servers: 

154 mcp_file = Path(directory) / "mcp.json" 

155 mcp_file.write_text(json.dumps(mcp_config(tools)), encoding="utf-8") 

156 command = self.command(contract_file=contract_file, tools=tools, budget=budget, mcp_file=mcp_file) 

157 started = time.monotonic() 

158 outcome = self._execute( 

159 command, 

160 input=prompt, 

161 cwd=cwd, 

162 env=subprocess_environment() | QUIET, 

163 timeout=timeout.total_seconds(), 

164 on_line=Relay(self._progress.emitter()), 

165 ) 

166 return interpret(outcome, timeout, time.monotonic() - started)

184class Relay: 

185 """Turns stream-json lines into `text` and `tool` events for a head to stream, as they arrive.""" 

186 

187 def __init__(self, emit: Emit) -> None: 

188 """Relay to `emit`.""" 

189 self._emit = emit 

190 

191 def __call__(self, line: str) -> None: 

192 """Emit what `line` adds: a text delta of the answer, or the name of a tool the agent calls.""" 

193 try: 

194 event = json.loads(line) 

195 except json.JSONDecodeError: 

196 return 

197 if not isinstance(event, dict): 

198 return 

199 event = cast("dict[str, Any]", event) # checked to be a dict just before 

200 if event.get("type") == "stream_event": 

201 inner = event.get("event") 

202 # Each cast follows an isinstance check: JSON objects, typed for the type checker. 

203 delta = cast("dict[str, Any]", inner).get("delta") if isinstance(inner, dict) else None 

204 if isinstance(delta, dict): 204 ↛ exitline 204 didn't return from function '__call__' because the condition on line 204 was always true

205 delta = cast("dict[str, Any]", delta) # checked to be a dict just before 

206 if delta.get("type") == "text_delta" and isinstance(delta.get("text"), str): 

207 self._emit("text", {"text": delta["text"]}) 

208 elif event.get("type") == "assistant": 208 ↛ exitline 208 didn't return from function '__call__' because the condition on line 208 was always true

209 for name in _tool_uses(event): 

210 self._emit("tool", {"name": name})

213def interpret(outcome: Outcome, timeout: timedelta, duration: float) -> AgentResult: 

214 """The result of a finished, failed or stopped `claude` process.""" 

215 result = parse_events(outcome.stdout.splitlines(), duration) 

216 if outcome.timed_out: 

217 detail = f"stopped after {timeout}, the run's timeout" 

218 return result.model_copy(update={"status": "timeout", "draft": "", "detail": detail}) 

219 cut = f"; output over {MAX_OUTPUT} bytes was cut" if outcome.truncated else "" 

220 if outcome.returncode != 0 and result.status in {"completed", "failed"}: 

221 why = result.detail or "Claude Code ended" 

222 stderr = outcome.stderr_tail() 

223 detail = f"{why} (exit code {outcome.returncode}){': ' + stderr if stderr else ''}{cut}" 

224 return result.model_copy(update={"status": "failed", "draft": "", "detail": detail}) 

225 if result.status == "completed" and not result.draft.strip(): 

226 return result.model_copy(update={"status": "failed", "detail": "Claude Code returned an empty result"}) 

227 if cut and result.status != "completed": 

228 return result.model_copy(update={"detail": result.detail + cut}) 

229 return result

232def _events(lines: Iterable[str]) -> Iterator[dict[str, Any]]: 

233 """The JSON objects among `lines`. 

234 

235 Yields: 

236 Each line that is a JSON object; anything else is skipped. 

237 """ 

238 for line in lines: 

239 try: 

240 event = json.loads(line) 

241 except json.JSONDecodeError: 

242 continue 

243 if isinstance(event, dict): 

244 yield cast("dict[str, Any]", event) # checked to be a dict just before

247def parse_events(lines: Iterable[str], duration: float) -> AgentResult: 

248 """An `AgentResult` from Claude Code's stream-json events; fields it doesn't report stay none.""" 

249 model: str | None = None 

250 calls: Counter[str] = Counter() 

251 final: dict[str, Any] | None = None 

252 api_error: str | None = None 

253 for event in _events(lines): 

254 kind = event.get("type") 

255 if kind == "system" and isinstance(event.get("model"), str): 

256 model = event["model"] 

257 elif kind == "assistant": 

258 calls.update(_tool_uses(event)) 

259 if isinstance(event.get("error"), str): 

260 api_error = event["error"] 

261 elif kind == "result": 

262 final = event 

263 if final is None: 

264 return AgentResult( 

265 runner=NAME, 

266 model=model, 

267 status="failed", 

268 tool_calls=dict(calls), 

269 duration_seconds=duration, 

270 detail="Claude Code printed no result", 

271 ) 

272 subtype = str(final.get("subtype", "")) 

273 status: RunStatus = "completed" 

274 if "budget" in subtype: 

275 status = "budget" 

276 elif final.get("is_error") is True or subtype != "success": # is_error wins: a failed login ends with success 

277 status = "failed" 

278 detail, fix = ("", "") if status == "completed" else explain(final, subtype, api_error) 

279 return AgentResult( 

280 runner=NAME, 

281 model=model, 

282 status=status, 

283 draft=str(final.get("result", "")) if status == "completed" else "", 

284 tool_calls=dict(calls), 

285 turns=whole_number(final.get("num_turns")), 

286 usage=_usage(final), 

287 duration_seconds=duration, 

288 detail=detail, 

289 fix=fix, 

290 )

317def explain(final: dict[str, Any], subtype: str, api_error: str | None) -> tuple[str, str]: 

318 """Why a run failed, in words, and the command that fixes it when there is one. 

319 

320 Claude Code reports a failed login or an API error as `is_error: true`, often with `subtype: success`, and puts 

321 what went wrong in `result` (and an `error` code on the assistant event); that text is shown, cleaned and cut. 

322 """ 

323 said = clean_text(str(final.get("result") or "")).strip() 

324 said = said if len(said) <= ERROR_TEXT else said[: ERROR_TEXT - 1] + "…" 

325 haystack = f"{api_error or ''} {said} {final.get('terminal_reason') or ''}".lower() 

326 for fragments, message, fix in KNOWN_ERRORS: 

327 if any(fragment in haystack for fragment in fragments): 

328 return (f"{message} ({said})" if said else message), fix 

329 if said: 

330 return f"Claude Code: {said}", "" 

331 return f"Claude Code ended with {api_error or subtype or 'an error'}", ""

334def _tool_uses(event: dict[str, Any]) -> list[str]: 

335 message = event.get("message") 

336 # Each cast follows an isinstance check: JSON objects and lists, typed for the type checker. 

337 content = cast("dict[str, Any]", message).get("content") if isinstance(message, dict) else None 

338 if not isinstance(content, list): 

339 return [] 

340 blocks = [cast("dict[str, Any]", block) for block in cast("list[object]", content) if isinstance(block, dict)] 

341 return [str(block["name"]) for block in blocks if block.get("type") == "tool_use" and "name" in block]

360def login_check(executable: str, execute: Execute = run_bounded) -> Check: 

361 """Whether Claude Code is logged in, from `claude auth status --json`; spends nothing. 

362 

363 Best effort: a Claude Code without `auth status`, or output it can't read, is reported as unknown, not as a 

364 problem. A login that expired on the server may still show as logged in until a run tries it; the chat then 

365 says so with the same fix. 

366 """ 

367 try: 

368 # In an empty directory, like a run: no project's Claude settings (and their helpers) come into play. 

369 with tempfile.TemporaryDirectory(prefix="hz-auth-") as directory: 

370 outcome = execute( 

371 [executable, "auth", "status", "--json"], 

372 input=None, 

373 cwd=Path(directory), 

374 env=subprocess_environment() | QUIET, 

375 timeout=LOGIN_TIMEOUT, 

376 ) 

377 except OSError as exc: 

378 return Check( 

379 name="claude-code login", 

380 status=Status.DEGRADED, 

381 detail=f"`claude auth status` didn't start: {exc}", 

382 fix=f"Reinstall Claude Code ({DOCS})", 

383 ) 

384 status: object = None 

385 try: # it exits with 1 when not logged in, so the JSON decides, not the exit code 

386 data = json.loads(outcome.stdout) 

387 except json.JSONDecodeError: 

388 data = None 

389 if isinstance(data, dict): 

390 status = cast("dict[str, Any]", data).get("loggedIn") # checked to be a dict just before 

391 if status is True: 

392 return Check(name="claude-code login", status=Status.UP, detail="logged in") 

393 if status is False: 

394 return Check( 

395 name="claude-code login", 

396 status=Status.DEGRADED, 

397 detail="Claude Code isn't logged in, so the chat and agent runs fail", 

398 fix=LOGIN, 

399 ) 

400 return Check( 

401 name="claude-code login", 

402 status=Status.UP, 

403 detail="unknown: this Claude Code has no `auth status`; a run says when it isn't logged in", 

404 )
