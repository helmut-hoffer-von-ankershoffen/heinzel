1"""`hz join <team repository>`: the front door of onboarding, and what the invite line posted in chat runs. 

2 

3The same line for everyone (the perfect-onboarding ADR). It says what hz keeps about you, finds out who you are (the 

4identity step: your gh and gws logins, this machine's clock; it asks only your role and working hours), joins the team 

5through the team module's `join` operation (a read-only clone with your own access, then you in the roster, or a pull 

6request asking its owners to add you), then runs the setup steps still missing and doctor. `hz setup` re-runs whatever 

7is still missing afterwards. 

8 

9The head knows no module: it finds the `team join` operation and the `org` settings through the kernel, like doctor. 

10Every question has a key, so `--answers`, `--yes`, `--propose`, `--role` and `--handle` answer them and a coding agent 

11runs the same line without a terminal; without one, a question nobody answered fails naming its key and its flag, and 

12nothing is changed without an explicit yes. 

13"""

104 def run(self, url: str, flags: dict[str, object], *, answers: Path | None = None, as_json: bool = False) -> None: 

105 """Say what hz keeps, find who you are, join the team, then set up what is still missing. 

106 

107 Raises: 

108 typer.Exit: With 1 when joining or a step failed, 130 after Ctrl-C. 

109 HeinzelError: Never past `setup.errors()`, which turns it into the error line and exit code 1. 

110 """ 

111 setup = SetupCommands(self.kernel, self.console, self.err_console) 

112 joined: dict[str, Any] | None = None 

113 results: list[StepResult] = [] 

114 error = "" 

115 with setup.errors(): 

116 operation = self._operation() 

117 presets = {**(load_answers(answers) if answers is not None else {}), **flags} 

118 plan = steps(self.kernel) 

119 prompter = setup.prompter(plan, presets, as_json=as_json) 

120 prompter.hints = HINTS 

121 state = SetupState.load(self.kernel.settings.state_dir) 

122 prompter.say(PRIVACY) 

123 runner = Runner(self.kernel, prompter, state, from_file=answers is not None) 

124 identity, rest = self._split(plan, SetupCommands.pending(plan, state), prompter) 

125 results = runner.run(identity, offer_skip=True) 

126 if not any(r.outcome in {Outcome.FAILED, Outcome.INTERRUPTED} for r in results): 

127 try: 

128 joined = self._join(operation, url, prompter) 

129 except KeyboardInterrupt: 

130 prompter.say("") 

131 results.append(StepResult(module="team", required=True, outcome=Outcome.INTERRUPTED)) 

132 except HeinzelError as exc: 

133 if not as_json: 

134 raise 

135 error = str(exc) 

136 results.append(StepResult(module="team", required=True, outcome=Outcome.FAILED, detail=error)) 

137 else: 

138 results += runner.run(rest, offer_skip=True) if joined.get("done") else [] 

139 self._next(prompter, url, joined) 

140 extra = {"joined": bool(joined and joined.get("done")), "team": joined, "error": error or None} 

141 if not as_json and joined is not None and not joined.get("done") and not results_failed(results): 

142 self.err_console.print( 

143 message("Not joined:", "nothing was changed; run the line again and say yes when asked."), 

144 highlight=False, 

145 soft_wrap=True, 

146 ) 

147 setup.finish(results, as_json=as_json, extra=extra)

149 def _operation(self) -> OperationLet: 

150 """The team module's `join` operation. 

151 

152 Raises: 

153 HeinzelError: If no installed module provides it. 

154 """ 

155 found = next( 

156 (let for owner, let in self.kernel.lets(OperationLet) if (owner.qualname, let.name) == TEAM_JOIN), None 

157 ) 

158 if found is None: 

159 msg = "This hz has no team module to join a team with; update hz (`uv tool upgrade heinzel`) and try again." 

160 raise HeinzelError(msg) 

161 return found

163 def _split( 

164 self, 

165 plan: list[tuple[ModuleRecord, SetupLet]], 

166 pending: list[tuple[ModuleRecord, SetupLet]], 

167 prompter: Prompter,

171 def wanted(record: ModuleRecord, let: SetupLet) -> bool: 

172 if any(key.startswith(f"{record.qualname}.") for key in prompter.presets): 

173 return True 

174 return (record, let) in pending and not is_configured(self.kernel, record) 

175 

176 fresh = [(record, let) for record, let in plan if wanted(record, let)] 

177 return [p for p in fresh if p[1].phase == "identity"], [p for p in fresh if p[1].phase != "identity"]

179 def _profile(self) -> dict[str, Any]: 

180 """The `[org]` settings, as the identity step left them on disk; empty without an org module.""" 

181 for owner, let in self.kernel.lets(SettingsLet): 

182 if owner.qualname == "org": 

183 fresh = Kernel(records=self.kernel.modules, errors=self.kernel.errors, config=Config.load()) 

184 return fresh.resolve(let.model).model_dump(mode="json") 

185 return {}

187 def _join(self, operation: OperationLet, url: str, prompter: Prompter) -> dict[str, Any]: 

188 """Join through the team module: show its plan, then carry it out on a yes, or propose you when not listed. 

189 

190 Raises: 

191 HeinzelError: If nobody is there to say yes and no yes was given in advance, or the team module refuses. 

192 """ 

193 profile = self._profile() 

194 login = profile.get("github") 

195 prompter.scope = SCOPE 

196 plan = _dump(self.kernel.invoke(operation, url=url, login=login, yes=False)) 

197 team = plan.get("team") or url 

198 prompter.say("") 

199 prompter.say(f"Joining {team}:") 

200 for action in plan.get("actions", []): 

201 prompter.say(f" {'done ' if action.get('done') else 'to do'} {action.get('what', '')}") 

202 if member := plan.get("member"): 

203 prompter.say(f"The roster lists you as {member['handle']} ({', '.join(member.get('roles', []))}).") 

204 if plan.get("done"): # nothing waited for a yes: it was your team already 

205 return plan 

206 if not self._yes(prompter, f"Make {team} your team?", "yes", what="joining", default=True): 

207 return plan 

208 return _dump(self.kernel.invoke(operation, url=url, login=login, yes=True)) 

209 prompter.say(f"The roster of {team} doesn't list you yet.") 

210 handle = profile.get("handle") 

211 if candidates := list(plan.get("candidates", [])): 211 ↛ 216line 211 didn't jump to line 216 because the condition on line 211 was always true

212 options = [*candidates, NONE_OF_THESE] 

213 mine = handle if handle in candidates else NONE_OF_THESE 

214 picked = prompter.choice("Which roster entry is you?", options, default=mine, key="handle") 

215 handle = handle if picked == NONE_OF_THESE else picked 

216 ask = f"Ask the owners of {team} to add you? This opens a pull request from your account." 

217 if not self._yes(prompter, ask, "propose", what="asking its owners to add you", default=False): 

218 return plan 

219 role = profile.get("role") or "engineer" 

220 return _dump( 

221 self.kernel.invoke(operation, url=url, login=login, handle=handle, role=role, propose=True, yes=True) 

222 )

224 @staticmethod 

225 def _yes(prompter: Prompter, question: str, key: str, *, what: str, default: bool) -> bool: 

226 """A yes or no; without a person only a yes given in advance counts, and its absence names the flag. 

227 

228 Raises: 

229 HeinzelError: If nobody can answer and no answer was given in advance. 

230 """ 

231 if not prompter.interactive and not prompter.has_preset(key): 

232 full = prompter.full_key(key) 

233 msg = f"Not joined: {what} needs a yes; {HINTS[full]} (or `{key} = true` under [{SCOPE}]). Nothing changed." 

234 raise HeinzelError(msg) 

235 return prompter.confirm(question, default=default, key=key)

238 for line in joined.get("next", []): 

239 prompter.say(str(line)) 

240 if joined.get("pull_request"): 

241 prompter.say(f"Pull request: {joined['pull_request']}")

245 prompter.say(" what is missing hz setup, any time") 

246 prompter.say(f" invite the next one post the same line: {invite_line(url)}")
