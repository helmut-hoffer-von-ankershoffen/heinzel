1"""The REST head: every operation becomes an HTTP endpoint with an OpenAPI schema (served at `/docs`). 

2 

3Read-only operations are `GET`, the others `POST`; parameters are query parameters described by their `Field`. A 

4`HeinzelError` becomes `400 {"error": ...}`; an operation whose exit code would be non-zero answers `503`, so 

5`GET /system/doctor` doubles as a health probe. 

6 

7The server has no authentication, so it answers only requests that name a loopback host (or one listed in 

8`[hz] api_allowed_hosts`) on its own port, and refuses requests from another origin with `403`: a web page in your 

9browser can reach `127.0.0.1` too, directly or through DNS rebinding, and must not be able to call operations. 

10Operations that spend money or act unattended (`hz workflow run`, `hz system daemon tick`) aren't exposed here. 

11""" 

12

33def build(kernel: Kernel, *, port: int | None = None, allowed_hosts: Iterable[str] = ()) -> FastAPI: 

34 """A FastAPI app with one endpoint per exposed operation, answering only local, same-origin requests. 

35 

36 Args: 

37 kernel: The kernel whose operations become endpoints. 

38 port: The port it listens on; a request naming another port is refused. Any port when None. 

39 allowed_hosts: Host names to answer besides localhost, 127.0.0.1 and [::1]. 

40 """ 

41 app = FastAPI( 

42 title="🧝 Heinzelmännchen", 

43 version=__version__, 

44 description="Open-source helpers that keep product teams running. Every endpoint is also a `hz` command.", 

45 ) 

46 hosts = LOOPBACK | {host.strip().lower() for host in allowed_hosts} 

47 

48 @app.middleware("http") 

49 async def local_only(request: Request, call_next: Callable[[Request], Awaitable[Response]]) -> Response: 

50 headers = request.headers 

51 if problem := refusal( 

52 headers.get("host", ""), headers.get("origin"), hosts, port, headers.get("sec-fetch-site") 

53 ): 

54 return JSONResponse(status_code=FORBIDDEN, content={"error": problem}) 

55 return await call_next(request) 

56 

57 @app.exception_handler(HeinzelError) 

58 async def heinzel_error(request: object, exc: HeinzelError) -> JSONResponse: # noqa: ARG001 - FastAPI signature 

59 return JSONResponse(status_code=400, content={"error": str(exc)}) 

60 

61 for item in exposed(kernel, HEAD): 

62 app.add_api_route( 

63 "/" + "/".join(item.path), 

64 endpoint(kernel, item), 

65 methods=["GET" if item.operation.read_only else "POST"], 

66 summary=item.operation.summary, 

67 description=inspect.getdoc(item.operation.func) or "", 

68 response_model=item.returns if item.returns is not inspect.Signature.empty else None, 

69 tags=[item.record.path[0]], 

70 operation_id=item.name, 

71 ) 

72 return app

75def refusal( 

76 host: str, origin: str | None, hosts: frozenset[str], port: int | None, fetch_site: str | None = None 

77) -> str | None: 

78 """Why a request is refused, or None: its Host must be an allowed name on our port, its Origin our own. 

79 

80 Without an Origin, a browser's `Sec-Fetch-Site: cross-site` or `same-site` still marks a request another page 

81 made. On port 80 or 443 a Host without a port is the same port. 

82 """ 

83 name, given = _split(host) 

84 default = port in DEFAULT_PORTS and given == "" 

85 if name not in hosts or (port is not None and given != str(port) and not default): 

86 where = f" on port {port}" if port is not None else "" 

87 return ( 

88 f"Refused: this server answers requests to {', '.join(sorted(hosts))}{where}, not {host!r}. If you serve " 

89 "on another interface, add its name to [hz] api_allowed_hosts in your user configuration." 

90 ) 

91 if origin is not None and origin.strip().lower() not in {f"http://{host.lower()}", f"https://{host.lower()}"}: 

92 return f"Refused: a request from another origin ({origin}) can't call operations; use the CLI or MCP." 

93 if origin is None and (fetch_site or "").strip().lower() in {"cross-site", "same-site"}: 

94 return f"Refused: a request another site made (Sec-Fetch-Site: {fetch_site}) can't call operations." 

95 return None

98def _split(host: str) -> tuple[str, str]: 

99 """A Host header's name and port; `[::1]:8000` keeps its brackets.""" 

100 host = host.strip().lower() 

101 if host.startswith("["): 

102 end = host.find("]") 

103 if end == -1: 

104 return host, "" 

105 rest = host[end + 1 :] 

106 return host[: end + 1], rest[1:] if rest.startswith(":") else "" 

107 name, _, port = host.partition(":") 

108 return name, port

111def endpoint(kernel: Kernel, item: Exposed) -> Callable[..., Any]: 

112 """A function with the operation's public parameters as query parameters, run through the kernel.""" 

113 

114 def run(response: Response, **kwargs: Any) -> Any: # noqa: ANN401 - whatever the operation returns 

115 result = kernel.invoke(item.operation, **kwargs) 

116 if item.operation.exit_code is not None and item.operation.exit_code(result): 

117 response.status_code = UNAVAILABLE 

118 return result 

119 

120 response_parameter = inspect.Parameter("response", inspect.Parameter.KEYWORD_ONLY, annotation=Response) 

121 run.__signature__ = inspect.Signature( # type: ignore[attr-defined] 

122 [_query(p) for p in item.parameters] + [response_parameter], return_annotation=item.returns 

123 ) 

124 run.__name__ = item.name 

125 return run

128def _query(parameter: inspect.Parameter) -> inspect.Parameter: 

129 annotation, description = parameter.annotation, None 

130 if get_origin(annotation) is Annotated: 130 ↛ 133line 130 didn't jump to line 133 because the condition on line 130 was always true

131 annotation, *metadata = get_args(annotation) 

132 description = next((m.description for m in metadata if isinstance(m, FieldInfo)), None) 

133 return parameter.replace( 

134 annotation=Annotated[annotation, Query(description=description)], kind=inspect.Parameter.KEYWORD_ONLY
