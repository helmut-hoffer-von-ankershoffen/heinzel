1"""The `chatgpt` option: a ChatGPT plan through OpenAI's "Sign in with ChatGPT", with Pydantic AI (ADR 0057). 

2 

3OpenAI lets open-source apps use a person's ChatGPT plan instead of an API key: the person signs in in their browser 

4(`hz ai access login`), approves hz by name, and can limit it in ChatGPT's settings; requests count toward the plan. 

5The sign-in's credentials, a refresh token among them, are kept as one credential in the keychain 

6(`ai.access/chatgpt-login`) and refreshed there; the installation's id (`<state_dir>/ai/host-id`, not a secret) lets a 

7later sign-in reuse the registration. `status` reads the keychain only; `test` sends one short request. Pydantic AI's 

8`openai-chatgpt` provider does the protocol; hz never uses the Codex CLI's stored credentials. 

9"""

92class KeychainCredentialSource: 

93 """Pydantic AI's credential source for the ChatGPT sign-in, kept in the keychain so refreshes outlive a process.

102 async def load(self) -> Any: # noqa: ANN401 - Pydantic AI's credentials, imported lazily 

103 """The stored sign-in. 

104 

105 Raises: 

106 HeinzelError: If there is none. 

107 """ 

108 from pydantic_ai.providers.openai_chatgpt import OpenAIChatGPTCredentials # noqa: PLC0415 - heavy

110 fields = _load(self._credentials) 

111 if fields is None: 

112 msg = f"Not signed in to ChatGPT: run `{LOGIN}`." 

113 raise HeinzelError(msg) 

114 known = {field.name for field in dataclasses.fields(OpenAIChatGPTCredentials)}

117 async def save(self, credentials: Any) -> None: # noqa: ANN401 - Pydantic AI's credentials 

118 """Store `credentials`, replacing the earlier sign-in; nothing without a keychain.""" 

119 if writable(self._credentials): 

120 self._credentials.set(CHATGPT_LOGIN, json.dumps(dataclasses.asdict(credentials)))

123def _without_hint(url: str) -> str: 

124 """`url` without its `id_token_hint` parameter.""" 

125 parts = urllib.parse.urlsplit(url) 

126 query = [(key, value) for key, value in urllib.parse.parse_qsl(parts.query) if key != "id_token_hint"] 

127 return urllib.parse.urlunsplit(parts._replace(query=urllib.parse.urlencode(query)))

131 """This installation's stable, opaque id for OpenAI (`urn:uuid:...`), created once in `root`. 

132 

133 Raises: 

134 HeinzelError: If it can't be kept there. 

135 """

139 if found.startswith("urn:uuid:"): 139 ↛ 143line 139 didn't jump to line 143 because the condition on line 139 was always true

140 return found 

141 except OSError: 

142 pass 

143 made = uuid.uuid4().urn 

144 try: 

145 root.mkdir(parents=True, exist_ok=True, mode=0o700) 

146 path.write_text(made, encoding="utf-8")

148 msg = f"Can't keep this installation's id for ChatGPT in {path}: {exc}" 

149 raise HeinzelError(msg) from None

174 def status(self) -> AccessStatus: 

175 """Whether OpenAI's client is installed, a sign-in with plan usage is kept and a model is set; offline.""" 

176 model = self._settings.chatgpt_model 

177 status = AccessStatus(title=TITLE, summary=SUMMARY, installed=True, models=[model] if model else []) 

178 if not self._installed("openai"): 

179 fix = "uv tool install --reinstall 'heinzel[openai]'" 

180 return status.model_copy( 

181 update={"installed": False, "detail": "OpenAI's client is not installed", "fix": fix} 

182 ) 

183 fields = _load(self._credentials) 

184 if fields is None: 

185 return status.model_copy(update={"signed_in": False, "detail": "not signed in", "fix": LOGIN}) 

186 if "chatgpt.tokens.use.direct" not in str(fields.get("scope", "")).split(): 

187 detail = "signed in, but plan usage wasn't granted to hz" 

188 return status.model_copy(update={"signed_in": False, "detail": detail, "fix": LOGIN}) 

189 if model is None: 

190 return status.model_copy( 

191 update={"signed_in": True, "detail": "signed in; no model is set", "fix": SET_MODEL} 

192 ) 

193 kept = "" if writable(self._credentials) else "; refreshed tokens last this process only (no keychain)" 

194 return status.model_copy(update={"signed_in": True, "detail": f"signed in; {model}{kept}"})

196 def login(self, progress: Progress, open_browser: Callable[[str], bool] = webbrowser.open) -> AccessStatus: 

197 """Sign in in the browser and keep the credentials in the keychain; then the option's status. 

198 

199 Raises: 

200 HeinzelError: If OpenAI's client is missing, or the sign-in fails, is declined or takes too long. 

201 """ 

202 if not self._installed("openai"): 

203 msg = "Signing in to ChatGPT needs OpenAI's client: uv tool install --reinstall 'heinzel[openai]'" 

204 raise HeinzelError(msg) 

205 if not writable(self._credentials): # before the browser: a sign-in nobody can keep would be wasted 

206 msg = ( 

207 "Signing in to ChatGPT needs a keychain to keep the sign-in, and credentials come from the " 

208 "environment here; run it where a keychain is available (`hz system credential list` says which)." 

209 ) 

210 raise HeinzelError(msg) 

211 from pydantic_ai.providers.openai_chatgpt import OpenAIChatGPTOAuthFlow # noqa: PLC0415 - heavy 

212 

213 earlier = _load(self._credentials) or {} 

214 flow = OpenAIChatGPTOAuthFlow( 

215 app_name=APP_NAME, 

216 host_id=host_id(self._root), 

217 client_id=earlier.get("client_id"), 

218 id_token_hint=earlier.get("id_token"), 

219 ) 

220 url = flow.authorization_url() 

221 # Shown only when no browser opened, and then without the earlier sign-in's id token it carries as a hint. 

222 where = "in the browser that opened" if open_browser(url) else f"at {_without_hint(url)}" 

223 minutes = int(SIGN_IN_TIMEOUT // 60) 

224 

225 async def signed_in() -> Any: # noqa: ANN401 - Pydantic AI's credentials 

226 return await asyncio.wait_for(flow.exchange_code_from_callback(), SIGN_IN_TIMEOUT) 

227 

228 try: 

229 with progress.task(f"Sign in to ChatGPT {where}; waiting up to {minutes} minutes"): 

230 credentials = self._run(signed_in()) 

231 except Exception as exc: # noqa: BLE001 - declined, timed out, or the exchange failed 

232 msg = f"Signing in to ChatGPT didn't finish: {clean_text(f'{type(exc).__name__}: {exc}')[:DETAIL_LIMIT]}" 

233 raise HeinzelError(msg) from None 

234 try: 

235 self._credentials.set(CHATGPT_LOGIN, json.dumps(dataclasses.asdict(credentials))) 

236 except HeinzelError as exc: 

237 msg = f"Signed in to ChatGPT, but keeping the sign-in failed: {exc}" 

238 raise HeinzelError(msg) from None 

239 status = self.status() 

240 if not credentials.plan_usage_granted: 

241 detail = "signed in, but plan usage wasn't granted: sign in again and allow hz to use your plan" 

242 return status.model_copy(update={"signed_in": False, "detail": detail, "fix": LOGIN}) 

243 return status

245 def test(self) -> AccessTest: 

246 """One request with a few output tokens through the plan; a failure is a result.""" 

247 found = self.status() 

248 if not found.signed_in or self._settings.chatgpt_model is None: 

249 return AccessTest(ok=False, detail=found.detail, fix=found.fix) 

250 started = time.monotonic() 

251 try: 

252 import pydantic_ai # noqa: PLC0415 - a heavy import, only on a test 

253 from pydantic_ai.usage import UsageLimits # noqa: PLC0415 

254 

255 pydantic_ai.BANNER_ENABLED = False # hz owns its output: stdout may be the MCP head's protocol 

256 model = self._build(self._settings.chatgpt_model, KeychainCredentialSource(self._credentials)) 

257 agent = pydantic_ai.Agent(model, instructions=TEST_INSTRUCTIONS) 

258 # One request, within a minute; the plan's models take no output cap, and reasoning counts as output. 

259 asked = agent.run("Are you there?", usage_limits=UsageLimits(request_limit=1)) 

260 result = self._run(asyncio.wait_for(asked, TEST_TIMEOUT)) 

261 except Exception as exc: # noqa: BLE001 - OpenAI's or the library's error is a failed check 

262 detail = clean_text(f"{type(exc).__name__}: {exc}")[:DETAIL_LIMIT] 

263 return AccessTest(ok=False, detail=detail, duration_seconds=time.monotonic() - started) 

264 answer = clean_text(str(result.output)).strip() 

265 return AccessTest( 

266 ok=bool(answer), 

267 detail=answer[:DETAIL_LIMIT] or "the model returned nothing", 

268 model=self._settings.chatgpt_model, 

269 duration_seconds=time.monotonic() - started, 

270 )
