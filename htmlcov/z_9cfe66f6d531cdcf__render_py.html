380def swatches(theme: ThemeDoc) -> str: 

381 """An SVG of the theme's characteristic colours: one row per mode, light above dark.""" 

382 tokens = [c for c in theme.colours if c.token in SWATCH_TOKENS] 

383 width = len(tokens) * SWATCH 

384 height = SWATCH * 2 

385 label = escape(f"{theme.title}: colours in light and dark mode", quote=True) 

386 box = f'width="{width}" height="{height}" viewBox="0 0 {width} {height}"' 

387 parts = [f'<svg xmlns="http://www.w3.org/2000/svg" {box} role="img" aria-label="{label}">'] 

388 parts.append(GENERATED) 

389 parts.append(f"<title>{escape(theme.title)}</title>") 

390 for row, mode in enumerate(("light", "dark")): 

391 for column, colour in enumerate(tokens): 

392 value = getattr(colour, mode) or "#808080" 

393 at = f'x="{column * SWATCH}" y="{row * SWATCH}" width="{SWATCH}" height="{SWATCH}"' 

394 title = f"<title>{escape(colour.token)} ({mode}): {escape(value)}</title>" 

395 parts.append(f'<rect {at} fill="{escape(value, quote=True)}">{title}</rect>') 

396 parts.append("</svg>") 

397 return "\n".join(parts) + "\n"

409def render_documentation(documentation: Documentation) -> RenderableType: 

410 """The model on a terminal: one table of plugins, then each plugin's operations.""" 

411 table = Table(title="Plugins documented", show_lines=False) 

412 for column in ("Plugin", "Version", "Kind", "Operations", "Settings", "Themes", "Scenes"): 

413 table.add_column(column) 

414 for doc in documentation.pages: 

415 table.add_row( 

416 doc.distribution, 

417 doc.version, 

418 ", ".join(doc.kinds) or "—", 

419 str(sum(len(m.operations) for m in doc.modules)), 

420 str(sum(len(m.settings) for m in doc.modules)), 

421 ", ".join(t.name for t in doc.themes) or "—", 

422 ", ".join(s.name for s in doc.scenes) or "—", 

423 ) 

424 parts: list[RenderableType] = [table] 

425 parts += [ 

426 Text(f" {op.command}: {op.summary}") 

427 for doc in documentation.pages 

428 for module in doc.modules 

429 for op in module.operations 

430 ] 

431 parts += [Text(f"! {warning}", style="yellow") for warning in documentation.warnings] 

432 return Group(*parts)
