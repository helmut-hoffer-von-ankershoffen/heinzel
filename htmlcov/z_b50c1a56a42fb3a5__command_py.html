1"""The generic command agent runner: any agent CLI from a configured command line (ADR 0029). 

2 

3The command line is a list of arguments with placeholders, run without a shell, from the user's configuration only. 

4It fails closed: it refuses to run until the user records that the command restricts its agent to read-only tools 

5(`command_enforces_read_only`), and passes only tools listed in `command_read_only_tools`. The prompt and the contract 

6are written to files in a private temporary directory; the result is read from stdout or from `{output_file}`, as 

7text or JSON. The environment is an allowlist. The runner enforces the timeout; the budget only if the command takes 

8`{max_usd}`. 

9"""

48def placeholders(argument: str) -> set[str]: 

49 """The placeholders an argument uses; `{{` and `}}` are literal braces. 

50 

51 Raises: 

52 HeinzelError: If the argument is not a valid template, uses a format spec or conversion (`{tools!r}`, 

53 `{model:>9}`), or names an unknown placeholder. 

54 """ 

55 try: 

56 fields = [(name, spec, conversion) for _, name, spec, conversion in string.Formatter().parse(argument)] 

57 except ValueError as exc: 

58 msg = f"[system.runner] command argument {argument!r} is not a valid template: {exc}" 

59 raise HeinzelError(msg) from exc 

60 names = {name for name, _, _ in fields if name is not None} 

61 if any(spec or conversion for name, spec, conversion in fields if name is not None): 

62 msg = f"[system.runner] command argument {argument!r} uses a format spec or conversion; write `{{name}}` only" 

63 raise HeinzelError(msg) 

64 if unknown := names - PLACEHOLDERS: 

65 msg = ( 

66 f"[system.runner] command argument {argument!r} uses unknown placeholders {sorted(unknown)}; " 

67 f"known: {', '.join(sorted(PLACEHOLDERS))}" 

68 ) 

69 raise HeinzelError(msg) 

70 return names

73def render(argv: list[str], values: Mapping[str, str]) -> list[str]: 

74 """Fill the placeholders of each argument; arguments stay separate, so nothing is ever split or interpreted. 

75 

76 Raises: 

77 HeinzelError: If an argument is not a valid template (see `placeholders`). 

78 """ 

79 rendered: list[str] = [] 

80 for argument in argv: 

81 placeholders(argument) 

82 rendered.append(argument.format_map(values)) 

83 return rendered

86def environment(names: list[str], source: Mapping[str, str] | None = None) -> dict[str, str]: 

87 """The command's environment: PATH, HOME, LANG, LC_* and TMPDIR plus `names`, never a credential variable.""" 

88 wanted = BASE_ENVIRONMENT | set(names) 

89 return { 

90 key: value for key, value in subprocess_environment(source).items() if key in wanted or key.startswith("LC_") 

91 }

94def read_output(path: Path, limit: int = MAX_OUTPUT) -> str | None: 

95 """The output file, never through a symlink, at most `limit` bytes; None if the command wrote none. 

96 

97 Raises: 

98 HeinzelError: If the file is a symlink or larger than `limit`. 

99 """ 

100 try: 

101 # O_NONBLOCK: a FIFO left at the path must not hang the read 

102 descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK) 

103 except FileNotFoundError: 

104 return None 

105 except OSError as exc: # ELOOP: a symlink, which could point anywhere the user can read 

106 msg = f"The command's output file is not a regular file ({exc.strerror}); it was not read." 

107 raise HeinzelError(msg) from None 

108 if not stat.S_ISREG(os.fstat(descriptor).st_mode): # a FIFO, a directory or a device 

109 os.close(descriptor) 

110 msg = "The command's output file is not a regular file; it was not read." 

111 raise HeinzelError(msg) 

112 with os.fdopen(descriptor, "rb") as file: 

113 data = file.read(limit + 1) 

114 if len(data) > limit: 

115 msg = f"The command's output file is larger than {limit} bytes; it was not read." 

116 raise HeinzelError(msg) 

117 return data.decode(errors="replace")

120class CommandRunner: 

121 """Runs one agent session with a configured command line."""

123 def __init__( 

124 self, 

125 settings: RunnerSettings, 

126 execute: Execute = run_bounded, 

127 which: Callable[[str], str | None] = shutil.which, 

128 ) -> None: 

129 """Create the runner; settings are injected, `execute` and `which` replace processes in tests.""" 

130 self._settings = settings 

131 self._execute = execute 

132 self._which = which

134 def _check(self, tools: ToolProfile) -> str: 

135 settings = self._settings 

136 if not settings.command: 

137 msg = f"The command runner has no command; set the agent CLI's command line, e.g.: {SET_COMMAND}" 

138 raise HeinzelError(msg) 

139 if not settings.command_enforces_read_only: 

140 msg = ( 

141 "The command runner refuses to run: nothing records that the configured command restricts its agent " 

142 f"to read-only tools. Configure that in the command, then record that it does: {TRUST_COMMAND}" 

143 ) 

144 raise HeinzelError(msg) 

145 if unsafe := [tool for tool in tools.tools if tool not in settings.command_read_only_tools]: 

146 msg = ( 

147 f"The tool profile asks for {', '.join(unsafe)}, which [system.runner] command_read_only_tools doesn't " 

148 f"list as read-only; runs never get tools that can change things." 

149 ) 

150 raise HeinzelError(msg) 

151 for argument in settings.command[1:]: 

152 placeholders(argument) 

153 return resolve(settings.command[0], self._which, "The command runner's executable")

155 def run(self, prompt: str, *, contract: str, tools: ToolProfile, timeout: timedelta, budget: Budget) -> AgentResult: 

156 """Write prompt and contract to files, run the command, read the result; stop at `timeout`. 

157 

158 Raises: 

159 HeinzelError: If the runner is not configured to run (no command, read-only not recorded, a tool not 

160 listed, a bad placeholder, a missing executable or working directory) or the output file is unsafe. 

161 """ 

162 settings = self._settings 

163 cwd = working_dir(tools) 

164 executable = self._check(tools) 

165 warnings = self._warnings(budget, tools) 

166 with tempfile.TemporaryDirectory(prefix="hz-run-") as directory: # 0700: only this user reads the prompt 

167 folder = Path(directory) 

168 (folder / "prompt.md").write_text(prompt, encoding="utf-8") 

169 (folder / "contract.md").write_text(contract, encoding="utf-8") 

170 output_file = folder / "output" 

171 (folder / "mcp.json").write_text(json.dumps(mcp_config(tools)), encoding="utf-8") 

172 values = { 

173 "prompt_file": str(folder / "prompt.md"), 

174 "contract_file": str(folder / "contract.md"), 

175 "output_file": str(output_file), 

176 "working_dir": str(cwd or Path.cwd()), 

177 "tools": ",".join(tools.tools), 

178 "model": settings.model or "", 

179 "timeout_seconds": str(int(timeout.total_seconds())), 

180 "max_usd": "" if budget.max_usd is None else f"{budget.max_usd:g}", 

181 "mcp_config": str(folder / "mcp.json"), 

182 } 

183 command = [executable, *render(settings.command[1:], values)] 

184 started = time.monotonic() 

185 outcome = self._execute( 

186 command, input=None, cwd=cwd, env=environment(settings.command_env), timeout=timeout.total_seconds() 

187 ) 

188 duration = time.monotonic() - started 

189 base: dict[str, Any] = {"runner": NAME, "model": settings.model, "duration_seconds": duration} 

190 if outcome.timed_out: 

191 detail = f"stopped after {timeout}, the run's timeout" 

192 return AgentResult(**base, status="timeout", detail=detail, warnings=warnings) 

193 if outcome.returncode != 0: 

194 detail = f"exit code {outcome.returncode}: {outcome.stderr_tail()}" 

195 return AgentResult(**base, status="failed", detail=detail, warnings=warnings) 

196 if settings.command_output == "file": 

197 try: # the agent ran: a bad output file is a failed run (it may have spent), not a misconfiguration 

198 output = read_output(output_file) 

199 except HeinzelError as exc: 

200 return AgentResult(**base, status="failed", detail=str(exc), warnings=warnings) 

201 if output is None: 

202 detail = "the command wrote no output file" 

203 return AgentResult(**base, status="failed", detail=detail, warnings=warnings) 

204 elif outcome.truncated: 

205 detail = f"the command printed more than {MAX_OUTPUT} bytes" 

206 return AgentResult(**base, status="failed", detail=detail, warnings=warnings) 

207 else: 

208 output = outcome.stdout 

209 return self._result(output, base, warnings)

211 def _warnings(self, budget: Budget, tools: ToolProfile) -> list[str]: 

212 used = set[str]().union(*(placeholders(argument) for argument in self._settings.command[1:])) 

213 warnings: list[str] = [] 

214 if budget.max_usd is not None and "max_usd" not in used: 

215 warnings.append(f"the budget of {budget.max_usd:g} USD is not passed to the command, so not enforced") 

216 if not tools.read_files: 

217 warnings.append("the command decides its agent's file tools itself; this run asked for none") 

218 if tools.mcp_servers and "mcp_config" not in used: 

219 names = ", ".join(sorted(tools.mcp_servers)) 

220 warnings.append(f"the command takes no {{mcp_config}}, so the agent had no tools of {names}") 

221 return warnings

223 def _result(self, output: str, base: dict[str, Any], warnings: list[str]) -> AgentResult: 

224 settings = self._settings 

225 if settings.command_format == "text": 

226 if not output.strip(): 

227 return AgentResult(**base, status="failed", detail="the command printed nothing", warnings=warnings) 

228 return AgentResult(**base, status="completed", draft=output.strip(), warnings=warnings) 

229 try: 

230 data = json.loads(output) 

231 except json.JSONDecodeError as exc: 

232 return AgentResult(**base, status="failed", detail=f"output is not JSON: {exc}", warnings=warnings) 

233 draft = cast("dict[str, Any]", data).get(settings.command_draft_key) if isinstance(data, dict) else None 

234 if not isinstance(draft, str) or not draft.strip(): 

235 detail = f"output has no text under {settings.command_draft_key!r}" 

236 return AgentResult(**base, status="failed", detail=detail, warnings=warnings) 

237 data = cast("dict[str, Any]", data) 

238 raw = data.get("usage") 

239 usage = cast("dict[str, Any]", raw) if isinstance(raw, dict) else {} 

240 return AgentResult( 

241 **base, 

242 status="completed", 

243 draft=draft.strip(), 

244 usage=TokenUsage( 

245 input_tokens=whole_number(usage.get("input_tokens")), 

246 output_tokens=whole_number(usage.get("output_tokens")), 

247 cost_usd=amount(data.get("total_cost_usd")), 

248 ), 

249 warnings=warnings, 

250 )
