15def render_modules(modules: list[ModuleInfo]) -> RenderableType: 

16 """A table of modules and their lets.""" 

17 table = Table(title="Modules") 

18 for column in ("Module", "Summary", "Distribution", "Lets"): 

19 table.add_column(column, overflow="fold") 

20 for info in modules: 

21 lets = "\n".join(f"{kind}: {', '.join(names)}" for kind, names in info.lets.items()) 

22 distribution = "built-in" if info.builtin else f"{info.distribution} {info.version}" 

23 if info.repository and not info.builtin: 

24 distribution += f"\n{info.repository}" 

25 table.add_row(Text(info.qualname), Text(info.summary), Text(distribution), Text(lets))
