1"""What `hz app install`, `uninstall` and `doctor` say to a person, in one place. 

2 

3Every failure is two lines: what happened, in the person's words (app, folder, download, open), and the one command 

4or action that fixes it. Internal names stay in the exception's details, shown only with `--details` or HZ_DEBUG=1. 

5Every folder is a full path with its plain name. Progress is step labels the installer reports, folded into 

6labelled rows by `install_rows`; code passes facts, never text. Design: the CLI install experience note of 

72026-11-07. 

8"""

185def _template(step: str) -> str: 

186 return step.split("{", 1)[0]

189_DOWNLOAD_STEPS = tuple( 

190 _template(step) for step in (RESOLVE, FETCH_MANIFEST, VERIFY_MANIFEST, CHECK_PRODUCER, DOWNLOAD) 

191) 

192_VERIFY_STEPS = (VERIFY_CHECKSUM, VERIFY_APP, CONFIRM_ASSETS, EXTRACT, CHECK_BUNDLE)

195def install_rows(*, keep_quarantine: bool) -> Callable[[str], Row | None]: 

196 """Which row each install step belongs to; steps of no row (the running check) show nothing. 

197 

198 Returns: 

199 A function from the installer's step description to its row. 

200 """ 

201 verified = Row("Verified", VERIFIED) 

202 rows = { 

203 **dict.fromkeys(_VERIFY_STEPS, verified), 

204 CHECK_SEAL: Row("Sealed", SEALED_KEEP_QUARANTINE if keep_quarantine else SEALED), 

205 WRITE_RECEIPT: Row("Installed"), 

206 VERIFY_INSTALLED: Row("Recorded", RECORDED), 

207 } 

208 

209 started = False 

210 

211 def row(step: str) -> Row | None: 

212 nonlocal started 

213 if step.startswith(_DOWNLOAD_STEPS): 

214 found: Row | None = Row(DOWNLOADING, download=True) 

215 elif step.startswith(_INSTALL_INTO): 

216 found = Row("Installed", str(Path(step.removeprefix(_INSTALL_INTO)) / BUNDLE_NAME)) 

217 else: 

218 found = rows.get(step) 

219 if step == VERIFY_INSTALLED and not started: 

220 return None # an install with nothing to do only checks the app it has: no rows 

221 started = started or found is not None 

222 return found 

223 

224 return row
