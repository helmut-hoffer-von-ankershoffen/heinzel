1"""Keep the verified, unsigned desktop bundle openable: a valid ad-hoc seal and honest quarantine handling. 

2 

3macOS on Apple silicon refuses to run arm64 code without a valid signature. A quarantined bundle whose 

4signature is missing or broken is reported as "damaged" with only a Move to Trash choice. A bundle with a 

5valid ad-hoc signature is merely "not notarized": it opens directly when it is not quarantined, and otherwise 

6macOS offers its standard Open Anyway path in System Settings. 

7 

8Heinzel downloads the release itself (no browser, so macOS adds no quarantine), checks its sha256 against the 

9release manifest and its build provenance. Only after that does the installer check the seal, add a local 

10ad-hoc seal when a release shipped without one, and leave the bundle unquarantined unless asked to keep it. 

11Nothing here ever launches the app. 

12""" 

13 

14import subprocess 

15from dataclasses import dataclass 

16from pathlib import Path 

17from typing import Literal 

18 

19from heinzel import HeinzelError 

20 

21_CODESIGN = "/usr/bin/codesign" 

22_XATTR = "/usr/bin/xattr" 

23_SPCTL = "/usr/sbin/spctl" 

24QUARANTINE = "com.apple.quarantine" 

25AGENT = "Heinzel" 

26"""The quarantine agent name the installer writes, so only our own mark is ever removed.""" 

27_OUR_MARK = f"0081;00000000;{AGENT};" 

28_TIMEOUT = 120 

29 

30type QuarantineState = Literal["none", "ours", "other"]

33class DesktopSealError(HeinzelError): 

34 """The desktop bundle's code signature could not be made valid."""

37def _run(*command: str) -> subprocess.CompletedProcess[str]: 

38 return subprocess.run( # noqa: S603 - fixed system executables and validated bundle paths 

39 command, check=False, capture_output=True, text=True, timeout=_TIMEOUT 

40 )

43def _last_line(result: subprocess.CompletedProcess[str]) -> str: 

44 lines = [line.strip() for line in (result.stderr + result.stdout).splitlines() if line.strip()] 

45 return lines[-1] if lines else f"exit status {result.returncode}"

48def seal_problem(bundle: Path) -> str | None: 

49 """Return why the bundle's signature is not valid, or None when `codesign --verify --deep --strict` passes.""" 

50 result = _run(_CODESIGN, "--verify", "--deep", "--strict", str(bundle)) 

51 return None if result.returncode == 0 else _last_line(result)

54def ensure_sealed(bundle: Path) -> bool: 

55 """Make the bundle's code seal valid, ad-hoc signing it locally only when it is broken. 

56 

57 Returns: 

58 True when the bundle had to be sealed locally, False when it already carried a valid seal. 

59 

60 Raises: 

61 DesktopSealError: If the bundle still has no valid signature afterwards. 

62 """ 

63 if seal_problem(bundle) is None: 

64 return False 

65 signed = _run(_CODESIGN, "--force", "--deep", "--sign", "-", str(bundle)) 

66 problem = _last_line(signed) if signed.returncode else seal_problem(bundle) 

67 if problem is not None: 

68 msg = f"macOS would report the desktop app as damaged: its code signature is not valid ({problem})." 

69 raise DesktopSealError(msg) 

70 return True

73def quarantine_state(bundle: Path) -> QuarantineState: 

74 """Say whether the bundle carries no quarantine, the installer's own mark or someone else's.""" 

75 result = _run(_XATTR, "-p", QUARANTINE, str(bundle)) 

76 if result.returncode: 

77 return "none" 

78 fields = result.stdout.strip().split(";") 

79 return "ours" if len(fields) > 2 and fields[2] == AGENT else "other" # noqa: PLR2004 - flags;time;agent

83 """Mark the bundle as downloaded so macOS asks once before the first open. 

84 

85 Raises: 

86 DesktopSealError: If the attribute cannot be written.

88 result = _run(_XATTR, "-w", QUARANTINE, _OUR_MARK, str(bundle)) 

89 if result.returncode: 89 ↛ 90line 89 didn't jump to line 90 because the condition on line 89 was never true

90 msg = f"The desktop app could not be marked as downloaded ({_last_line(result)})." 

91 raise DesktopSealError(msg)

94def clear_our_quarantine(bundle: Path) -> bool: 

95 """Remove the installer's own quarantine mark from the bundle root; never another program's. 

96 

97 Returns: 

98 True when our mark was removed. 

99 

100 Raises: 

101 DesktopSealError: If our mark is present but cannot be removed. 

102 """ 

103 if quarantine_state(bundle) != "ours": 

104 return False 

105 result = _run(_XATTR, "-d", QUARANTINE, str(bundle)) 

106 if result.returncode: 106 ↛ 107line 106 didn't jump to line 107 because the condition on line 106 was never true

107 msg = f"The desktop app's quarantine mark could not be removed ({_last_line(result)})." 

108 raise DesktopSealError(msg) 

109 return True

112def gatekeeper_verdict(bundle: Path) -> str: 

113 """Return the `spctl --assess` verdict line, which never launches the app.""" 

114 return _last_line(_run(_SPCTL, "--assess", "-vv", "--type", "exec", str(bundle)))

117@dataclass(frozen=True) 

118class SealReport: 

119 """What macOS will make of the installed bundle, in plain words."""

121 bundle: Path 

122 problem: str | None 

123 quarantine: QuarantineState 

124 gatekeeper: str

126 @property 

127 def opens(self) -> bool:

129 return self.problem is None

131 def as_dict(self) -> dict[str, object]: 

132 """The same facts for scripts and agents.""" 

133 return { 

134 "app": str(self.bundle), 

135 "opens": self.opens, 

136 "signature_problem": self.problem, 

137 "quarantine": self.quarantine, 

138 "gatekeeper": self.gatekeeper, 

139 }

141 def lines(self) -> list[str]: 

142 """Describe the state for a person, one fact per line.""" 

143 signature = ( 

144 "Signature: valid ad-hoc seal (unsigned app from this project, not notarized by Apple)." 

145 if self.problem is None 

146 else f"Signature: broken ({self.problem}). macOS will call the app damaged." 

147 ) 

148 quarantine = { 

149 "none": "Quarantine: none. A double-click opens the app.", 

150 "ours": "Quarantine: set by Heinzel. macOS asks once; open it via System Settings, Privacy & Security, " 

151 "Open Anyway.", 

152 "other": "Quarantine: set by another program. macOS asks once; open it via System Settings, " 

153 "Privacy & Security, Open Anyway.", 

154 }[self.quarantine] 

155 return [f"App: {self.bundle}", signature, quarantine, f"Gatekeeper assessment: {self.gatekeeper}"]

158def inspect_bundle(bundle: Path) -> SealReport: 

159 """Check signature, quarantine and Gatekeeper's verdict without launching anything.""" 

160 return SealReport(bundle, seal_problem(bundle), quarantine_state(bundle), gatekeeper_verdict(bundle))
