1"""Native notifications, for when no web app is open (ADR 0043). 

2 

3macOS: Heinzel Notifier, hz's own app bundled in the wheel (`_macos.py`): it shows with the Heinzel icon and a click 

4opens the web app. osascript only as the last resort, when this installation carries no app: it shows as Script 

5Editor and a click opens Script Editor, so the text names the command to run. Linux: notify-send, when there is a 

6desktop session. 

7 

8Text reaches every notifier only as separate arguments, never inside a script or a shell command: osascript runs a 

9fixed `on run argv` script that reads its strings from `argv`, so a title such as `" & do shell script "…` stays text. 

10Every string is cleaned of control and invisible characters, flattened to one line, stripped of leading characters 

11a notifier could read as an option, escaped where notify-send reads markup, and bounded. 

12"""

14import html 

15import re 

16import subprocess 

17from collections.abc import Callable, Mapping, Sequence 

18from dataclasses import dataclass, field

23APP = "Heinzelmännchen" 

24TITLE_LIMIT = 80 

25BODY_LIMIT = 240 

26TIMEOUT = 10 

27"""Seconds a notifier may take; they answer at once unless something is wrong.""" 

28_SPACE = re.compile(r"\s+") 

29_LEADING = "-[\\ " 

30"""Characters a notifier could read as the start of an option.""" 

31_OSASCRIPT = ( 

32 "on run argv", 

33 "display notification (item 2 of argv) with title (item 1 of argv) subtitle (item 3 of argv)", 

34 "end run", 

35) 

36"""A fixed script: the strings are its arguments, never part of its source.""" 

37_SESSION = ("DBUS_SESSION_BUS_ADDRESS", "DISPLAY", "WAYLAND_DISPLAY") 

38"""Without one of these there is no desktop session to show a notification in.""" 

39HEINZEL = "heinzel-notifier" 

40BANNERS = (

43 "app's alert style must not be None: System Settings > Notifications > {app}." 

44) 

45"""The usual reasons a macOS notification is delivered but no banner shows, with where to change them.""" 

46ALLOW = (

63def sanitise(text: str, limit: int) -> str: 

64 """`text` as one clean line of at most `limit` characters, never starting like an option.""" 

65 line = _SPACE.sub(" ", clean_text(text)).strip().lstrip(_LEADING) 

66 return line if len(line) <= limit else line[: limit - 1].rstrip() + "…" 

67 

68 

69def run(command: Sequence[str]) -> subprocess.CompletedProcess[str]: 

70 """Run a notifier with a timeout; a timeout becomes exit code 124, like timeout(1).""" 

71 try: 

72 return subprocess.run(command, capture_output=True, text=True, check=False, timeout=TIMEOUT) # noqa: S603 - executables resolved with shutil.which, arguments only, no shell 

73 except subprocess.TimeoutExpired: 

74 return subprocess.CompletedProcess(command, 124, "", f"{command[0]} timed out after {TIMEOUT}s")

77@dataclass(frozen=True, slots=True) 

78class Native: 

79 """The native notifier of this machine, or why there is none.""" 

80 

81 name: str | None 

82 """`heinzel-notifier`, `osascript`, `notify-send`, or None when there is none.""" 

83 executable: str | None = None 

84 detail: str = "" 

85 """How notices pop up, in words, for doctor and `hz system notify list`.""" 

86 fix: str = "" 

87 """What makes it better, or empty.""" 

88 clicks: bool = False 

89 """Whether a click can open a link.""" 

90 runner: Runner = field(default=run, repr=False, compare=False) 

91 app: NotifierApp | None = field(default=None, repr=False, compare=False) 

92 

93 def argv(self, title: str, body: str) -> list[str]: 

94 """The command that shows one notification with osascript or notify-send; arguments only. 

95 

96 Raises: 

97 ValueError: If there is no such notifier. 

98 """ 

99 if self.name not in {"osascript", "notify-send"} or self.executable is None: 

100 msg = f"no native notifier: {self.detail}" 

101 raise ValueError(msg) 

102 title, body = sanitise(title, TITLE_LIMIT), sanitise(body, BODY_LIMIT) 

103 if self.name == "osascript": 

104 return [self.executable, *(arg for line in _OSASCRIPT for arg in ("-e", line)), APP, body or title, title] 

105 # notify-send: the body may be read as markup by the notification server, the summary isn't. 

106 return [self.executable, f"--app-name={APP}", "--", title, html.escape(body, quote=False)] 

107 

108 def post(self, title: str, body: str, *, url: str | None = None, asking: bool = False) -> str | None: 

109 """Show one notification; None when it was shown, else why not. 

110 

111 `asking` says a person is at the screen (`hz system notify install`): only then may Heinzel Notifier make 

112 macOS ask whether to allow it. The daemon doesn't, or an unanswered question would hold every tick. 

113 """ 

114 if self.name == HEINZEL and self.app is not None: 

115 return _through_app(self.app, title, body, url=url, asking=asking) 

116 try: 

117 command = self.argv(title, body) 

118 except ValueError as exc: 

119 return str(exc) 

120 result = self.runner(command) 

121 if result.returncode != 0: 

122 return f"{self.name} failed: {(result.stderr or result.stdout).strip() or f'exit {result.returncode}'}" 

123 return None

126def _through_app(app: NotifierApp, title: str, body: str, *, url: str | None, asking: bool) -> str | None: 

127 """Show one notification through Heinzel Notifier, installing it first if needed; None when it was shown.""" 

128 try: 

129 installed = app.ensure() 

130 except OSError as exc: 

131 return f"Heinzel Notifier can't be installed: {exc.strerror or exc}" 

132 if not app.path.exists(): 132 ↛ 133line 132 didn't jump to line 133 because the condition on line 132 was never true

133 return f"Heinzel Notifier isn't installed: {installed.detail}" 

134 if not asking and app.status() == "not-determined": 

135 return NOT_ASKED 

136 said = app.post(sanitise(title, TITLE_LIMIT), sanitise(body, BODY_LIMIT), url, asking=asking) 

137 if said == "posted": 

138 return None 

139 return DENIED if said == "denied" else f"Heinzel Notifier: {said}"

142def native( 

143 platform: str, 

144 which: Callable[[str], str | None], 

145 environ: Mapping[str, str], 

146 runner: Runner = run, 

147 app: NotifierApp | None = None, 

148) -> Native: 

149 """The best native notifier this machine has.""" 

150 if platform == "darwin": 

151 return _darwin(which, runner, app or NotifierApp()) 

152 if platform.startswith("linux"): 

153 return _linux(which, environ, runner) 

154 return Native(None, detail=f"no native notifications on {platform}; `hz system notify list` shows them")

157def _darwin(which: Callable[[str], str | None], runner: Runner, app: NotifierApp) -> Native: 

158 if app.available: 

159 return Native( 

160 HEINZEL, 

161 str(app.path), 

162 detail="desktop notifications through Heinzel Notifier; a click opens the screen", 

163 fix=BANNERS.format(app="Heinzel Notifier"), 

164 clicks=True, 

165 runner=runner, 

166 app=app, 

167 ) 

168 if found := which("osascript"): 

169 return Native( 

170 "osascript", 

171 found, 

172 detail="desktop notifications through osascript; they show as Script Editor and name the command", 

173 fix="This installation carries no Heinzel Notifier; a release installed with `uv tool install` does. " 

174 + BANNERS.format(app="Script Editor"), 

175 runner=runner, 

176 ) 

177 return Native(None, detail="osascript isn't on PATH, so the daemon can't pop up notifications")

180def _linux(which: Callable[[str], str | None], environ: Mapping[str, str], runner: Runner) -> Native: 

181 found = which("notify-send") 

182 if found is None: 

183 return Native( 

184 None, 

185 detail="notify-send isn't on PATH, so the daemon can't pop up notifications", 

186 fix="install libnotify (notify-send), e.g. `sudo apt install libnotify-bin`", 

187 ) 

188 if not any(environ.get(name) for name in _SESSION): 

189 return Native( 

190 None, 

191 detail="no desktop session (headless), so the daemon can't pop up notifications; " 

192 "`hz system notify list` shows them", 

193 ) 

194 return Native("notify-send", found, detail=f"desktop notifications through {found}", runner=runner)
