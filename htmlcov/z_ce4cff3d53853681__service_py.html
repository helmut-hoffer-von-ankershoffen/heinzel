54def _branch(stack: LayerStack, layer: str) -> str | None: 

55 """`site` for the site, `org` for a layer beside it on the org line, None for the company and the team.""" 

56 if layer == stack.site: 

57 return "site" 

58 if stack.site is None or layer in stack.site_ancestors or layer == stack.layer: 

59 return None 

60 return "org"

81 def catalog(self) -> Catalog: 

82 """Each skill by name; of several offering one name, the most specific source wins. 

83 

84 General to specific: a plugin outside the person's stack, the core, a layer of the stack (the team over the 

85 site and the orgs, those over the company), the person's own directory. So a plugin outside the stack never 

86 hides a core skill. An org and the site offering one name, with nothing more specific offering it, is a 

87 conflict nobody decided (ADR 0040): the name is left out and reported. 

88 """ 

89 stack = self._kernel.layer_stack() 

90 rank = {name: index for index, name in enumerate(stack.names)} 

91 candidates: list[tuple[tuple[int, int], Offered]] = [] 

92 for record, let in self._kernel.lets(SkillLet): 

93 top = record.path[0] 

94 if top in rank: 

95 candidates.append(((2, rank[top]), Offered(let.skill, let.directory, "layer", record))) 

96 elif record.origin.builtin: 

97 candidates.append(((1, 0), Offered(let.skill, let.directory, "core", record))) 

98 else: 

99 candidates.append(((0, 0), Offered(let.skill, let.directory, "plugin", record))) 

100 personal, problems = self._personal() 

101 candidates += [((3, 0), offered) for offered in personal] 

102 by_name: dict[str, list[tuple[tuple[int, int], Offered]]] = {} 

103 for priority, offered in sorted(candidates, key=lambda pair: (-pair[0][0], -pair[0][1], pair[1].module or "")): 

104 by_name.setdefault(offered.skill.name, []).append((priority, offered)) 

105 skills: dict[str, Offered] = {} 

106 conflicts: list[str] = [] 

107 for name in sorted(by_name): 

108 (_, winner), *others = by_name[name] 

109 if winner.source == "layer" and winner.record is not None: 

110 branch = _branch(stack, winner.record.path[0]) 

111 rivals = [ 

112 o.record.path[0] 

113 for _, o in others 

114 if o.source == "layer" and o.record and _branch(stack, o.record.path[0]) not in {None, branch} 

115 ] 

116 if branch is not None and rivals: 

117 conflicts.append( 

118 f"{name}: offered by {winner.record.path[0]} and {', '.join(rivals)}, an org and the site; " 

119 f"nobody decided which applies, so neither does until {stack.layer or 'your team'} offers it" 

120 ) 

121 continue 

122 shadows = [o.module or PERSONAL for _, o in others] 

123 skills[name] = Offered(winner.skill, winner.directory, winner.source, winner.record, shadows) 

124 return Catalog(skills, conflicts, problems)

126 def _personal(self) -> tuple[list[Offered], list[str]]: 

127 """The person's own skills, each checked like a plugin's; one that fails is a problem, not an error.""" 

128 root = self.personal_dir() 

129 if not root.is_dir(): 

130 return [], [] 

131 resolved = root.resolve() 

132 for target in self.export_dirs(): 

133 if resolved == target.resolve() or resolved.is_relative_to(target.resolve()): 

134 return [], [f"{root} is inside {target}, where exports go; keep your own skills elsewhere"] 

135 offered: list[Offered] = [] 

136 problems: list[str] = [] 

137 try: 

138 directories = sorted(root.iterdir()) 

139 except OSError as exc: 

140 return [], [f"{root} can't be listed ({exc.strerror or type(exc).__name__})"] 

141 for directory in directories: 

142 if directory.name.startswith(".") or not directory.is_dir(): 

143 continue 

144 try: 

145 offered.append(Offered(read_skill(directory), directory, "personal", None)) 

146 except ValueError as exc: 

147 problems.append(str(exc)) 

148 return offered, problems

196 if allowed is not None and name not in allowed: 

197 msg = f"Skill {name!r} isn't one this may use; it may use {', '.join(sorted(allowed)) or 'none'}."

219 def listing(self) -> SkillList: 

220 """Every skill in effect, with what hides what, the chat's set and where each is exported.""" 

221 catalog = self.catalog() 

222 chat = set(self.chat_allowed()) 

223 exported = self._exporter().exported() 

224 views = [self._view(offered, chat, exported) for offered in catalog.skills.values()] 

225 next_steps = [f"hz system skill show {views[0].name}"] if views else [] 

226 if views and not any(view.exported for view in views): 

227 next_steps.append(f"hz system skill export --name {views[0].name}") 

228 return SkillList(skills=views, conflicts=catalog.conflicts, problems=catalog.problems, next=next_steps)

240 def export(self, names: Collection[str], *, everything: bool = False, sync: bool = False) -> ExportResult: 

241 """Export the named skills (or every one but your own) for Claude Code; with `sync`, only remove stale ones. 

242 

243 Raises: 

244 HeinzelError: If nothing is named, a name is unknown, or `sync` comes with names. 

245 """ 

246 catalog = self.catalog() 

247 if sync: 

248 if names or everything: 

249 msg = "`--sync` only removes stale exports; run it without names or `--all-skills`." 

250 raise HeinzelError(msg) 

251 return self._exporter().sync(catalog.skills) 

252 if everything: 

253 chosen = [name for name, offered in catalog.skills.items() if offered.source != "personal"] 

254 elif names: 

255 if unknown := sorted(set(names) - set(catalog.skills)): 

256 msg = f"No skill is called {', '.join(unknown)}; `hz system skill list` shows the skills in effect." 

257 raise HeinzelError(msg) 

258 chosen = sorted(set(names)) 

259 else: 

260 msg = "Name the skills to export (`--name review-pull-request`), or pass `--all-skills`." 

261 raise HeinzelError(msg) 

262 return self._exporter().export([catalog.skills[name] for name in chosen])
