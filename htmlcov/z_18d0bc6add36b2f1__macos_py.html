1"""Heinzel Notifier, hz's own macOS notifier app (ADR 0043): bundled in the wheel, installed into ~/Applications. 

2 

3The release build compiles `macos/main.swift` into a universal, ad-hoc signed `HeinzelNotifier.app` (`make 

4notifier`) that ships inside the package; nothing is compiled on a person's Mac. hz copies it to `~/Applications/Heinzel 

5Notifier.app` when it is missing or differs from the bundled one, and checks the copy against the bundle's digest. 

6Installed by Python from a wheel, it carries no quarantine flag, so Gatekeeper doesn't stop it. 

7 

8hz starts it with `open -n`, through LaunchServices, which puts it in the person's GUI session: that is what lets the 

9daemon (launchd) and SSH sessions post notifications. Run directly, or as the child of a sandboxed process, macOS 

10refuses ("Notifications are not allowed for this application"). The first notification shows macOS's question, with 

11the Heinzel icon, whether to allow them; the person answers once. 

12"""

47def source_version(sources: Path = SOURCES) -> str: 

48 """The app's version: a digest of its sources, so a changed source means a new app.""" 

49 digest = hashlib.sha256() 

50 for name in ( 

51 "Info.plist", 

52 "heinzel-swarm.svg", 

53 "heinzel.svg", 

54 "icon.swift", 

55 "main.swift", 

56 ): # scripts/notifier.py SOURCE_NAMES 

57 digest.update(name.encode() + b"\0" + (sources / name).read_bytes() + b"\0") 

58 return digest.hexdigest()[:16]

61def tree_digest(app: Path) -> str | None: 

62 """SHA-256 over every file of an app bundle, by relative path; None if it isn't one.""" 

63 if not (app / EXECUTABLE).is_file(): 

64 return None 

65 digest = hashlib.sha256() 

66 for path in sorted(p for p in app.rglob("*") if p.is_symlink() or p.is_file()): 

67 content = b"link:" + str(path.readlink()).encode() if path.is_symlink() else path.read_bytes() 

68 digest.update(path.relative_to(app).as_posix().encode() + b"\0" + content + b"\0") 

69 return digest.hexdigest()

72def bundle_version(app: Path) -> str | None: 

73 """`CFBundleVersion` of an app bundle; None if it has none.""" 

74 try: 

75 with (app / "Contents" / "Info.plist").open("rb") as handle: 

76 version = plistlib.load(handle).get("CFBundleVersion")

119 @property 

120 def available(self) -> bool: 

121 """Whether there is an app to use: bundled with this installation, or installed before.""" 

122 return tree_digest(self.bundled) is not None or tree_digest(self.path) is not None

124 def rebuild(self) -> str | None: 

125 """In a source checkout, rebuild the bundled app when its sources changed; why it failed, else None. 

126 

127 A wheel ships the app built when it was installed and has no build script: nothing to do there. 

128 """ 

129 script = self._build_script 

130 if script is None or not script.is_file(): 

131 return None 

132 if self._runner([sys.executable, str(script), "--check"]).returncode == 0: 

133 return None 

134 built = self._runner([sys.executable, str(script)], BUILD_TIMEOUT) 

135 if built.returncode == 0: 

136 return None 

137 lines = (built.stderr or built.stdout).strip().splitlines() 

138 return lines[-1] if lines else f"the build exited with {built.returncode}"

140 def inspect(self) -> Install: 

141 """Whether the installed app matches the bundled one, without changing anything.""" 

142 wanted, have = tree_digest(self.bundled), tree_digest(self.path) 

143 if have is None: 

144 return Install(self.path, "missing", "not in ~/Applications yet; the first notification installs it") 

145 if wanted is not None and have != wanted: 

146 return Install(self.path, "outdated", "it differs from the app this installation bundles") 

147 return Install(self.path, "current")

149 def ensure(self) -> Install: 

150 """Copy the bundled app into ~/Applications if it is missing or differs, and verify the copy.""" 

151 wanted = tree_digest(self.bundled) 

152 have = tree_digest(self.path) 

153 if wanted is None: 

154 state = "outdated" if have is not None else "missing" 

155 return Install( 

156 self.path, 

157 state, 

158 "this installation carries no Heinzel Notifier: install hz again on a Mac with the Xcode Command Line " 

159 "Tools (xcode-select --install)", 

160 ) 

161 if have == wanted: 

162 return Install(self.path, "current") 

163 try: 

164 self._copy() 

165 except OSError as exc: 

166 return Install(self.path, "outdated" if have else "missing", f"can't install it: {exc.strerror or exc}")

168 return Install(self.path, "outdated", "the installed copy doesn't match the bundled app") 

169 return Install(self.path, "installed")

171 def _copy(self) -> None: 

172 """Replace the installed app at once: copy beside it, then swap, so a failed copy leaves the old one. 

173 

174 Raises: 

175 OSError: If it can't be copied or moved into place; the old app, if any, is put back. 

176 """ 

177 self.path.parent.mkdir(parents=True, exist_ok=True) 

178 staging = Path(tempfile.mkdtemp(prefix=".heinzel-notifier-", dir=self.path.parent)) 

179 try: 

180 fresh = staging / INSTALLED_NAME 

181 shutil.copytree(self.bundled, fresh, symlinks=True) 

182 (fresh / EXECUTABLE).chmod(0o755) # a wheel may not keep the executable bit 

183 old = staging / "old.app" 

184 if self.path.exists(): 

185 self.path.rename(old) 

186 try: 

187 fresh.rename(self.path) 

188 except OSError: 

189 if old.exists(): 189 ↛ 191line 189 didn't jump to line 191 because the condition on line 189 was always true

190 old.rename(self.path) # never leave no app at all 

191 raise 

192 finally: 

193 shutil.rmtree(staging, ignore_errors=True)

195 def quarantined(self) -> bool: 

196 """Whether the installed app carries the quarantine flag, which makes Gatekeeper stop it.""" 

197 result = self._runner(["/usr/bin/xattr", "-p", QUARANTINE, str(self.path)]) 

198 return result.returncode == 0

201 """MacOS's answer for the app: `authorized`, `denied`, `not-determined`, `provisional`, or why unknown."""

204 def post(self, title: str, message: str, url: str | None, *, asking: bool = False) -> str: 

205 """Post one notification through LaunchServices; `posted`, `denied`, or what went wrong. 

206 

207 With `asking`, a person is there to answer macOS's question, so it waits longer for the app. 

208 """ 

209 arguments = ["--title", title, "--message", message] 

210 if url is not None: 

211 arguments += ["--open", url] 

212 return self._launch(arguments, ASKING_TIMEOUT if asking else TIMEOUT) 

213 

214 def _launch(self, arguments: list[str], timeout: float = TIMEOUT) -> str: 

215 with tempfile.TemporaryDirectory(prefix="hz-notifier-") as directory: 

216 out = Path(directory) / "out" 

217 out.touch(mode=0o600) 

218 command = [self._open, "-n", "-g", "-W", "--stdout", str(out), "--stderr", os.devnull, str(self.path)] 

219 result = self._runner([*command, "--args", *arguments], timeout) 

220 said = out.read_text(encoding="utf-8", errors="replace").strip().splitlines() 

221 if result.returncode == 124 or said[-1:] == ["waiting"]: # noqa: PLR2004 - timeout(1)'s exit code 

222 return "waiting: macOS is asking whether to allow Heinzel Notifier's notifications" 

223 if result.returncode != 0: 

224 return f"open failed: {(result.stderr or result.stdout).strip() or f'exit {result.returncode}'}" 

225 return said[-1] if said else "no answer from Heinzel Notifier"
