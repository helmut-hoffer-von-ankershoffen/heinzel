1"""Heinzel Notifier, hz's own macOS notifier app (ADR 0043): bundled in the wheel, installed into ~/Applications. 

2 

3The release build compiles `macos/main.swift` into a universal, ad-hoc signed `HeinzelNotifier.app` (`make 

4notifier`) that ships inside the package; nothing is compiled on a person's Mac. hz copies it to `~/Applications/Heinzel 

5Notifier.app` when it is missing or differs from the bundled one, and checks the copy against the bundle's digest. 

6Installed by Python from a wheel, it carries no quarantine flag, so Gatekeeper doesn't stop it. 

7 

8hz starts it with `open -n`, through LaunchServices, which puts it in the person's GUI session: that is what lets the 

9daemon (launchd) and SSH sessions post notifications. Run directly, or as the child of a sandboxed process, macOS 

10refuses ("Notifications are not allowed for this application"). The first notification shows macOS's question, with 

11the Heinzel icon, whether to allow them; the person answers once. 

12"""

66def source_version(sources: Path = SOURCES) -> str: 

67 """The app's version: a digest of its sources, so a changed source means a new app."""

80def tree_digest(app: Path) -> str | None: 

81 """SHA-256 over every file of an app bundle, by relative path; None if it isn't one.""" 

82 if not (app / EXECUTABLE).is_file(): 

83 return None 

84 digest = hashlib.sha256() 

85 for path in sorted(p for p in app.rglob("*") if p.is_symlink() or p.is_file()): 

86 content = b"link:" + str(path.readlink()).encode() if path.is_symlink() else path.read_bytes() 

87 digest.update(path.relative_to(app).as_posix().encode() + b"\0" + content + b"\0") 

88 return digest.hexdigest()

124def parse_etime(text: str) -> int | None: 

125 """Seconds in `ps`'s `[[dd-]hh:]mm:ss`; None if it isn't that.""" 

126 match = _ETIME.fullmatch(text.strip()) 

127 if match is None: 

128 return None 

129 parts = {name: int(value or 0) for name, value in match.groupdict().items()} 

130 return ((parts["d"] * 24 + parts["h"]) * 60 + parts["m"]) * 60 + parts["s"]

169 def rebuild(self) -> str | None: 

170 """In a source checkout, rebuild the bundled app when its sources changed; why it failed, else None. 

171 

172 A wheel ships the app built when it was installed and has no build script: nothing to do there. 

173 """ 

174 script = self._build_script 

175 if script is None or not script.is_file(): 

176 return None 

177 if self._runner([sys.executable, str(script), "--check"]).returncode == 0: 

178 return None 

179 built = self._runner([sys.executable, str(script)], BUILD_TIMEOUT) 

180 if built.returncode == 0: 

181 return None 

182 lines = (built.stderr or built.stdout).strip().splitlines() 

183 return lines[-1] if lines else f"the build exited with {built.returncode}"

185 def inspect(self) -> Install: 

186 """Whether the installed app matches the bundled one, without changing anything.""" 

187 wanted, have = tree_digest(self.bundled), tree_digest(self.path) 

188 if have is None: 

189 return Install(self.path, "missing", "not in ~/Applications yet; the first notification installs it") 

190 if wanted is not None and have != wanted: 

191 return Install(self.path, "outdated", "it differs from the app this installation bundles") 

192 return Install(self.path, "current")

216 def _copy(self) -> None: 

217 """Replace the installed app at once: copy beside it, then swap, so a failed copy leaves the old one. 

218 

219 Raises: 

220 OSError: If it can't be copied or moved into place; the old app, if any, is put back. 

221 """ 

222 self.path.parent.mkdir(parents=True, exist_ok=True) 

223 staging = Path(tempfile.mkdtemp(prefix=".heinzel-notifier-", dir=self.path.parent)) 

224 try: 

225 fresh = staging / INSTALLED_NAME 

226 shutil.copytree(self.bundled, fresh, symlinks=True) 

227 (fresh / EXECUTABLE).chmod(0o755) # a wheel may not keep the executable bit 

228 old = staging / "old.app" 

229 if self.path.exists(): 

230 self.path.rename(old)

232 fresh.rename(self.path) 

233 except OSError:

235 old.rename(self.path) # never leave no app at all

245 def instances(self) -> list[Instance]: 

246 """The Heinzel Notifier processes running now, found by the bundle id in launchd's label, never by a name. 

247 

248 `launchctl list` names what LaunchServices started `application.<bundle id>.<ids>`; only labels with exactly 

249 this app's bundle id and a live pid count, so no other process is ever matched. 

250 """ 

251 listed = self._runner(["/bin/launchctl", "list"]) 

252 if listed.returncode != 0: 252 ↛ 253line 252 didn't jump to line 253 because the condition on line 252 was never true

253 return [] 

254 found: list[Instance] = [] 

255 for line in listed.stdout.splitlines(): 

256 fields = line.split("\t") 

257 if len(fields) != 3 or not fields[2].startswith(LABEL_PREFIX) or not fields[0].isdecimal(): # noqa: PLR2004 

258 continue 

259 pid = int(fields[0]) 

260 aged = self._runner(["/bin/ps", "-o", "etime=", "-p", str(pid)]) 

261 found.append(Instance(pid, fields[2], parse_etime(aged.stdout) if aged.returncode == 0 else None)) 

262 return found

264 def registered(self) -> list[Path]: 

265 """Every app LaunchServices knows by this bundle id, wherever it is or was. 

266 

267 macOS registers a copy the first time it is launched from a new path. A test, a build in a worktree or a 

268 staging folder leaves one each, and System Settings > Notifications lists them all under one name. 

269 """ 

270 dumped = self._runner([LSREGISTER, "-dump"], REGISTRATION_TIMEOUT) 

271 if dumped.returncode != 0: 271 ↛ 272line 271 didn't jump to line 272 because the condition on line 271 was never true

272 return [] 

273 paths: list[Path] = [] 

274 for block in _BLOCK.split(dumped.stdout): 

275 path: str | None = None 

276 ours = False 

277 for line in block.splitlines(): 

278 field, _, value = line.partition(":") 

279 if field == "path": 

280 path = value.strip().rpartition(" (0x")[0] or None # `<path> (0x<id>)` 

281 elif field == "identifier": 

282 ours = value.strip() == IDENTIFIER 

283 if path is not None and ours: 

284 paths.append(Path(path)) 

285 return paths

338 def _launch(self, arguments: list[str], timeout: float = TIMEOUT) -> str: 

339 # A test in a temporary home would start an app from a path macOS has never seen, and macOS would list it in 

340 # System Settings > Notifications for good: hundreds of "Heinzel Notifier" entries. Fakes are not refused. 

341 if self._runner is run and os.environ.get("PYTEST_CURRENT_TEST") and not os.environ.get("HZ_REAL_NOTIFIER"): 

342 return f"open failed: {TESTS_REFUSED}" 

343 with tempfile.TemporaryDirectory(prefix="hz-notifier-") as directory: 

344 out = Path(directory) / "out" 

345 out.touch(mode=0o600) 

346 command = [self._open, "-n", "-g", "-W", "--stdout", str(out), "--stderr", os.devnull, str(self.path)] 

347 result = self._runner([*command, "--args", *arguments], timeout) 

348 said = out.read_text(encoding="utf-8", errors="replace").strip().splitlines() 

349 if result.returncode == 124 or said[-1:] == ["waiting"]: # noqa: PLR2004 - timeout(1)'s exit code 

350 return "waiting: macOS is asking whether to allow Heinzel Notifier's notifications" 

351 # An app that ends at once can be gone before `open -W` starts watching it; it had already written its answer. 

352 if result.returncode != 0 and not (said and ALREADY_GONE in result.stderr): 

353 return f"open failed: {(result.stderr or result.stdout).strip() or f'exit {result.returncode}'}" 

354 return said[-1] if said else "no answer from Heinzel Notifier"
