1"""GitHub's REST API through `gh api`: pagination, schema checks, and errors people can act on. 

2 

3Every failure becomes a `HeinzelError` that says what happened and what to do. `gh`'s messages are data from outside 

4the process: they are redacted of anything shaped like a GitHub token first, then stripped of control and 

5bidirectional characters, and only then cut to length, so a cut can never leave half a token behind. 

6""" 

7 

8import json 

9import logging 

10import re 

11import time 

12from collections.abc import Iterator, Mapping 

13from typing import Any, cast 

14 

15from heinzel import HeinzelError, clean_text 

16from heinzel_github._gh import MAX_BYTES, Gh, GhResult, login_fix 

17 

18API_VERSION = "2022-11-28" 

19"""The REST API version requested, so GitHub doesn't change answers under us.""" 

20PER_PAGE = 100 

21MAX_PAGES = 100 

22"""Pages read for one call (10,000 items at 100 a page) before asking for a narrower window.""" 

23READ_TIMEOUTS = 10 

24"""A whole read, all its pages, may take this many times `github.timeout`.""" 

25MAX_READ_BYTES = 50 * 1024 * 1024 

26"""Bytes a whole read may answer with, all its pages.""" 

27MAX_SAID = 300 

28"""Characters of GitHub's or `gh`'s own error message shown, after redaction.""" 

29NOT_LOGGED_IN = 4 

30"""`gh`'s exit code when it has no login for the host.""" 

31_LOG = logging.getLogger(__name__) 

32_STATUS = re.compile(r"\(HTTP (\d{3})\)\s*$", re.MULTILINE) 

33# Prefixed tokens anywhere, also glued to other text; a classic 40-hex token only as a word of its own. 

34_TOKEN = re.compile(r"(?:gh[opsur]_|github_pat_)[A-Za-z0-9_]{16,}|\b[0-9a-f]{40}\b")

37class GhLoginError(HeinzelError): 

38 """`gh` has no login for the host, or GitHub rejected it: the message names `gh auth login`.""" 

39

41def redact(text: str) -> str: 

42 """`text` with anything shaped like a GitHub token (`ghp_…`, `github_pat_…`, 40 hex digits) replaced by `***`.""" 

43 return _TOKEN.sub("***", text)

46def said(text: str) -> str: 

47 """A message from outside, safe to show: redacted, cleaned, then cut to `MAX_SAID` characters.""" 

48 return clean_text(redact(text)).strip()[:MAX_SAID] or "no details"

51class GhClient: 

52 """GETs JSON from GitHub's REST API with `gh api`, as the account `gh` is logged in with on one host. 

53 

54 That login usually may write too; this client only ever sends GET, to paths the adapter builds under `repos/`. 

55 """ 

56 

57 def __init__(self, gh: Gh, host: str, timeout: float) -> None: 

58 """Read from `host` through `gh`, giving each call `timeout` seconds.""" 

59 self._gh = gh 

60 self._host = host 

61 self._timeout = timeout 

62 

63 def pages( 

64 self, path: str, params: Mapping[str, str], *, too_many: str = "ask for a shorter window" 

65 ) -> Iterator[list[dict[str, Any]]]: 

66 """Each page of a list endpoint such as `repos/acme/widgets/pulls`, 100 items a page, until a short page. 

67 

68 Stop iterating early to stop asking for pages. `too_many` is the advice given when there are more than 

69 `MAX_PAGES` full pages; what helps depends on which bound of the window the paging depends on. 

70 

71 Yields: 

72 The items of one page. 

73 

74 Raises: 

75 HeinzelError: If `gh` is missing or not logged in, GitHub refuses, rate-limits or answers unexpectedly, 

76 or there are more than `MAX_PAGES` pages. 

77 """ 

78 # One budget for the whole read, not only per call (ADR 0028's window rule, ADR 0035 statement 4). 

79 deadline = time.monotonic() + self._timeout * READ_TIMEOUTS 

80 read = 0 

81 for page in range(1, MAX_PAGES + 1): 

82 if (left := deadline - time.monotonic()) <= 0: 

83 msg = f"Reading {path} took more than {self._timeout * READ_TIMEOUTS:g} seconds; {too_many}." 

84 raise HeinzelError(msg) 

85 items, size = self._list(path, {**params, "per_page": str(PER_PAGE), "page": str(page)}, left) 

86 read += size 

87 if read > MAX_READ_BYTES: 

88 msg = f"{path} answered with more than {MAX_READ_BYTES} bytes in all; {too_many}." 

89 raise HeinzelError(msg) 

90 yield items 

91 if len(items) < PER_PAGE: 

92 return 

93 msg = f"{path} has {MAX_PAGES} or more full pages of results; {too_many}." 

94 raise HeinzelError(msg) 

95 

96 def get(self, path: str) -> dict[str, Any]: 

97 """GET one object, such as `user` (who gh is logged in as), within the client's timeout. 

98 

99 Raises: 

100 HeinzelError: If `gh` is missing or not logged in, GitHub refuses or answers with something else. 

101 """ 

102 found, _ = self._call(path, {}, self._timeout) 

103 if not isinstance(found, dict): 103 ↛ 104line 103 didn't jump to line 104 because the condition on line 103 was never true

104 msg = f"GitHub answered {path} with something other than an object; is {self._host} a GitHub server?" 

105 raise HeinzelError(msg) 

106 return cast("dict[str, Any]", found) 

107 

108 def _list(self, path: str, params: Mapping[str, str], left: float) -> tuple[list[dict[str, Any]], int]: 

109 items, size = self._call(path, params, left) 

110 if not isinstance(items, list) or not all(isinstance(item, dict) for item in cast("list[object]", items)): 

111 msg = ( 

112 f"GitHub answered {path} with something other than a list of objects; is {self._host} a GitHub server?" 

113 ) 

114 raise HeinzelError(msg) 

115 return cast("list[dict[str, Any]]", items), size # checked above 

116 

117 def _call(self, path: str, params: Mapping[str, str], left: float) -> tuple[Any, int]: 

118 args = [ 

119 "api", 

120 "--hostname", 

121 self._host, 

122 "--method", 

123 "GET", 

124 "--header", 

125 "Accept: application/vnd.github+json", 

126 "--header", 

127 f"X-GitHub-Api-Version: {API_VERSION}", 

128 path, 

129 ] 

130 for key, value in params.items(): 

131 args += ["--raw-field", f"{key}={value}"] # raw: a value is never read as @file or a number 

132 timeout = min(self._timeout, left) 

133 result = self._gh.run(args, host=self._host, timeout=timeout) 

134 if result.returncode is None: 

135 msg = ( 

136 f"`gh api {path}` did not finish within {timeout:g} seconds; check the network, or allow more " 

137 "time with `hz system config set github.timeout 120`." 

138 ) 

139 raise HeinzelError(msg) 

140 if result.truncated: 

141 msg = f"GitHub's answer for {path} is larger than {MAX_BYTES} bytes; ask for a shorter window." 

142 raise HeinzelError(msg) 

143 if result.returncode != 0: 

144 problem = self._problem(path, result) 

145 raise (GhLoginError if login_fix(self._host) in problem else HeinzelError)(problem) 

146 return _json(result.stdout, path), len(result.stdout) 

147 

148 def _problem(self, path: str, result: GhResult) -> str: # noqa: PLR0911 - one message per kind of failure 

149 """What went wrong, for a person: GitHub's or `gh`'s words, redacted and cleaned, plus what to do.""" 

150 host, stderr = self._host, result.stderr 

151 _LOG.debug("gh api %s failed (exit %s): %s", path, result.returncode, clean_text(redact(stderr[-2000:]))) 

152 # gh's own line ends with the status; GitHub's message before it could quote one: take the last. 

153 statuses = _STATUS.findall(stderr) 

154 status = int(statuses[-1]) if statuses else None 

155 text = said(_message(result.stdout) or _first_line(stderr)) 

156 # gh adds "Try authenticating with: gh auth login" to a 401 too: the status decides first. 

157 if status == 401: # noqa: PLR2004 - Unauthorized 

158 return ( 

159 f"GitHub rejected gh's login for {host} (401: {text}); log in again with `{login_fix(host)}`, " 

160 "or replace GH_TOKEN or GH_ENTERPRISE_TOKEN if one is set, as it takes precedence." 

161 ) 

162 if status is None and (result.returncode == NOT_LOGGED_IN or "gh auth login" in stderr): 

163 return f"`gh` is not logged in to {host}; run `{login_fix(host)}`, then try again." 

164 if status == 429 or (status == 403 and "rate limit" in text.lower()): # noqa: PLR2004 - Too Many Requests 

165 return ( 

166 f"GitHub's rate limit is reached for {path} ({status}: {text}); wait a few minutes and try again. " 

167 f"`gh api rate_limit --hostname {host}` shows when it resets." 

168 ) 

169 account = f"the account gh is logged in with (`gh auth status --hostname {host}`)" 

170 if status == 404: # noqa: PLR2004 - Not Found 

171 return f"GitHub has no {path}: check the owner/repo spelling, and that {account} can see it." 

172 if status == 403: # noqa: PLR2004 - Forbidden 

173 return f"GitHub refused {path} (403: {text}); {account} needs read access to its pull requests and issues." 

174 if status is not None: 

175 return f"GitHub answered {path} with {status}: {text}." 

176 return ( 

177 f"`gh api {path}` failed (exit code {result.returncode}): {text}. " 

178 f"Check the network and `gh auth status --hostname {host}`." 

179 )

182def _json(body: bytes, path: str) -> Any: # noqa: ANN401 - free-form JSON 

183 try: 

184 return json.loads(body) 

185 except ValueError: 

186 msg = f"`gh api {path}` printed something that is not JSON; is `gh` the GitHub CLI?" 

187 raise HeinzelError(msg) from None

190def _message(body: bytes) -> str: 

191 """GitHub's `message` from an error body `gh` printed, or ""; never the raw body, which could be anything.""" 

192 try: 

193 data = json.loads(body) 

194 except ValueError: 

195 return "" 

196 message = cast("dict[str, Any]", data).get("message") if isinstance(data, dict) else None 

197 return message if isinstance(message, str) else ""

200def _first_line(text: str) -> str: 

201 return next((line for line in text.splitlines() if line.strip()), "")
