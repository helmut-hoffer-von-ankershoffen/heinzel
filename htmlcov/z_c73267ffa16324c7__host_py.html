1"""The machine the prerequisite engine looks at: platform, search path and the only ways it starts programs (ADR 0112). 

2 

3Everything that touches the real machine (a search path, a file mode, a child process, Homebrew's location) goes 

4through one `ToolHost`, so tests describe a machine instead of patching the engine, and no test can reach the 

5Homebrew of the machine it runs on. 

6""" 

7 

8import os 

9import shutil 

10import sys 

11from collections.abc import Callable, MutableMapping 

12from dataclasses import dataclass, field 

13from pathlib import Path 

14 

15from heinzel.kernel import ProcessExecute, ProcessOutcome, run_bounded_process, subprocess_environment 

16 

17BREW_LOCATIONS = ("/opt/homebrew/bin/brew", "/usr/local/bin/brew", "/home/linuxbrew/.linuxbrew/bin/brew") 

18"""Where Homebrew installs itself; its directory is often missing from the `PATH` of a launchd or GUI process.""" 

19OUTPUT_LIMIT = 65536 

20 

21 

22def _which(name: str, path: str) -> str | None: 

23 return shutil.which(name, path=path) 

24 

25 

26def _is_executable(path: str) -> bool: 

27 candidate = Path(path) 

28 return candidate.is_file() and os.access(candidate, os.X_OK) 

29 

30 

31def _same_file(path: str, reference: str) -> bool: 

32 """Compare physical file identity without execution; the service handles filesystem lookup failures.""" 

33 return Path(path).samefile(reference) 

34 

35 

36@dataclass(frozen=True, slots=True, kw_only=True) 

37class ToolHost: 

38 """A machine as the engine sees it; every field has a real default and a fake in tests.""" 

39 

40 platform: str = sys.platform 

41 environment: MutableMapping[str, str] = field(default_factory=lambda: os.environ) 

42 execute: ProcessExecute = run_bounded_process 

43 which: Callable[[str, str], str | None] = _which 

44 is_executable: Callable[[str], bool] = _is_executable 

45 same_file: Callable[[str, str], bool] = _same_file 

46 brew_locations: tuple[str, ...] = BREW_LOCATIONS 

47 

48 @classmethod 

49 def system(cls) -> "ToolHost": 

50 """The real machine; the one place a test suite replaces to keep the real Homebrew out of reach.""" 

51 return cls() 

52 

53 @property 

54 def search_path(self) -> str: 

55 """The `PATH` programs are looked up in, including directories added by `add_to_path`.""" 

56 return self.environment.get("PATH") or os.defpath 

57 

58 def add_to_path(self, directory: str, *, first: bool = False) -> None: 

59 """Add a process-local directory, promoting it when an incompatible executable would shadow it.""" 

60 entries = [entry for entry in self.search_path.split(os.pathsep) if entry] 

61 if first: 

62 self.environment["PATH"] = os.pathsep.join([directory, *(entry for entry in entries if entry != directory)]) 

63 elif directory not in entries: 

64 self.environment["PATH"] = os.pathsep.join([*entries, directory]) 

65 

66 def run(self, command: list[str], timeout: float, flags: tuple[tuple[str, str], ...] = ()) -> ProcessOutcome: 

67 """Run a command without a shell or input, from the home directory, with `flags` added to the environment. 

68 

69 Raises: 

70 OSError: If the program cannot be started. 

71 """ 

72 env = subprocess_environment(self.environment) 

73 env.update(flags) 

74 return self.execute( 

75 command, 

76 input=None, 

77 cwd=self._working_directory(), 

78 env=env, 

79 timeout=timeout, 

80 limit=OUTPUT_LIMIT, 

81 ) 

82 

83 def _working_directory(self) -> Path: 

84 """The home directory, or the root when it does not exist, so a broken HOME never hides a tool.""" 

85 home = Path(self.environment.get("HOME") or Path.home()) 

86 return home if home.is_dir() else Path(home.anchor or "/") 

87 

88 def find_brew(self) -> str | None: 

89 """Homebrew's absolute executable on `PATH` or in its standard locations; None without a safe candidate.""" 

90 found = self.which("brew", self.search_path) 

91 for candidate in ([found] if found else []) + list(self.brew_locations): 

92 if Path(candidate).is_absolute() and self.is_executable(candidate): 

93 return candidate 

94 return None
