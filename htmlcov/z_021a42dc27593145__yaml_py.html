1"""Reading and writing the YAML data files of a team repository: `hz.yaml`, `team/roster.yaml`, `people/*.yaml`. 

2 

3Repository content is untrusted: anyone who can push to the team repository writes it. It is read 

4with the same rules as a contract's frontmatter (ADR 0021 §1): YAML 1.2 core scalars only, no anchors or aliases, no 

5tags but the core scalar ones, bounded size, depth and node count, no control characters. The workflow module's 

6loader can't be imported (modules don't import each other's private files), so the rules are repeated here; keep 

7them in step. 

8""" 

9 

10import re 

11from typing import TYPE_CHECKING, Any, cast 

12 

13import yaml 

14from yaml.nodes import MappingNode, Node, ScalarNode, SequenceNode 

15 

16if TYPE_CHECKING: 

17 from collections.abc import Callable, Iterator 

18 

19_STR = "tag:yaml.org,2002:str" 

20_BOOL = "tag:yaml.org,2002:bool" 

21_INT = "tag:yaml.org,2002:int" 

22_FLOAT = "tag:yaml.org,2002:float" 

23_NULL = "tag:yaml.org,2002:null" 

24_CORE_TAGS = frozenset({_STR, _BOOL, _INT, _FLOAT, _NULL}) 

25MAX_BYTES = 256 * 1024 

26"""Data files larger than this aren't read: a manifest or a member list is small.""" 

27MAX_DEPTH = 32 

28MAX_NODES = 10_000 

29CONTROLS = re.compile(r"[\x00-\x08\x0b-\x1f\x7f-\x9f]") 

30"""C0 controls but tab and line feed, DEL and C1 controls: what a terminal acts on rather than shows.""" 

31 

32 

33class YamlError(ValueError): 

34 """A data file that isn't the YAML hz reads; the message says what is wrong, and where.""" 

35

37class _Loader(yaml.SafeLoader): 

38 """PyYAML's safe loader resolving plain scalars like YAML 1.2's core schema: `on`, `yes` and dates stay text.""" 

39 

40 

41_Loader.yaml_implicit_resolvers = { 

42 first: [(tag, regexp) for tag, regexp in resolvers if tag == _NULL] 

43 for first, resolvers in yaml.SafeLoader.yaml_implicit_resolvers.items() 

44} 

45_CORE_SCALARS = ( 

46 (_BOOL, r"^(?:true|True|TRUE|false|False|FALSE)$", "tTfF"), 

47 (_INT, r"^[-+]?(?:0|[1-9][0-9]*)$", "-+0123456789"), 

48 (_FLOAT, r"^[-+]?(?:0|[1-9][0-9]*)\.[0-9]+$", "-+0123456789"), 

49) 

50for _tag, _pattern, _first in _CORE_SCALARS: 

51 _Loader.add_implicit_resolver(_tag, re.compile(_pattern), list(_first)) # pyright: ignore[reportUnknownMemberType] - PyYAML stubs

54def load(text: str) -> dict[str, Any]: 

55 """Parse a YAML mapping within the limits above. 

56 

57 Raises: 

58 YamlError: If the text is too large, holds control characters, doesn't parse, breaks a limit, has duplicate or 

59 non-text keys, or isn't a mapping. 

60 """ 

61 if len(text.encode()) > MAX_BYTES: 

62 msg = f"larger than {MAX_BYTES // 1024} KiB" 

63 raise YamlError(msg) 

64 if CONTROLS.search(text.replace("\r\n", "\n")): 

65 msg = "holds control characters; only tabs and line breaks may appear" 

66 raise YamlError(msg) 

67 try: 

68 _scan(text) 

69 loader = _Loader(text) 

70 try: 

71 node = loader.get_single_node() 

72 data: object = _build(loader, node) if node is not None else {} 

73 finally: 

74 loader.dispose() 

75 except yaml.MarkedYAMLError as exc: 

76 mark = exc.problem_mark or exc.context_mark 

77 where = f" (line {mark.line + 1})" if mark else "" 

78 msg = f"doesn't parse{where}: {exc.problem or exc}" 

79 raise YamlError(msg) from exc 

80 except yaml.YAMLError as exc: 

81 msg = f"doesn't parse: {exc}" 

82 raise YamlError(msg) from exc 

83 except ValueError as exc: # PyYAML's scanner lets an escape past U+10FFFF (`"\U00110000"`) escape as chr()'s error 

84 msg = f"doesn't parse: an escape names no character ({exc})" 

85 raise YamlError(msg) from exc 

86 if not isinstance(data, dict): 

87 msg = "must be a mapping of keys to values" 

88 raise YamlError(msg) 

89 return cast("dict[str, Any]", data)

92def dump(data: object, *, header: str = "") -> str: 

93 """YAML text for plain data, keys in the given order, preceded by `header` as comment lines.""" 

94 comments = "".join(f"# {line}\n" if line else "#\n" for line in header.splitlines()) 

95 body = yaml.safe_dump(data, sort_keys=False, allow_unicode=True, default_flow_style=False) 

96 return comments + body

99def _scan(text: str) -> None: 

100 """Refuse anchors, aliases, non-core tags and oversized structures before any node is built. 

101 

102 Raises: 

103 YamlError: At the first one. 

104 """ 

105 depth = 0 

106 nodes = 0 

107 parse = cast("Callable[..., Iterator[yaml.Event]]", yaml.parse) # pyright: ignore[reportUnknownMemberType] - PyYAML stubs leave its Loader untyped 

108 for event in parse(text, Loader=_Loader): 

109 line = (event.start_mark.line + 1) if event.start_mark is not None else 1 

110 if isinstance(event, yaml.AliasEvent) or getattr(event, "anchor", None) is not None: 

111 msg = f"anchors and aliases aren't allowed (line {line})" 

112 raise YamlError(msg) 

113 tag = getattr(event, "tag", None) 

114 if (isinstance(event, yaml.CollectionStartEvent) and tag is not None) or ( 

115 isinstance(event, yaml.ScalarEvent) and tag not in {None, "!"} and tag not in _CORE_TAGS 

116 ): 

117 msg = f"the tag {tag} isn't allowed (line {line})" 

118 raise YamlError(msg) 

119 if isinstance(event, yaml.NodeEvent): 

120 nodes += 1 

121 if nodes > MAX_NODES: 

122 msg = f"more than {MAX_NODES} values" 

123 raise YamlError(msg) 

124 if isinstance(event, yaml.CollectionStartEvent): 

125 depth += 1 

126 if depth > MAX_DEPTH: 

127 msg = f"nested deeper than {MAX_DEPTH} levels (line {line})" 

128 raise YamlError(msg) 

129 elif isinstance(event, yaml.CollectionEndEvent): 

130 depth -= 1

133def _build(loader: _Loader, node: Node) -> Any: # noqa: ANN401 - YAML 

134 if isinstance(node, MappingNode): 

135 mapping: dict[str, Any] = {} 

136 for key_node, value_node in node.value: 

137 line = key_node.start_mark.line + 1 

138 if not isinstance(key_node, ScalarNode) or key_node.tag != _STR: 

139 msg = f"keys must be text (line {line})" 

140 raise YamlError(msg) 

141 key = _plain(cast("str", key_node.value), line) 

142 if key in mapping: 

143 msg = f"duplicate key {key!r} (line {line})" 

144 raise YamlError(msg) 

145 mapping[key] = _build(loader, value_node) 

146 return mapping 

147 if isinstance(node, SequenceNode): 

148 return [_build(loader, item) for item in node.value] 

149 value = loader.construct_object(node, deep=True) # pyright: ignore[reportUnknownMemberType] - PyYAML stubs leave node untyped 

150 return _plain(value, node.start_mark.line + 1) if isinstance(value, str) else value

153def _plain(text: str, line: int) -> str: 

154 r"""`text`, unless an escape such as `\e` decoded into a control character the text scan can't see. 

155 

156 Raises: 

157 YamlError: If it holds a control character other than a tab or line break. 

158 """ 

159 if CONTROLS.search(text): 

160 msg = f"control characters aren't allowed (line {line})" 

161 raise YamlError(msg) 

162 return text
