1"""Themes: how the web app looks, as data a plugin ships (ADR 0044). 

2 

3A theme is a name, a title, the colours of the web app's tokens for light and dark, two font stacks, its type scale, 

4density and corner radii, a border width, and optionally font files of its own. Every value ends up as a CSS custom 

5property, so each is checked against a grammar that can't leave its declaration: hex colours, a CSS length, an integer 

6weight, and font stacks of family names from a small character set. A font file is a `.woff2` inside the theme's own 

7directory, checked at import and again when the web head serves it from its own origin: the web head's CSP 

8(`font-src 'self'`) loads no remote font. A logo is SVG files in the same directory, held to the same path rules and 

9refused if they could run anything or load anything from outside themselves. 

10"""

122def font_stack_problem(stack: str) -> str | None: 

123 """Why `stack` isn't a safe font-family stack, or None. 

124 

125 A stack is printable ASCII: family names separated by commas. A name is unquoted words that start with a letter 

126 (one leading dash allowed) and hold letters, digits and dashes, or a double-quoted name of letters, digits, 

127 spaces, dots, underscores and dashes. Nothing else passes: no `;`, `}`, `(` (so no `url(` or `var(`), no 

128 backslash, no single quote, no quote inside a name, no newline, and no CSS-wide keyword such as `inherit`. 

129 """ 

130 if not (stack.isascii() and stack.isprintable()): 

131 return "a font stack is one line of printable ASCII characters" 

132 if not stack.strip(): 

133 return "a font stack names at least one family, e.g. system-ui, sans-serif" 

134 if len(stack) > MAX_FONT_STACK: 

135 return f"a font stack is at most {MAX_FONT_STACK} characters" 

136 for family in stack.split(","): 

137 if family.strip().lower() in CSS_WIDE: 

138 return f"{family.strip()!r} is a CSS keyword, not a font family; quote it if a font has that name" 

139 if not _FAMILY.fullmatch(family.strip()): 

140 return ( 

141 f"{family.strip()!r} isn't a font family name: use letters, digits, spaces and dashes, and double " 

142 'quotes around a name with other characters, e.g. "SF Mono"' 

143 ) 

144 return None

435def _channel(value: int) -> float:

440def _rgb(colour: str) -> tuple[int, int, int]: 

441 digits = colour.removeprefix("#") 

442 if len(digits) == 3: # noqa: PLR2004 - #rgb 

443 digits = "".join(digit * 2 for digit in digits)

447def _luminance(rgb: tuple[float, float, float]) -> float: 

448 red, green, blue = (_channel(round(part)) for part in rgb) 

449 return 0.2126 * red + 0.7152 * green + 0.0722 * blue

452def contrast(first: str, second: str) -> float: 

453 """The WCAG contrast ratio of two hex colours, 1 to 21.""" 

454 lighter, darker = sorted((_luminance(_rgb(first)), _luminance(_rgb(second))), reverse=True) 

455 return (lighter + 0.05) / (darker + 0.05)

458def _blend(top: str, bottom: str, opacity: float) -> str: 

459 opacity = round(opacity * 100) / 100 # as the web app draws it: `color-mix` at a whole percent 

460 mixed = (t * opacity + b * (1 - opacity) for t, b in zip(_rgb(top), _rgb(bottom), strict=True)) 

461 return "#" + "".join(f"{round(part):02x}" for part in mixed)

464def _surfaces(colours: ThemeColours) -> list[tuple[str, str]]: 

465 """What the main area's text sits on, by name: the background, the tint, each backdrop stop over the background.""" 

466 surfaces = [("background", colours.background)] 

467 if colours.main_tint: 

468 surfaces.append(("main_tint", colours.main_tint)) 

469 if backdrop := colours.backdrop: 

470 for stop in backdrop.stops: 

471 value = stop if stop.startswith("#") else getattr(colours, stop) or colours.background 

472 surfaces.append((f"backdrop stop {stop}", _blend(value, colours.background, backdrop.opacity))) 

473 return surfaces

477 """Where text in `colours` falls under 4.5:1, in words. 

478 

479 `foreground` and `muted_foreground` are checked on cards, the sidebar, popovers and every main surface. 

480 """ 

481 pairs = [ 

482 (text, surface, getattr(colours, text), getattr(colours, surface)) 

483 for surface in ("card", "sidebar", "popover") 

484 for text in ("foreground", "muted_foreground") 

485 ] 

486 for surface, value in _surfaces(colours):

490 ] 

491 return [ 

492 f"{text} on {surface} is {ratio:.2f}:1, under {MIN_CONTRAST}:1" 

493 for text, surface, first, second in pairs 

494 if (ratio := contrast(first, second)) < MIN_CONTRAST

575 """Refuse a theme whose text is hard to read, in either mode, as it is or in any variant (ADR 0060).

713def font_file(directory: Path, font: ThemeFont) -> Path: 

714 """The resolved path of `font` in `directory`, after checking it is a woff2 file of at most 2 MB inside it. 

715 

716 Symlinks are resolved before the check, so one pointing out of the theme's directory is refused. 

717 

718 Raises: 

719 ValueError: If the file is missing, leads out of the directory, isn't a woff2 file or is too large. 

720 """ 

721 where = f"Font file {font.file!r} of {directory}" 

722 try: 

723 root = directory.resolve(strict=True) 

724 path = (root / font.file).resolve(strict=True) 

725 except OSError as exc: 

726 msg = f"{where} doesn't exist; its path is relative to the theme file's directory" 

727 raise ValueError(msg) from exc 

728 if not path.is_relative_to(root): 

729 msg = f"{where} leads out of the theme's directory; keep font files beside the theme file" 

730 raise ValueError(msg) 

731 if not path.is_file(): 

732 msg = f"{where} isn't a file" 

733 raise ValueError(msg) 

734 if path.stat().st_size > MAX_FONT_BYTES: 

735 msg = f"{where} is larger than {MAX_FONT_BYTES // 1024 // 1024} MB; subset the font" 

736 raise ValueError(msg) 

737 with path.open("rb") as handle: 

738 if handle.read(len(WOFF2_SIGNATURE)) != WOFF2_SIGNATURE: 

739 msg = f"{where} isn't a woff2 file; convert it, e.g. with woff2_compress" 

740 raise ValueError(msg) 

741 return path

744def read_font(directory: Path, font: ThemeFont) -> bytes: 

745 """The content of `font`, checked by `font_file` and again on the bytes read, so a file swapped meanwhile is caught. 

746 

747 Raises: 

748 ValueError: If the file doesn't pass the checks of `font_file`, can't be read, or changed meanwhile. 

749 """ 

750 path = font_file(directory, font) 

751 content = _read_regular(path, MAX_FONT_BYTES, f"Font file {font.file!r} of {directory}") 

752 if len(content) > MAX_FONT_BYTES or not content.startswith(WOFF2_SIGNATURE): 

753 msg = f"Font file {font.file!r} of {directory} changed into something that isn't a woff2 file of up to 2 MB" 

754 raise ValueError(msg) 

755 return content

758def _read_regular(path: Path, limit: int, where: str) -> bytes: 

759 """Up to `limit + 1` bytes of `path`, through one descriptor that follows no symlink and must be a regular file. 

760 

761 Raises: 

762 ValueError: If `path` is a symlink, a FIFO or anything else that isn't a regular file, or can't be read. 

763 """ 

764 try: 

765 # One descriptor for the check and the read: no symlink swapped in, no FIFO to block on, a regular file. 

766 descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK) 

767 with os.fdopen(descriptor, "rb") as handle: 

768 if not stat.S_ISREG(os.fstat(handle.fileno()).st_mode): 

769 msg = f"{where} isn't a file" 

770 raise ValueError(msg) 

771 return handle.read(limit + 1) 

772 except OSError as exc: 

773 msg = f"{where} can't be read: {exc}" 

774 raise ValueError(msg) from exc

778 """An XML name without its `{namespace}`, lowercase.""" 

779 return name.rpartition("}")[2].lower()

782def _svg_text_problem(text: str) -> str | None: 

783 """Why the text of an SVG file could run or load something, or None; checked before it is parsed.""" 

784 lowered = text.lower() 

785 for marker, problem in ( 

786 ("<!doctype", "declares a DOCTYPE; entities aren't allowed"), 

787 ("<!entity", "declares an entity; entities aren't allowed"), 

788 ("<script", "holds a <script>"), 

789 ("<foreignobject", "holds a <foreignObject>"), 

790 ("xml-stylesheet", "links a stylesheet"), 

791 ("@import", "imports a stylesheet"), 

792 ("image-set(", "loads an image (image-set)"), 

793 ): 

794 if marker in lowered: 

795 return problem 

796 for match in re.finditer(r"url\(\s*['\"]?\s*(.?)", lowered): 

797 if match.group(1) != "#": 

798 return "refers to something outside itself (url(...)); only url(#id) is allowed" 

799 return None

801 

802def _svg_element_problem(element: ET.Element) -> str | None: 

803 """Why one element of an SVG file could run or load something, or None.""" 

804 tag = _local(element.tag) 

805 if tag in {"script", "foreignobject"}: 

806 return f"holds a <{tag}>" 

807 for name, value in element.attrib.items(): 

808 attribute = _local(name) 

809 if attribute.startswith("on"): 

810 return f"has an event handler ({attribute}=)" 

811 if attribute in _REFERENCES and not value.strip().startswith("#"): 

812 return f"refers to something outside itself ({attribute}={value[:40]!r}); only #id is allowed" 

813 if tag in {"set", "animate"} and attribute == "attributename": 

814 target = value.strip().lower().rpartition(":")[2] 

815 if target in _REFERENCES or target.startswith("on"): 

816 return f"animates {target}, which could point outside the file" 

817 return _svg_css_problem(tag, element) 

818 

819 

820def _svg_css_problem(tag: str, element: ET.Element) -> str | None: 

821 """Why the decoded text and style of one element could load something, or None.""" 

822 styles = [value for name, value in element.attrib.items() if _local(name) == "style"] 

823 if tag == "style": 

824 styles.append(element.text or "") 

825 if any("\\" in style for style in styles): 

826 return f"escapes characters in a {'<style>' if tag == 'style' else 'style'} (\\), which could hide a url(...)" 

827 # Character references are decoded by now: `&#117;rl(` is `url(` here, so the text checks run again. 

828 decoded = [element.text, element.tail, *element.attrib.values()] 

829 return next((problem for text in decoded if text and (problem := _svg_text_problem(text))), None)

832def svg_problem(content: bytes) -> str | None: 

833 """Why `content` isn't a safe SVG logo, or None. 

834 

835 A logo is UTF-8 XML of at most 200 KB whose root is `<svg>`, without a DOCTYPE or entities, a `<script>`, a 

836 `<foreignObject>`, an event-handler attribute (`on...=`), or a reference (`href`, `xlink:href`, `src`, `url(...)`) 

837 to anything but an element of the same file (`#id`): no external file, no `data:` URI. 

838 """ 

839 if len(content) > MAX_LOGO_BYTES: 

840 return f"is larger than {MAX_LOGO_BYTES // 1024} KB; simplify it, e.g. with svgo" 

841 try: 

842 text = content.decode("utf-8") 

843 except UnicodeDecodeError: 

844 return "isn't UTF-8 text" 

845 return _svg_text_problem(text) or _svg_tree_problem(content)

848def _svg_tree_problem(content: bytes) -> str | None: 

849 """Why the parsed elements of an SVG file aren't a safe logo, or None; its text passed `_svg_text_problem`.""" 

850 try: 

851 root = ET.fromstring(content) # noqa: S314 - DOCTYPE and entities are refused before: nothing expands 

852 except ET.ParseError as exc: 

853 return f"isn't well-formed XML: {exc}" 

854 if root.tag not in {"svg", f"{_SVG}svg"}: 

855 return "isn't an SVG image: its root element isn't <svg>" 

856 return next((problem for element in root.iter() if (problem := _svg_element_problem(element))), None)

1001 Raises: 

1002 ValueError: If a file isn't a valid theme, or a directory holds none. 

1003 OSError: If a file can't be read. 

1004 """ 

1005 if isinstance(source, Theme): 

1006 return ((source, None),) 

1007 files = sorted(source.glob(f"*{THEME_SUFFIX}")) if source.is_dir() else [source] 

1008 if not files: 

1009 msg = f"{source} holds no theme files (*{THEME_SUFFIX})" 

1010 raise ValueError(msg) 

1011 themes: list[tuple[Theme, Path | None]] = [] 

1012 for file in files: 

1013 if file.suffix != THEME_SUFFIX: 

1014 msg = f"Theme file {file} must end in {THEME_SUFFIX}" 

1015 raise ValueError(msg) 

1016 try: 

1017 themes.append((Theme.model_validate(tomllib.loads(file.read_text(encoding="utf-8"))), file.parent)) 

1018 except tomllib.TOMLDecodeError as exc: 

1019 msg = f"Theme file {file} isn't valid TOML: {exc}" 

1020 raise ValueError(msg) from exc 

1021 except ValidationError as exc: 

1022 msg = f"Theme file {file} isn't a valid theme: {problems_of(exc)}" 

1023 raise ValueError(msg) from exc 

1024 return tuple(themes)
