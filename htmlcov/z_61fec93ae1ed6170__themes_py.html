1"""Themes: how the web app looks, as data a plugin ships (ADR 0044). 

2 

3A theme is a name, a title, the colours of the web app's tokens for light and dark, two font stacks, a corner radius 

4and a border width. Every value ends up as a CSS custom property, so each is checked against a grammar that can't 

5leave its declaration: hex colours, a CSS length, and font stacks of family names from a small character set. Fonts 

6are those installed on the person's machine: the web head's CSP (`font-src 'self'`) loads no remote font. 

7"""

22MAX_NAME = 64 

23MAX_TITLE = 64 

24MAX_FONT_STACK = 300 

25NAME = r"^[a-z](?:-?[a-z0-9])*$" 

26HEX = r"^#(?:[0-9a-fA-F]{3}|[0-9a-fA-F]{6})$" 

27LENGTH = r"^(?:0|(?:[0-9]{1,3}(?:\.[0-9]{1,4})?|\.[0-9]{1,4})(?:px|rem|em))$" 

28"""A CSS length: `0`, or a number with `px`, `rem` or `em`, e.g. `0.5rem` or `1px`; no `calc()`, no `var()`.""" 

29_FAMILY = re.compile(r'^(?:-?[A-Za-z][A-Za-z0-9-]*(?: [A-Za-z][A-Za-z0-9-]*)*|"[A-Za-z0-9][A-Za-z0-9 ._-]*")$') 

30"""One family in a stack: unquoted identifiers (`system-ui`, `-apple-system`) or a double-quoted name (`"SF Mono"`).""" 

31CSS_WIDE = frozenset({"inherit", "initial", "unset", "revert", "revert-layer", "default"}) 

32"""Keywords CSS reserves; unquoted inside a stack they make the whole declaration invalid.""" 

33_RULES = { 

34 "string_pattern_mismatch": { 

35 "name": "lowercase words joined by single dashes, e.g. acme", 

36 "radius": "a CSS length: 0, or a number with px, rem or em, e.g. 0.75rem", 

37 "border_width": "a CSS length: 0, or a number with px, rem or em, e.g. 1px", 

38 }, 

39}

47def font_stack_problem(stack: str) -> str | None: 

48 """Why `stack` isn't a safe font-family stack, or None. 

49 

50 A stack is printable ASCII: family names separated by commas. A name is unquoted words that start with a letter 

51 (one leading dash allowed) and hold letters, digits and dashes, or a double-quoted name of letters, digits, 

52 spaces, dots, underscores and dashes. Nothing else passes: no `;`, `}`, `(` (so no `url(` or `var(`), no 

53 backslash, no single quote, no quote inside a name, no newline, and no CSS-wide keyword such as `inherit`. 

54 """ 

55 if not (stack.isascii() and stack.isprintable()): 

56 return "a font stack is one line of printable ASCII characters" 

57 if not stack.strip(): 

58 return "a font stack names at least one family, e.g. system-ui, sans-serif" 

59 if len(stack) > MAX_FONT_STACK: 

60 return f"a font stack is at most {MAX_FONT_STACK} characters" 

61 for family in stack.split(","): 

62 if family.strip().lower() in CSS_WIDE: 

63 return f"{family.strip()!r} is a CSS keyword, not a font family; quote it if a font has that name" 

64 if not _FAMILY.fullmatch(family.strip()): 

65 return ( 

66 f"{family.strip()!r} isn't a font family name: use letters, digits, spaces and dashes, and double " 

67 'quotes around a name with other characters, e.g. "SF Mono"' 

68 ) 

69 return None

72def _font_stack(value: str) -> str: 

73 if problem := font_stack_problem(value): 

74 raise ValueError(problem) 

75 return ", ".join(family.strip() for family in value.split(","))

138 def __post_init__(self) -> None: 

139 """Check the theme again, so one built with `model_construct` can't skip the grammar. 

140 

141 Raises: 

142 ValueError: If the theme isn't valid, is called `default`, or the let isn't named after it. 

143 """ 

144 try: 

145 theme = Theme.model_validate(self.theme.model_dump()) 

146 except ValidationError as exc: 

147 msg = f"Theme {self.theme.name!r} isn't valid: {problems_of(exc)}" 

148 raise ValueError(msg) from exc 

149 if theme.name == DEFAULT: 

150 msg = f"No plugin can offer a theme named {DEFAULT!r}: that is the built-in one" 

151 raise ValueError(msg) 

152 if self.name != theme.name: 

153 msg = f"A theme let is named after its theme: {theme.name!r}, not {self.name!r}" 

154 raise ValueError(msg) 

155 object.__setattr__(self, "theme", theme)

158def problems_of(exc: ValidationError) -> str: 

159 """Each problem of a theme as `field: rule`, in words a person editing a theme file reads.""" 

160 problems: list[str] = [] 

161 for error in exc.errors(): 

162 where = ".".join(map(str, error["loc"])) or "theme" 

163 field = str(error["loc"][-1]) if error["loc"] else "" 

164 if error["type"] == "string_pattern_mismatch": 164 ↛ 167line 164 didn't jump to line 167 because the condition on line 164 was always true

165 rule = _RULES["string_pattern_mismatch"].get(field, "a hex colour, #rgb or #rrggbb, e.g. #0f766e") 

166 else: 

167 rule = error["msg"].removeprefix("Value error, ") 

168 problems.append(f"{where}: {rule}") 

169 return "; ".join(problems)

172def themes_in(source: Path | Theme) -> tuple[Theme, ...]: 

173 """The themes of `source`: a `Theme`, a `*.toml` file of one, or a directory of them, read now. 

174 

175 Raises: 

176 ValueError: If a file isn't a valid theme, or a directory holds none. 

177 OSError: If a file can't be read. 

178 """ 

179 if isinstance(source, Theme): 

180 return (source,) 

181 files = sorted(source.glob(f"*{THEME_SUFFIX}")) if source.is_dir() else [source] 

182 if not files: 

183 msg = f"{source} holds no theme files (*{THEME_SUFFIX})" 

184 raise ValueError(msg) 

185 themes: list[Theme] = [] 

186 for file in files: 

187 if file.suffix != THEME_SUFFIX: 

188 msg = f"Theme file {file} must end in {THEME_SUFFIX}" 

189 raise ValueError(msg) 

190 try: 

191 themes.append(Theme.model_validate(tomllib.loads(file.read_text(encoding="utf-8")))) 

192 except tomllib.TOMLDecodeError as exc: 

193 msg = f"Theme file {file} isn't valid TOML: {exc}" 

194 raise ValueError(msg) from exc 

195 except ValidationError as exc: 

196 msg = f"Theme file {file} isn't a valid theme: {problems_of(exc)}" 

197 raise ValueError(msg) from exc 

198 return tuple(themes)
