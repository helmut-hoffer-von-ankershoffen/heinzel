171 @property

176 @model_validator(mode="after") 

177 def _method_and_arguments(self) -> Self: 

178 if (upstream := self.upstream) is not None: 

179 return self._contract_input(upstream) 

180 if self.stage is not None: 

181 raise at(("stage",), "only an input that reads a contract's output has a stage: remove `stage`") 

182 return self._capability_input()

184 def _capability_input(self) -> Self: 

185 spec = CAPABILITIES.get(self.capability) 

186 if spec is None or self.capability == RUNNER: 

187 known = ", ".join(sorted(name for name in CAPABILITIES if name != RUNNER)) 

188 raise at(("read",), f"{self.capability!r} is not a capability an input can read; known: {known}") 

189 mode = spec.methods.get(self.method) 

190 if mode is None: 

191 methods = ", ".join(f"{self.capability}.{m}" for m, mo in spec.methods.items() if mo == "read") 

192 raise at(("read",), f"{self.capability} has no method {self.method!r}; read methods: {methods}") 

193 if mode != "read": 

194 raise at(("read",), f"{self.read} writes; inputs only read, and writes happen in publish operations") 

195 hints = get_type_hints(getattr(spec.protocol, self.method)) 

196 parameters = [p for p in inspect.signature(getattr(spec.protocol, self.method)).parameters if p != "self"] 

197 windows = [p for p in parameters if hints.get(p) is Window] 

198 named = [p for p in parameters if p not in windows] 

199 if windows and self.window is None: 

200 raise at(("window",), f"{self.read} lists changes in a time window: set `window`, e.g. 7d") 

201 if not windows and self.window is not None: 

202 raise at(("window",), f"{self.read} takes no time window: remove `window`") 

203 if unknown := sorted(set(self.args) - set(named)): 

204 raise at(("args", unknown[0]), f"{self.read} has no argument {unknown[0]!r}; it takes {', '.join(named)}") 

205 if missing := [p for p in named if p not in self.args]: 

206 raise at(("args",), f"{self.read} needs `args: {{{missing[0]}: ...}}`") 

207 for name in named: 

208 if problem := self._argument_problem(spec, name, hints[name]): 

209 raise problem 

210 return self

212 def _argument_problem(self, spec: Capability, name: str, hint: Any) -> PydanticCustomError | None: # noqa: ANN401 - a type hint 

213 """Why the argument `name` lacks the method's type or fails the capability's own check of it; None if not.""" 

214 try: 

215 TypeAdapter(hint).validate_python(self.args[name]) 

216 except ValidationError as exc: 

217 return at(("args", name), f"{name}: {exc.errors()[0]['msg']}") 

218 if check := spec.checks.get((self.method, name)): 

219 try: 

220 check(str(self.args[name])) # a str: validated above 

221 except ValueError as exc: 

222 return at(("args", name), f"{name}: {exc}") 

223 return None

225 def _contract_input(self, upstream: tuple[str, str, str]) -> Self: 

226 if len(upstream[1]) > MAX_NAME_LENGTH: 

227 raise at(("read",), f"contract names are at most {MAX_NAME_LENGTH} characters") 

228 if self.args or self.window is not None: 

229 key = "args" if self.args else "window" 

230 raise at((key,), f"{self.read} is a contract's output: it takes no `{key}`; remove it") 

231 if self.stage != "draft": 

232 raise at( 

233 ("stage",), 

234 f"{self.read}: nothing is published yet, so set `stage: draft` to read its newest draft; " 

235 "published truth comes with `hz inbox`", 

236 ) 

237 return self 

238 

239 def arguments(self, run: datetime) -> dict[str, Any]: 

240 """The method's keyword arguments for a run at `run`: `args`, plus the window back from or ahead of it.""" 

241 if self.upstream is not None: 

242 return {} 

243 spec = CAPABILITIES[self.capability] 

244 hints = get_type_hints(getattr(spec.protocol, self.method)) 

245 arguments: dict[str, Any] = dict(self.args) 

246 for name, hint in hints.items(): 

247 if hint is Window and self.window is not None: 

248 arguments[name] = self.window.at(run) 

249 return arguments

367 @model_validator(mode="after") 

368 def _consistent(self) -> Self: 

369 if self.schedule is not None and (problem := cron_problem(self.schedule)): 

370 raise at(("schedule",), problem) 

371 if not timedelta(0) < self.timeout <= MAX_TIMEOUT: 

372 raise at(("timeout",), f"timeout must be at most {format_duration(MAX_TIMEOUT)}") 

373 granted = self.requirements() 

374 for name, spec in self.inputs.items(): 

375 if spec.upstream is None and spec.capability not in granted: 

376 raise at(("inputs", name, "read"), f"{spec.read} needs `{spec.capability}:read` in `hz.requires`") 

377 if len(self.outputs) != 1: 

378 raise at(("outputs",), "name exactly one output; several outputs come with output facets") 

379 for index, publish in enumerate(self.publishes): 

380 if publish.output not in self.outputs: 

381 raise at(("publishes", index, "output"), f"no output named {publish.output!r} in `hz.outputs`") 

382 return self

384 def requirements(self) -> dict[str, Requirement]: 

385 """What it requires, by capability, `agent-runner:read` included; checked when the model is validated.""" 

386 return self._parse_requirements() 

387 

388 def _parse_requirements(self) -> dict[str, Requirement]: 

389 found: dict[str, Requirement] = {} 

390 for index, text in enumerate(self.requires): 

391 try: 

392 requirement = Requirement.parse(text) 

393 except ValueError as exc: 

394 raise at(("requires", index), str(exc)) from exc 

395 if requirement.adapter is not None: 

396 raise at( 

397 ("requires", index), 

398 f"{text!r} pins the adapter {requirement.adapter!r}: a contract names capabilities; " 

399 "configuration decides the adapter ([capabilities] in your user file)", 

400 ) 

401 if "write" in requirement.modes: 

402 raise at( 

403 ("requires", index), 

404 f"{text!r} writes: a contract only reads; writes go through `publishes`, whose operation declares " 

405 "write itself", 

406 ) 

407 if requirement.capability in found: 

408 raise at(("requires", index), f"{requirement.capability} is required twice") 

409 found[requirement.capability] = requirement 

410 found.setdefault(RUNNER, Requirement(RUNNER, frozenset({"read"}))) 

411 return found

435def cron_problem(schedule: str) -> str | None: 

436 """Why a cron expression can't schedule a job, or None if it can. 

437 

438 The shape is the kernel's own check, the one `JobLet` applies (five ASCII fields or an alias such as `@daily`), so 

439 a contract that passes here always compiles; croniter then checks the values. 

440 """ 

441 from croniter import CroniterBadDateError, croniter # noqa: PLC0415 - only needed when a contract has a schedule 

442 

443 if problem := schedule_problem(schedule): 

444 return f"invalid {problem}; e.g. 0 7 * * MON" 

445 if not schedule.startswith("@") and not croniter.is_valid(schedule): 

446 return f"invalid schedule {schedule!r}: a field is out of range, e.g. minute 0-59, hour 0-23, weekday MON-SUN" 

447 # The daemon's own never-fires check (`system.daemon`'s schedule module), repeated because modules don't import 

448 # each other's private files; keep the two in step. 

449 try: 

450 croniter(schedule, datetime(2000, 1, 1)).get_next(datetime) # noqa: DTZ001 - wall-clock arithmetic is naive 

451 except CroniterBadDateError: 

452 return f"invalid schedule {schedule!r}: it never fires, e.g. 30 February" 

453 return None
