123@module.operation(name="calendar-reminder-worker", read_only=False, heads={"cli"}) 

124def calendar_reminder_worker(kernel: Kernel) -> None: 

125 """Run the supervised calendar reminder worker until its periodic restart.""" 

126 selected = ReminderProfile.selected() 

127 with lifecycle(selected): 

128 verify_binding() 

129 notifications = kernel.resolve(Notifications) 

130 run_reminders(WorkerReads(lambda: kernel.call(agenda), lambda: lifecycle(selected)), notifications)

148@module.health 

149def profile(service: ProfileService, recommendations: SyncService) -> list[Check]: 

150 """Your profile names your handle, role and time zone; and drift from what your team recommends.""" 

151 found = service.profile() 

152 drift = recommendations.health() 

153 if not found.complete: 

154 return [ 

155 Check( 

156 name="profile", 

157 status=Status.DEGRADED, 

158 detail=f"missing {', '.join(found.missing)}", 

159 fix=f"Run `{found.fix}`", 

160 ), 

161 *drift, 

162 ] 

163 detail = f"{found.handle}, {found.role}, {found.timezone} {found.working_hours}" 

164 return [Check(name="profile", status=Status.UP, detail=detail), *drift]
