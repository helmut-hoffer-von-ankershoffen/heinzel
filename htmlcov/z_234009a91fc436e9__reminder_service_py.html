210def _owned(path: Path, content: str) -> os.stat_result: 

211 try: 

212 descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK) 

213 with os.fdopen(descriptor, encoding="utf-8") as stream: 

214 info = os.fstat(stream.fileno()) 

215 matches = ( 

216 stat.S_ISREG(info.st_mode) 

217 and info.st_uid == os.getuid() 

218 and not info.st_mode & 0o022 

219 and stream.read() == content 

220 ) 

221 except OSError as exc: 

222 msg = "The reminder service file could not be verified safely; it was left alone." 

223 raise HeinzelError(msg) from exc 

224 if not matches: 

225 msg = "The reminder service file is not an unchanged file owned by this profile; it was left alone." 

226 raise HeinzelError(msg) 

227 return info

230def _write(path: Path, content: str) -> None: 

231 path.parent.mkdir(parents=True, exist_ok=True) 

232 descriptor, name = tempfile.mkstemp(prefix=f".{path.name}.", dir=path.parent) 

233 temporary = Path(name) 

234 try: 

235 with os.fdopen(descriptor, "w", encoding="utf-8") as target: 

236 target.write(content) 

237 target.flush() 

238 os.fsync(target.fileno()) 

239 os.link(temporary, path) # never replace an existing unit, even in a race 

240 except FileExistsError as exc: 

241 msg = "The calendar reminder service file appeared; it was not overwritten." 

242 raise HeinzelError(msg) from exc 

243 finally: 

244 temporary.unlink(missing_ok=True)
