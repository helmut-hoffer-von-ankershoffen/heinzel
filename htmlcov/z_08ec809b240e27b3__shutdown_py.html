1"""How `hz ui` stops: the responses that stay open end first, so a Ctrl-C stops it at once and quietly. 

2 

3The notice stream and the terminal keep their requests open for minutes. uvicorn's graceful shutdown waits for them, 

4then cancels them, and each cancellation would land in the log as an "Exception in ASGI application" with its 

5traceback. So the server tells the app it is stopping before it waits (`Stopping`): the stream ends its response and 

6each terminal hangs up its shell and closes its socket with "going away" and a reason the app shows. What is still 

7running when the short backstop runs out (an operation streaming its progress) is cancelled as before; the log 

8records of that cancellation are dropped while stopping, and only then, so a real error still logs while serving. 

9 

10Ctrl-C is how a session of `hz ui` ends, so the server doesn't raise it again once it stopped, as uvicorn does: `serve` 

11returns and the CLI says so in one line. A second Ctrl-C while it stops ends the wait at once, as in uvicorn, and one 

12after that, while the process exits, ends it at once without a traceback. 

13"""

31"""How long open responses get to end before uvicorn cancels them; ours end within a fraction of it.""" 

32SETTLE_SECONDS = 0.1 

33"""How long the app gets to close its terminals itself before uvicorn closes what is left with its own code.""" 

34UVICORN_LOGGER = "uvicorn.error"

37class Stopping: 

38 """Whether the server is stopping, for the responses that would otherwise stay open: the stream, the terminals. 

39 

40 `watch` hands each one an event of its own event loop, so a test's client, which runs each request in a loop of 

41 its own, works like the server. 

42 """ 

43 

44 def __init__(self) -> None: 

45 """Not stopping yet.""" 

46 self._stopped = False 

47 self.interrupted = False 

48 """Whether Ctrl-C (SIGINT) stopped the server, rather than a restart or a test.""" 

49 self._lock = threading.Lock() 

50 self._events: set[tuple[asyncio.AbstractEventLoop, asyncio.Event]] = set() 

51 

52 @property 

53 def stopped(self) -> bool: 

54 """Whether `stop` was called.""" 

55 return self._stopped 

56 

57 def stop(self) -> None: 

58 """Set every watcher's event; one that watches later finds it set at once.""" 

59 with self._lock: 

60 self._stopped = True 

61 events = list(self._events) 

62 for loop, event in events: 

63 with contextlib.suppress(RuntimeError): # its loop is closed: nothing waits there any more 

64 loop.call_soon_threadsafe(event.set) 

65 

66 @contextlib.contextmanager 

67 def watch(self) -> Generator[asyncio.Event]: 

68 """An event of the running loop, set when the server stops. 

69 

70 Yields: 

71 The event; already set when the server stopped before. 

72 """ 

73 entry = (asyncio.get_running_loop(), asyncio.Event()) 

74 with self._lock: 

75 if self._stopped: 

76 entry[1].set() 

77 self._events.add(entry) 

78 try: 

79 yield entry[1] 

80 finally: 

81 with self._lock: 

82 self._events.discard(entry)

85class QuietCancellation(logging.Filter): 

86 """Drops uvicorn's records of responses it cancels while the server stops; every other record passes.""" 

87 

88 def __init__(self, stopping: Stopping) -> None: 

89 """Quiet only once `stopping` is stopped.""" 

90 super().__init__() 

91 self.stopping = stopping 

92 

93 def filter(self, record: logging.LogRecord) -> bool: 

94 """Whether `record` is logged.""" 

95 if not self.stopping.stopped: 

96 return True 

97 if record.getMessage().startswith("Cancel ") and "graceful shutdown exceeded" in record.getMessage(): 

98 return False 

99 return not (record.exc_info and _cancelled(record.exc_info[1])) 

100

102def _cancelled(exc: BaseException | None) -> bool: 

103 """Whether `exc` is a cancellation, alone or as the only kind in a group.""" 

104 if isinstance(exc, BaseExceptionGroup): 

105 group = cast("BaseExceptionGroup[BaseException]", exc) # isinstance leaves the members' type unknown 

106 return all(_cancelled(inner) for inner in group.exceptions) 

107 return isinstance(exc, asyncio.CancelledError)

110def _exit_quietly(sig: int, frame: FrameType | None) -> None: # noqa: ARG001 - a signal handler's signature 

111 """End the process at once, as a third Ctrl-C asks, without a traceback from wherever it was.""" 

112 os._exit(0)

115def server(config: "uvicorn.Config", stopping: Stopping) -> "uvicorn.Server": 

116 """A uvicorn server that tells the app through `stopping` that it stops, before it waits for open responses. 

117 

118 The class is made here, not at import: uvicorn is imported only when serving, and a test can replace 

119 `uvicorn.Server` to record the server. 

120 """ 

121 import uvicorn # noqa: PLC0415 - only needed when serving 

122 

123 quiet = QuietCancellation(stopping) 

124 

125 class _Server(uvicorn.Server): 

126 def run(self, sockets: "list[socket.socket] | None" = None) -> None: 

127 # Around the whole event loop: after a second Ctrl-C, the loop cancels what is left only as it closes. 

128 logger = logging.getLogger(UVICORN_LOGGER) 

129 logger.addFilter(quiet) 

130 try: 

131 super().run(sockets) 

132 finally: 

133 logger.removeFilter(quiet) 

134 if stopping.interrupted and threading.current_thread() is threading.main_thread(): 

135 signal.signal(signal.SIGINT, _exit_quietly) # the process ends now; another Ctrl-C only hurries it 

136 

137 def handle_exit(self, sig: int, frame: FrameType | None) -> None: 

138 # Unlike uvicorn's, this doesn't record SIGINT to raise it again once stopped; the exit code test in 

139 # tests/integration/ui_stop_test.py notices if uvicorn starts raising it some other way. 

140 if sig != signal.SIGINT: 140 ↛ 141line 140 didn't jump to line 141 because the condition on line 140 was never true

141 super().handle_exit(sig, frame) # SIGTERM: uvicorn raises it again once stopped, so the process ends 

142 return 

143 stopping.interrupted = True 

144 self.force_exit = self.should_exit # the second Ctrl-C stops waiting 

145 self.should_exit = True 

146 

147 async def shutdown(self, sockets: "list[socket.socket] | None" = None) -> None: 

148 stopping.stop() 

149 await asyncio.sleep(SETTLE_SECONDS) 

150 await super().shutdown(sockets) 

151 

152 return _Server(config)
