1"""Soundscapes: background sound for the web app, as data a plugin ships, chosen like a theme (ADR 0058). 

2 

3A soundscape is either layered audio, licensed files in the plugin that the web app mixes live (each layer's gain bound 

4to a signal by a curve), or a stream in the official player of a host `STREAM_PROVIDERS` allows. No URL or host is 

5data: a stream names a provider and an id, and the frame origin comes from the table here. Audio files are checked at 

6import and again when the web head serves them, like a theme's fonts: inside the soundscape's directory, a regular 

7file, within the size limits, and of the type its signature says. 

8"""

147 """One layer of a layered soundscape: a looped file whose gain follows a signal through a curve.""" 

148 

149 gain: float = Field(default=1.0, ge=0, le=1, description="Its loudness, 0 to 1, before the curve.") 

150 signal: Literal["none", "daypart", "weather.rain", "weather.wind"] = Field( 

151 default="none", description="The signal its gain follows: `none`, `daypart`, `weather.rain` or `weather.wind`." 

152 ) 

153 curve: tuple[tuple[float, float], ...] | dict[str, float] | None = Field( 

154 default=None, 

155 description="For `weather.*`: 2 to 8 points `[x, y]` in 0..1, x rising, read piecewise linear; for `daypart`: " 

156 "`{dawn, morning, day, dusk, evening, night}` to 0..1, a missing part is 0; none for `none`.", 

157 ) 

158 

159 @model_validator(mode="after") 

160 def _curve(self) -> Self: 

161 if self.signal == "none": 

162 if self.curve is not None: 

163 msg = "a layer without a signal has no curve" 

164 raise ValueError(msg) 

165 return self 

166 if self.curve is None: 

167 msg = f"a layer that follows {self.signal} needs a curve" 

168 raise ValueError(msg) 

169 if self.signal in NUMERIC_SIGNALS: 

170 _points(self.curve) 

171 else: 

172 _parts(self.curve) 

173 return self

176def _points(curve: tuple[tuple[float, float], ...] | dict[str, float]) -> None: 

177 if not isinstance(curve, tuple) or not 2 <= len(curve) <= MAX_CURVE: # noqa: PLR2004 - a line needs two points 

178 msg = f"a curve of a weather signal is 2 to {MAX_CURVE} points [x, y], e.g. [[0, 0], [1, 1]]" 

179 raise ValueError(msg) 

180 xs = [x for x, _ in curve] 

181 if any(not (0 <= value <= 1) for point in curve for value in point) or xs != sorted(set(xs)): 

182 msg = "a curve's points are in 0..1 and its x values rise, e.g. [[0, 0], [0.5, 0.8], [1, 1]]" 

183 raise ValueError(msg)

186def _parts(curve: tuple[tuple[float, float], ...] | dict[str, float]) -> None: 

187 if not isinstance(curve, dict) or not curve: 

188 msg = "a curve of daypart is a table of dawn, morning, day, dusk, evening and night, e.g. {morning = 1}"

256def audio_type(content: bytes, extension: str) -> str | None: 

257 """The media type of `content` if its signature matches `extension` (`ogg`, `opus`, `mp3`), else None. 

258 

259 `.ogg` and `.opus` start with `OggS`. `.mp3` starts with an ID3 tag or with two consecutive valid MPEG Layer III 

260 frame headers: one 11-bit sync word alone is too weak a signature. 

261 """ 

262 if extension in {"ogg", "opus"}: 

263 return _TYPES[extension] if content.startswith(_OGG) else None 

264 if extension == "mp3" and (content.startswith(_ID3) or _mp3_frames(content)): 

265 return _TYPES[extension] 

266 return None 

267 

268 

269def _mp3_frame_length(header: bytes) -> int | None: 

270 """The length of the MPEG Layer III frame that `header` (4 bytes) starts, or None if it isn't a valid header.""" 

271 if len(header) < _MP3_FRAME_HEADER or header[0] != 0xFF or header[1] & 0xE0 != 0xE0: # noqa: PLR2004 - sync word 

272 return None 

273 version, layer = (header[1] >> 3) & 3, (header[1] >> 1) & 3 

274 bitrate_index, rate_index, padding = header[2] >> 4, (header[2] >> 2) & 3, (header[2] >> 1) & 1 

275 if version not in _MP3_BITRATES or layer != 1 or bitrate_index in {0, 15} or rate_index == 3: # noqa: PLR2004 

276 return None 

277 bitrate = _MP3_BITRATES[version][bitrate_index] * 1000 

278 rate = _MP3_RATES[version][rate_index] 

279 return (144 if version == 3 else 72) * bitrate // rate + padding # noqa: PLR2004 - MPEG-1 has 1152 samples 

280 

281 

282def _mp3_frames(content: bytes) -> bool: 

283 first = _mp3_frame_length(content[:_MP3_FRAME_HEADER]) 

284 return first is not None and _mp3_frame_length(content[first : first + _MP3_FRAME_HEADER]) is not None

287def read_audio(directory: Path, asset: SoundAsset) -> tuple[bytes, str]: 

288 """The content of `asset` and its media type, checked as it is read, so a file swapped meanwhile is caught. 

289 

290 Raises: 

291 ValueError: If the file is missing, leads out of the directory, isn't a regular file, is larger than 2 MB, or 

292 its signature doesn't match its extension. 

293 """ 

294 where = f"Audio file {asset.file!r} of {directory}" 

295 if not re.fullmatch(AUDIO_FILE, asset.file): 

296 msg = f"{where} isn't a path inside the soundscape's directory ending in .ogg, .opus or .mp3" 

297 raise ValueError(msg) 

298 try: 

299 root = directory.resolve(strict=True) 

300 path = (root / asset.file).resolve(strict=True) 

301 except OSError as exc: 

302 msg = f"{where} doesn't exist; its path is relative to the soundscape file's directory" 

303 raise ValueError(msg) from exc 

304 if not path.is_relative_to(root): 

305 msg = f"{where} leads out of the soundscape's directory; keep audio files beside the soundscape file" 

306 raise ValueError(msg) 

307 if not path.is_file(): 

308 msg = f"{where} isn't a file" 

309 raise ValueError(msg) 

310 content = _read_regular(path, MAX_ASSET_BYTES, where) 

311 if len(content) > MAX_ASSET_BYTES: 

312 msg = f"{where} is larger than {MAX_ASSET_BYTES // 1024 // 1024} MB; shorten or compress it" 

313 raise ValueError(msg) 

314 media_type = audio_type(content, asset.file.rpartition(".")[2]) 

315 if media_type is None: 

316 msg = f"{where} isn't the audio its extension says: .ogg and .opus are Ogg files, .mp3 is MPEG audio" 

317 raise ValueError(msg) 

318 return content, media_type

329 def __post_init__(self) -> None: 

330 """Check the soundscape again, so one built with `model_construct` can't skip the checks, and its files. 

331 

332 Raises: 

333 ValueError: If the soundscape isn't valid, is called `off`, the let isn't named after it, or an audio 

334 file doesn't pass the checks of `read_audio`, or all of them are larger than 8 MB. 

335 """ 

336 try: 

337 soundscape = Soundscape.model_validate(self.soundscape.model_dump()) 

338 except ValidationError as exc: 

339 msg = f"Soundscape {self.soundscape.name!r} isn't valid: {problems_of(exc)}" 

340 raise ValueError(msg) from exc 

341 if soundscape.name == OFF: 

342 msg = f"No plugin can offer a soundscape named {OFF!r}: that is no sound" 

343 raise ValueError(msg) 

344 if self.name != soundscape.name: 

345 msg = f"A soundscape let is named after its soundscape: {soundscape.name!r}, not {self.name!r}" 

346 raise ValueError(msg) 

347 assets = soundscape.assets() 

348 if assets and self.directory is None: 

349 msg = f"Soundscape {soundscape.name!r} ships audio files, so it comes from a soundscape file" 

350 raise ValueError(msg) 

351 total = sum(len(read_audio(self.directory, asset)[0]) for asset in assets.values()) if self.directory else 0 

352 if total > MAX_SOUNDSCAPE_BYTES: 

353 msg = f"Soundscape {soundscape.name!r} ships more than {MAX_SOUNDSCAPE_BYTES // 1024 // 1024} MB of audio" 

354 raise ValueError(msg) 

355 object.__setattr__(self, "soundscape", soundscape)

357 def asset(self, part: str) -> tuple[bytes, str, str]: 

358 """The content, media type and extension of asset `part` (`layer-0`, `earcon-draft`), checked as it is read. 

359 

360 Raises: 

361 ValueError: If there is no such asset, or its file no longer passes the checks of `read_audio`. 

362 """ 

363 asset = self.soundscape.assets().get(part) 

364 if asset is None or self.directory is None: 

365 msg = f"Soundscape {self.soundscape.name!r} has no {part!r}" 

366 raise ValueError(msg) 

367 content, media_type = read_audio(self.directory, asset) 

368 return content, media_type, asset.file.rpartition(".")[2]

405def problems_of(exc: ValidationError) -> str: 

406 """Each problem of a soundscape as `field: rule`, in words a person editing a soundscape file reads.""" 

407 rules = { 

408 "name": "lowercase words joined by single dashes, e.g. rain", 

409 "file": "a path inside the soundscape's directory ending in .ogg, .opus or .mp3, e.g. sounds/rain.ogg", 

410 "video": "the video's id: 11 letters, digits, dashes or underscores, e.g. tRsQsTMvPNg", 

411 } 

412 problems: list[str] = [] 

413 for error in exc.errors(): 

414 where = ".".join(map(str, error["loc"])) or "soundscape" 

415 field = next((str(part) for part in reversed(error["loc"]) if isinstance(part, str)), "") 

416 if error["type"] == "string_pattern_mismatch": 

417 rule = rules.get(field, "lowercase words joined by single dashes, e.g. acme") 

418 else: 

419 rule = error["msg"].removeprefix("Value error, ") 

420 problems.append(f"{where}: {rule}") 

421 return "; ".join(problems)

424def sourced_soundscapes(source: Path | Soundscape) -> tuple[tuple[Soundscape, Path | None], ...]: 

425 """Each soundscape of `source` (one, a `*.toml` file of one, or a directory of them) with its file's directory. 

426 

427 Raises: 

428 ValueError: If a file isn't a valid soundscape, or a directory holds none. 

429 OSError: If a file can't be read. 

430 """ 

431 if isinstance(source, Soundscape): 

432 return ((source, None),) 

433 files = sorted(source.glob(f"*{SOUNDSCAPE_SUFFIX}")) if source.is_dir() else [source] 

434 if not files: 

435 msg = f"{source} holds no soundscape files (*{SOUNDSCAPE_SUFFIX})" 

436 raise ValueError(msg) 

437 found: list[tuple[Soundscape, Path | None]] = [] 

438 for file in files: 

439 if file.suffix != SOUNDSCAPE_SUFFIX: 

440 msg = f"Soundscape file {file} must end in {SOUNDSCAPE_SUFFIX}" 

441 raise ValueError(msg) 

442 try: 

443 found.append((Soundscape.model_validate(tomllib.loads(file.read_text(encoding="utf-8"))), file.parent)) 

444 except tomllib.TOMLDecodeError as exc: 

445 msg = f"Soundscape file {file} isn't valid TOML: {exc}" 

446 raise ValueError(msg) from exc 

447 except ValidationError as exc: 

448 msg = f"Soundscape file {file} isn't a valid soundscape: {problems_of(exc)}" 

449 raise ValueError(msg) from exc 

450 return tuple(found)
