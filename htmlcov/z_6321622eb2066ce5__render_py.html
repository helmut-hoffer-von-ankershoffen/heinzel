3from rich.console import Group, RenderableType

9def render_reported(reported: Reported) -> RenderableType: 

10 """The entry in one line, the log it went to, and what comes next.""" 

11 entry = reported.entry 

12 what = f"{entry.heinzel_id} {entry.verb}" + (f" {entry.target}" if entry.target else "") 

13 lines = [ 

14 Text(f"Reported: {what}"), 

15 Text(f" {entry.time.isoformat()} by {entry.user}, entry {reported.entries} of {reported.log}", style="dim"), 

16 ] 

17 return Group(*lines, *(Text(line) for line in reported.next))

20def render_synced(synced: Synced) -> RenderableType: 

21 """Each log copied with its new entries, what couldn't be, and the command that shares them.""" 

22 lines = [Text(f"Into {synced.checkout}:")] 

23 lines += [Text(f" {log.heinzel_id}: {log.entries} entries, {log.new} new ({log.path})") for log in synced.logs] 

24 if not synced.logs and not synced.problems: 

25 lines.append(Text(" nothing to copy yet; report first: hz report <id> heartbeat")) 

26 lines += [Text(f" not copied: {problem}", style="yellow") for problem in synced.problems] 

27 return Group(*lines, *(Text(line) for line in synced.next))
