30@module.operation(render=render_release) 

31def release(service: SystemService, installation: Installation) -> ReleaseCheck: 

32 """Ask where Heinzelmännchen was installed from for its latest release, and whether to update. 

33 

34 One unauthenticated GET to the GitHub releases API of the repository in the uv tool receipt (or the default 

35 repository) with only a `heinzel/<version>` User-Agent (ADR 0031); offline or rate-limited, the result carries a 

36 warning. Always asks, even when the hourly check (setting `update_check`) asked a minute ago; the answer is 

37 cached for the update notice. 

38 """ 

39 requirements = installation.requirements() 

40 return service.release(requirements[0] if requirements else None)

43@module.operation(read_only=False) 

44def update( 

45 installation: Installation, 

46 source: Annotated[ 

47 str | None, 

48 Field(description="Install from here instead, e.g. git+https://github.com/org/heinzel@v1.2.0 or a wheel."), 

49 ] = None, 

50) -> str: 

51 """Update Heinzelmännchen, keeping plugins; see what it brings with `hz system changelog --newer`. 

52 

53 Shows the version now installed. 

54 

55 Raises: 

56 HeinzelError: If Heinzelmännchen wasn't installed with `uv tool install`, or the update fails. 

57 """ 

58 requirements = installation.requirements() 

59 if requirements is None: 

60 msg = "Heinzelmännchen wasn't installed with `uv tool install`; update it the way you installed it." 

61 raise HeinzelError(msg) 

62 command = ( 

63 installation.upgrade_command() 

64 if source is None 

65 else installation.tool_install_command(source, requirements[1:]) 

66 ) 

67 installation.run(command, "Failed to update Heinzelmännchen.", "Updating Heinzelmännchen") 

68 return installation.hz("--version").splitlines()[0]
