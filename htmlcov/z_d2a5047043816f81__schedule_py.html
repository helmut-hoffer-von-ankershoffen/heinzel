1"""Schedule arithmetic: cron slots in a window, through croniter rather than by hand (ADR 0020). 

2 

3Cron expressions mean wall-clock time in the user's time zone, so slots are computed on naive local times and then 

4placed in the real zone (a `ZoneInfo`, never a fixed offset, which would be wrong for half the year). Daylight saving 

5time follows two rules: 

6 

7- A wall-clock time that happens twice when the clocks go back runs once, at its first occurrence. 

8- A wall-clock time that doesn't exist when the clocks go forward runs shifted by the gap: 02:30 becomes 03:30. 

9""" 

10 

11import os 

12from datetime import UTC, datetime, timedelta, tzinfo 

13from pathlib import Path 

14from zoneinfo import ZoneInfo, ZoneInfoNotFoundError 

15 

16from croniter import CroniterBadDateError, croniter 

17 

18MAX_SLOTS_PER_TICK = 100 

19"""`catch_up=all` replays at most this many slots per tick, oldest first; the next tick continues.""" 

20FIELDS = 5 

21LOCALTIME = Path("/etc/localtime") 

22 

23 

24def local_zone(localtime: Path = LOCALTIME) -> tzinfo: 

25 """The user's time zone with its DST rules: `TZ` if set, else what /etc/localtime points at. 

26 

27 Falls back to the current fixed offset only when neither names a zone, e.g. in a bare container. 

28 """ 

29 name = os.environ.get("TZ", "").removeprefix(":") 

30 if name: 

31 try: 

32 if not name.startswith("/"): 

33 return ZoneInfo(name) 

34 with Path(name).open("rb") as handle: 

35 return ZoneInfo.from_file(handle, key=name) 

36 except (OSError, ValueError, ZoneInfoNotFoundError): 

37 pass 

38 resolved = localtime.resolve() 

39 if "zoneinfo" in resolved.parts: 

40 key = "/".join(resolved.parts[resolved.parts.index("zoneinfo") + 1 :]) 

41 try: 

42 return ZoneInfo(key) 

43 except (ValueError, ZoneInfoNotFoundError): 

44 pass 

45 if localtime.is_file(): 

46 with localtime.open("rb") as handle: 

47 return ZoneInfo.from_file(handle, key="localtime") 

48 fixed = datetime.now(UTC).astimezone().tzinfo 

49 return fixed if fixed is not None else UTC

52def problem(schedule: str) -> str | None: 

53 """Why a cron expression can't be used, or None if it can.""" 

54 if not (schedule.startswith("@") or len(schedule.split()) == FIELDS) or not croniter.is_valid(schedule): 

55 return f"invalid schedule {schedule!r}: use five cron fields (minute hour day month weekday) or @daily" 

56 try: 

57 croniter(schedule, datetime(2000, 1, 1)).get_next(datetime) # noqa: DTZ001 - wall-clock arithmetic is naive 

58 except CroniterBadDateError: 

59 return f"invalid schedule {schedule!r}: it never fires" 

60 return None

63def _place(naive: datetime, zone: tzinfo) -> datetime: 

64 """A naive wall-clock time in `zone`: the first occurrence of a repeated time, a missing one shifted by the gap.""" 

65 return naive.replace(tzinfo=zone, fold=0).astimezone(UTC).astimezone(zone) 

66 

67 

68def _naive(moment: datetime, zone: tzinfo) -> datetime: 

69 return moment.astimezone(zone).replace(tzinfo=None) 

70 

71 

72def slots( 

73 schedule: str, after: datetime, until: datetime, zone: tzinfo, limit: int = MAX_SLOTS_PER_TICK 

74) -> list[datetime]: 

75 """Slots `s` with `after < s <= until`, oldest first, at most `limit`, each wall-clock time once.""" 

76 found: list[datetime] = [] 

77 iterator = croniter(schedule, _naive(after, zone)) 

78 while len(found) < limit: 

79 slot = _place(iterator.get_next(datetime), zone) 

80 if slot > until: 

81 break 

82 if slot > after and (not found or slot > found[-1]): 

83 found.append(slot) 

84 return found

87def latest(schedule: str, after: datetime, until: datetime, zone: tzinfo) -> datetime | None: 

88 """The most recent slot with `after < slot <= until`, or None.""" 

89 # croniter's get_prev excludes its start, so start just past `until` to include a slot that falls on it. 

90 iterator = croniter(schedule, _naive(until, zone) + timedelta(seconds=1)) 

91 slot = _place(iterator.get_prev(datetime), zone) 

92 while slot > until: # a slot just before a gap can land after `until` once shifted 

93 slot = _place(iterator.get_prev(datetime), zone) 

94 return slot if slot > after else None

97def next_slot(schedule: str, now: datetime, zone: tzinfo) -> datetime: 

98 """The first slot after `now`.""" 

99 iterator = croniter(schedule, _naive(now, zone)) 

100 slot = _place(iterator.get_next(datetime), zone) 

101 while slot <= now: 101 ↛ 102line 101 didn't jump to line 102 because the condition on line 101 was never true

102 slot = _place(iterator.get_next(datetime), zone) 

103 return slot
